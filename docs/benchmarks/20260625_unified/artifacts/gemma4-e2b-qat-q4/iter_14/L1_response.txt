class JobQueue:
    """
    A basic job queue implementation following FIFO principles.
    Note: For this basic structure, job processing (state change) is simulated 
    within the system, allowing get_result to retrieve a value.
    """
    def __init__(self):
        # Stores pending jobs as a list (FIFO)
        self._queue = []
        # Stores results keyed by job_id once processed
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Add a job to the queue.
        """
        job = {"job_id": job_id, "data": data, "completed": False}
        self._queue.append(job)
        return job_id

    def process_job(self, job_id: str) -> dict | None:
        """
        Simulates processing a job from the queue.
        In a real system, this would run the actual task.
        """
        job_index = -1
        
        # Find the job and remove it from the queue (FIFO)
        for i, job in enumerate(self._queue):
            if job["job_id"] == job_id:
                job_index = i
                job = self._queue.pop(i)
                break

        if job_index != -1:
            job_id = job["job_id"]
            data = job["data"]
            
            # Simulate processing logic
            # In a real application, this is where the computation happens
            # For demonstration, we use the task data as the result
            result_data = {"status": "completed", "task_data": data.get("task"), "input": data}
            
            self._results[job_id] = result_data
            return result_data
        
        return None

    def get_result(self, job_id: str) -> dict | None:
        """
        Get the result of a completed job.
        """
        return self._results.get(job_id)

if __name__ == "__main__":
    queue = JobQueue()
    
    # 1. Add job
    job_id_1 = queue.add_job("job1", {"task": "process_data"})
    job_id_2 = queue.add_job("job2", {"task": "calculate_sum"})
    
    print(f"Added job 1 with ID: {job_id_1}")
    print(f"Added job 2 with ID: {job_id_2}\n")

    # 2. Process job 1 (FIFO)
    print("Processing job 1...")
    result_1 = queue.process_job(job_id_1)
    if result_1:
        print(f"Job 1 Result: {result_1}\n")
    
    # 3. Process job 2
    print("Processing job 2...")
    result_2 = queue.process_job(job_id_2)
    if result_2:
        print(f"Job 2 Result: {result_2}\n")

    # 4. Get result directly (demonstrates get_result functionality)

from collections import deque
from typing import Dict, Optional, List

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._data = {}

    def add_job(self, job_id: str, data: Dict) -> str:
        if job_id in self._data:
            raise ValueError(f"Job {job_id} already exists")
        self._queue.append(job_id)
        self._data[job_id] = data
        return job_id

    def get_result(self, job_id: str) -> Optional[Dict]:
        # If job not known, return None
        if job_id not in self._data:
            return None

        # Process any jobs that arrived before the requested one
        while self._queue and self._queue[0] != job_id:
            front_id = self._queue.popleft()
            # Remove from map as it's considered completed
            del self._data[front_id]

        # Now the job should be at front
        if not self._queue or self._queue[0] != job_id:
            # Job ID not in queue (maybe it was added then removed? Or not in order)
            return None

        # Process the target job
        self._queue.popleft()
        result = self._data.pop(job_id)
        return result

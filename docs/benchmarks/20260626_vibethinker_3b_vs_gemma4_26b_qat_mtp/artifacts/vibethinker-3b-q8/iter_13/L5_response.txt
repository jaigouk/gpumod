class JobQueue:
    def __init__(self):
        self._jobs = {}  # id -> Job
        self._priority_map = {}  # priority -> deque of job ids (or jobs)
        self._counter = 0

    def add_job(self, job_id: str, data: dict, priority: int = 0) -> None:
        job = Job(id=job_id, data=data, priority=priority)
        self._jobs[job_id] = job
        if priority not in self._priority_map:
            self._priority_map[priority] = deque()
            if not self._max_priority or priority > self._max_priority:
                self._max_priority = priority
        else:
            self._priority_map[priority].append(job)
            if priority > self._max_priority:
                self._max_priority = priority
        self._counter += 1

    def get_next_job(self) -> Optional[Tuple[str, dict]]:
        if self._max_priority is None:
            return None
        # get the job at the front of the deque
        job = self._priority_map[self._max_priority].popleft()
        # If the deque is empty after popleft, delete it and update max_priority
        if not self._priority_map[self._max_priority]:
            del self._priority_map[self._max_priority]
            # new max priority is max of remaining keys, or None
            self._max_priority = max(self._priority_map.keys()) if self._priority_map else None
        # Remove from global job dict (optional)
        del self._jobs[job.id]
        return (job.id, job.data)

        self._jobs = []  # list of Job

    def add_job(self, job_id: str, data: dict, priority: int = 0) -> None:

    def get_next_job(self) -> Optional[Tuple[str, dict]]:
        if not self._jobs:
            return None
        # Find index of job with highest priority (and earliest if tie)
        max_index = 0
        max_priority = self._jobs[0].priority
        for i, job in enumerate(self._jobs):

                max_index = i

import heapq
from typing import List, Tuple, Dict

class JobQueue:
    def __init__(self) -> None:
        self._heap: List[Tuple[int, int, str, Dict]] = []
        self._counter: int = 0

    def add_job(self, job_name: str, job_data: Dict, priority: int = 0) -> None:
        """Add a job; higher ``priority`` values mean higher priority."""
        self._counter += 1  # maintain FIFO order for same priority
        # Negative priority makes higher numeric priority appear first in a min‑heap
        heapq.heappush(self._heap, (-priority, self._counter, job_name, job_data))

    def get_next_job(self) -> Tuple[str, Dict] | None:
        """Return the highest‑priority job (name and data), or None if empty."""
        if not self._heap:
            return None
        _, _, job_name, job_data = heapq.heappop(self._heap)
        return job_name, job_data
```
   
This implementation uses a min‑heap where the priority is stored as a negative value so that larger priority numbers (e.g., 2 > 1 > 0) are extracted first. The counter provides the FIFO tie‑breaker for jobs with identical priority. The `add_job` method accepts an optional `priority` argument (defaulting to 0), and `get_next_job` returns a tuple of the job name and its data, or `None` if the queue is empty.

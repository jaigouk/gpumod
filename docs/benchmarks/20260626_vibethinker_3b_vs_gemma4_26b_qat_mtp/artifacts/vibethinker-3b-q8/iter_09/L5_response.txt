import heapq

@dataclass
class Job:
    id: str
    data: Dict[str, Any]
    priority: int = 0
    retries: int = 0

class RetryPolicy:
    def __init__(self, max_attempts: int = 4):
        self.max_attempts = max_attempts

    def run(self, fn: Callable, data: Any) -> Tuple[bool, int]:
        attempt = 0
        while attempt < self.max_attempts:
            try:
                fn(data)
                # success if call completes without exception
                return (True, attempt + 1)
            except Exception:
                attempt += 1
                continue
        # Exceeded attempts
        return (False, self.max_attempts)

class JobQueue:
    def __init__(self):
        self._heap: List[Tuple[int, int, Any]] = []  # (-priority, index, job)
        self._index: int = 0
        self._retry_policy = RetryPolicy()

    def add_job(self, job_id: str, data: Dict[str, Any], priority: int = 0) -> None:
        job = Job(id=job_id, data=data, priority=priority, retries=0)
        self._heap.append((-priority, self._index, job))
        heapq.heapify(self._heap)
        self._index += 1

    def process_job(self, job_id: str, processor: Callable[[Dict[str, Any]], Any]) -> bool:
        if not self._heap:
            return False
        # pop the highest priority job
        _, _, job = heapq.heappop(self._heap)
        success, attempts = self._retry_policy.run(processor, job.data)
        # Optionally store or raise on failure; just return success per spec
        return success

    def get_next_job(self) -> Optional[Tuple[str, Dict[str, Any]]]:
        if not self._heap:
            return None
        # Peek at the highest priority job
        _, _, job = self._heap[0]
        return (job.id, job.data)

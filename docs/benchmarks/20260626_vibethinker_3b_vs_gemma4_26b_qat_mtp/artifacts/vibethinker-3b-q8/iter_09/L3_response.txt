class JobQueue:
    def __init__(self):
        self.deques = [collections.deque() for _ in range(3)]

    def add_job(self, job_id, job_data, priority=0):
        # optionally validate priority
        if priority not in (0,1,2):
            raise ValueError("Priority must be 0,1,2")
        self.deques[priority].append((job_id, job_data))

    def get_next_job(self):
        for p in range(2, -1, -1):
            if self.deques[p]:
                job_id, job_data = self.deques[p].popleft()
                return (job_id, job_data)
        return None

import collections

class JobQueue:
    def __init__(self):
        # indices 0,1,2 for priority levels
        self._deques = [collections.deque() for _ in range(3)]

    def add_job(self, job_id, job_data, priority=0):
        """Add a job to the queue.
        Parameters
        ----------
        job_id : str
            Unique identifier for the job.
        job_data : dict
            Data associated with the job.
        priority : int, optional
            Priority level (0=normal, 1=high, 2=critical). Default is 0.
        """
        if priority not in (0, 1, 2):
            raise ValueError("Priority must be 0, 1, or 2")
        self._deques[priority].append((job_id, job_data))

    def get_next_job(self):
        """Return the highest‑priority job that is waiting.
        Returns a tuple (job_id, job_data) or None if the queue is empty.
        """
        for p in range(2, -1, -1):
            if self._deques[p]:
                job_id, job_data = self._deques[p].popleft()
                return (job_id, job_data)
        return None

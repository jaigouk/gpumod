from collections import deque
from typing import Dict, Tuple, Optional

class JobQueue:
    """Simple priority queue for jobs with FIFO order within the same priority."""
    _PRIORITY_LIMIT = 2  # critical = 2, high = 1, normal = 0

    def __init__(self) -> None:
        # One deque per priority level (0 .. _PRIORITY_LIMIT)
        self._deques: list[deque[Tuple[str, Dict]]] = [deque() for _ in range(self._PRIORITY_LIMIT + 1)]

    def add_job(self, job_name: str, job_data: Dict, priority: int = 0) -> None:
        """Add a job to the queue.

        Parameters
        ----------
        job_name : str
            Identifier for the job.
        job_data : dict
            Data associated with the job.
        priority : int, optional
            Prioritization level. Default is 0 (normal). Must be in 0..2.
        """
        if not 0 <= priority <= self._PRIORITY_LIMIT:
            raise ValueError(f"Priority must be between 0 and {self._PRIORITY_LIMIT}, got {priority}")
        self._deques[priority].append((job_name, job_data))

    def get_next_job(self) -> Tuple[str, Dict] | None:
        """Return the highest‑priority job (critical > high > normal) or None if empty.

        Returns
        -------
        tuple[str, dict] | None
        """
        for p in range(self._PRIORITY_LIMIT, -1, -1):
            if self._deques[p]:
                job_name, job_data = self._deques[p].popleft()
                return job_name, job_data
        return None

    _MAX_PRIORITY = 2

        # One deque per priority level (0 .. _MAX_PRIORITY)

            deque() for _ in range(self._MAX priority + 1)

    def add_job(self, job_name: str, job_data: Dict, priority: int = 0) -> None:
        """Add a job.

        Args:
            job_name (str): Identifier for the job.
            job_data (dict): Data associated with the job.

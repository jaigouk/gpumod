from dataclasses import dataclass
import heapq

@dataclass
class Job:
    id: str
    data: Dict[str, Any]
    priority: int = 0
    retries: int = 0

class RetryPolicy:
    def __init__(self, max_attempts: int = 4):
        self.max_attempts = max_attempts

    def run(self, fn: Callable[[], Any], data: Dict[str, Any]) -> Tuple[bool, int]:
        attempts = 0
        while attempts < self.max_attempts:
            attempts += 1
            try:
                fn(data)
                return True, attempts
            except Exception as e:
                # Don't actually sleep; just continue
                pass
        # Exceeded max attempts
        return False, self.max_attempts

class JobQueue:
    def __init__(self):
        self._heap: List[Tuple[int, int, Job]] = []
        self._counter: int = 0

    def add_job(self, job_id: str, data: Dict[str, Any], priority: int = 0) -> None:
        job = Job(id=job_id, data=data, priority=priority, retries=0)
        heapq.heappush(self._heap, (-priority, self._counter, job))
        self._counter += 1

    def process_job(self, job_id: str, processor: Callable[[Dict[str, Any]], Any]) -> bool:
        # If the queue is empty, nothing to process
        if not self._heap:
            return False
        # Pop the next job (highest priority, FIFO for same priority)
        _, _, job = heapq.heappop(self._heap)
        # Determine the max number of attempts based on the job's retry count
        max_attempts = job.retries + 1  # initial attempt plus retries
        policy = RetryPolicy(max_attempts)
        success, attempts = policy.run(processor, job.data)
        # Optionally you could log attempts, but not required
        return success

    def get_next_job(self) -> Optional[Tuple[str, Dict[str, Any]]]:
        if not self._heap:
            return None
        neg_priority, _, job = self._heap[0]
        return (job.id, job.data)

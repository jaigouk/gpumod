from collections import deque

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        # Ensure unique job_id
        if job_id in self._results or job_id in self._queue:
            raise ValueError(f"Job {job_id} cannot be added; already exists.")
        self._queue.append((job_id, data))
        return job_id

    def get_result(self, job_id: str):
        # If already processed, return result
        if job_id in self._results:
            return self._results[job_id]

        # Process jobs in FIFO order until we find the requested job
        while self._queue:
            pid, data = self._queue.popleft()
            # Process the job: for this basic implementation, we return the data as result
            result = data  # simple identity
            self._results[pid] = result
            if pid == job_id:
                return result
        # Job not found; return None
        return None

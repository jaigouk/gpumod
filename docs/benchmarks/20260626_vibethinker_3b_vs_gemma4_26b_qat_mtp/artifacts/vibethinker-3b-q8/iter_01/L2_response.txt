from typing import Callable, Dict

class JobQueue:
    def __init__(self):
        self._jobs = {}  # job_id -> data dict
        self._retries = {}  # job_id -> attempts made (0-4)

    def add_job(self, job_id: str, data: Dict) -> None:
        self._jobs[job_id] = data
        self._retries[job_id] = 0

    def process_job(self, job_id: str, processor: Callable[[Dict], Callable[None, bool]]) -> bool:
        data = self._jobs.get(job_id)
        if data is None:
            raise KeyError(f"Job {job_id} not found")
        attempts = self._retries.get(job_id, 0)
        while attempts < 4:
            # Determine exponential backoff delay for the next attempt
            next_attempt = attempts + 1
            # delays list for attempts 2,3,4
            if next_attempt > 1:
                # index: next_attempt-2 -> 0:1, 1:2, 2:4
                delay = [1, 2, 4][next_attempt - 2]
            else:
                delay = 0

            # Record the delay (if any) in the data dict
            if 'delays' not in data:
                data['delays'] = []
            if delay:
                data['delays'].append(delay)

            try:
                # Pass the possibly augmented data to the processor
                success = processor(data)
                return True
            except Exception:
                # Failure: increment attempt count and loop for retry
                attempts += 1
                # Do not sleep; the delay has already been recorded in data
        # All four attempts failed
        self._retries[job_id] = 4
        return False

class JobQueue:
    def __init__(self):
        self._jobs: Dict[str, Dict] = {}
        self._retries: Dict[str, int] = {}

    def add_job(self, job_id: str, data: Dict) -> None:
        self._jobs[job_id] = data
        self._retries[job_id] = 0

    def process_job(self, job_id: str, processor: Callable[[Dict], Callable[None, bool]]) -> bool:
        data = self._jobs.get(job_id)
        if data is None:
            raise KeyError(f"Job {job_id} not found")
        attempts = self._retries.get(job_id, 0)

        while attempts < 4:
            # Determine the backoff delay for the next attempt
            next_attempt = attempts + 1
            if next_attempt > 1:
                # Exponential backoff: 1, 2, 4 seconds
                delay = [1, 2, 4][next_attempt - 2]
            else:

            # Record the delay in the data dict (if non‑zero)

                data['delays'].append(delay)

            try:
                # Pass the (possibly augmented) data to the processor
                _ = processor(data)
                return True
            except Exception:
                # Failure – increment attempt counter and loop for retry
                attempts += 1
                self._retries[job_id] = attempts
                # No actual sleep; the delay has already been captured in `data`
        # Four attempts exhausted
        self._retries[job_id] = 4

from typing import Callable, Dict

class JobQueue:
    def __init__(self):
        # Stores the data for each job and the number of attempts already made
        self._jobs: Dict[str, Dict] = {}
        self._retries: Dict[str, int] = {}

    def add_job(self, job_id: str, data: Dict) -> None:
        """Add a job to be processed."""
        self._jobs[job_id] = data
        self._retries[job_id] = 0

    def process_job(self, job_id: str, processor: Callable)[None] -> bool:
        """
        Process a job with up to 4 attempts, using exponential backoff to
        simulate delays by recording them in the data dictionary.
        Returns True on first successful execution, False if all attempts fail.
        """
        data = self._jobs.get(job_id)
        if data is None:
            raise KeyError(f"Job {job_id} not found")

        attempts = self._retries.get(job_id, 0)

        while attempts < 4:
            # Determine exponential backoff delay for the next attempt (seconds)
            next_attempt = attempts + 1
            if next_attempt > 1:
                # Delays: 1, 2, 4 seconds for attempts 2, 3, 4
                delay = [1, 2, 4][next_attempt - 2]
            else:
                delay = 0

            # Record the delay in the data dict (if non‑zero)
            if 'delays' not in data:
                data['delays'] = []
            if delay:
                data['delays'].append(delay)

            try:
                # Execute the processor; success is defined as no exception raised
                processor(data)
                return True
            except Exception:
                # Failure – increment attempt count and repeat
                attempts += 1
                self._retries[job_id] = attempts
                # No actual sleep; the delay has been recorded in `data`

        # All four attempts exhausted
        self._retries[job_id] = 4
        return False

from typing import Callable, Dict, Any, Deque
import collections

class JobQueue:
    def __init__(self) -> None:
        self.queue: Deque[tuple[str, Dict[str, Any]]] = collections.deque()
        self.data_by_id: Dict[str, Dict[str, Any]] = {}
        self.retry_counts: Dict[str, int] = {}
        self.delay_log: Dict[str, Dict[int, float]] = {}

    def add_job(self, job_id: str, data: Dict[str, Any]) -> None:
        self.data_by_id[job_id] = data
        self.queue.append((job_id, data))
        self.retry_counts[job_id] = 0  # zero attempts made yet

    def process_job(self, job_id: str, processor: Callable[[Dict[str, Any]], Any]) -> bool:
        data = self.data_by_id.get(job_id)
        if not data:
            return False

        attempts_made = self.retry_counts.get(job_id, 0)
        if attempts_made >= 4:
            return False

        attempts = attempts_made
        delay = 1.0  # first backoff delay

        while attempts < 4:
            attempt_no = attempts + 1
            # embed current delay into the data for this attempt (only for attempts after the first)
            attempt_data = data.copy()
            if attempt_no > 1:
                attempt_data['delay'] = delay

            try:
                # Process the job; any exception will trigger retry
                result = processor(attempt_data)
                # Success – mark the job as processed and clear stored data
                self.retry_counts[job_id] = attempt_no
                del self.data_by_id[job_id]
                # Remove from queue (linear scan for simplicity)
                for i, item in enumerate(self.queue):
                    if item[0] == job_id:
                        self.queue.pop(i)
                        break
                return True
            except Exception:
                # Record that we tried with the current delay
                if job_id not in self.delay_log:
                    self.delay_log[job_id] = {}
                self.delay_log[job_id][attempt_no] = delay
                # Move to the next attempt
                attempts += 1
                delay *= 2
                # If we hit the fourth attempt, the loop condition will stop after this iteration

        # Exhausted all allowed attempts
        self.retry_counts[job_id] = attempts
        # Optionally log the final attempt details (if not already logged)
        if job_id in self.delay_log:
            self.delay_log[job_id][attempts] = delay
        # Clear the job data as a convention (already not successful)
        del self.data_by_id[job_id]
        return False

        self.data_by_id: Dict[str, Dict[str, Any]] = {}

    def process_job(self, job_id: str, processor: Callable[[Dict[str, Any]], Any]) -> bool:

        if not data:
            return False

        attempts_made = self.retry_counts.get(job_id, 0)
        if attempts_made >= 4:
            return False

        # Iterate from current attempts to a maximum of 4 attempts total
        for i in range(attempts_made, 4):
            attempt_no = i + 1  # 1-indexed attempt number
            # Compute exponential backoff delay for this attempt (skip for first attempt)
            delay = 2**(attempt_no - 2) if attempt_no > 1 else 0

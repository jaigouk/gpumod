```python
"""
Core job data structures and basic FIFO queue operations.
"""
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, List, Optional


@dataclass
class Job:
    """Represents a unit of work in the queue."""
    id: str
    payload: Any
    status: str = "pending"
    created_at: datetime = field(default_factory=datetime.utcnow)
    retries: int = 0


class JobQueue:
    """A basic FIFO job queue."""
    def __init__(self):
        self._jobs: List[Job] = []

    def add(self, job: Job) -> None:
        """Enqueue a job."""
        self._jobs.append(job)

    def get_next(self) -> Optional[Job]:
        """Dequeue and return the next pending job."""
        if not self._jobs:
            return None
        return self._jobs.pop(0)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 📄 `queue/priority.py`
```python
"""
Priority-based job scheduling using a min-heap.
"""
import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Manages jobs where lower priority values are processed first."""
    def __init__(self):
        self._heap: list = []
        self._counter = 0  # Tie-breaker to preserve FIFO order for equal priorities

    def add(self, job: Job, priority: int) -> None:
        """Add a job with a given priority (lower number = higher priority)."""
        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def get_next(self) -> Optional[Job]:
        """Retrieve and remove the highest-priority job."""
        if self.is_empty():
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def is_empty(self) -> bool:
        return len(self._heap) == 0

    def __len__(self) -> int:
        return len(self._heap)
```

### 📄 `queue/retry.py`
```python
"""
Retry logic with exponential backoff for job processing.
"""
import time
import functools
from typing import Callable, Any


def process_with_retry(max_retries: int = 3, base_delay: float = 1.0):
    """
    Decorator that retries a job processing function with exponential backoff.
    
    Args:
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
    """
    def decorator(func: Callable[..., Any]) -> Callable[..., Any]:
        @functools.wraps(func)
        def wrapper(job, *args, **kwargs) -> Any:
            last_exception = None
            for attempt in range(max_retries + 1):
                try:
                    return func(job, *args, **kwargs)
                except Exception as e:
                    last_exception = e
                    job.retries += 1
                    if attempt < max_retries:
                        delay = base_delay * (2 ** attempt)
                        time.sleep(delay)
                else:
                    job.status = "completed"
                    break
            else:
                job.status = "failed"
                raise last_exception
        return wrapper
    return decorator
```

### 🔑 Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py`: Data modeling & basic FIFO queue
   - `priority.py`: Heap-based scheduling logic
   - `retry.py`: Fault tolerance & backoff strategy
   - `__init__.py`: Public API surface
2. **Relative Imports**: All cross-module imports use `.` notation (`from .core import Job`) to avoid namespace collisions and ensure the package works correctly when installed or imported.
3. **State Tracking**: The retry decorator updates `job.retries` and `job.status` automatically, keeping job state synchronized with processing outcomes.
4. **Priority Tie-Breaking**: Uses a monotonic counter in the heap tuple to guarantee FIFO behavior when multiple jobs share the same priority level.

### `queue/core.py`
```python
"""Core data model and basic FIFO queue implementation."""

from dataclasses import dataclass, field
from typing import Any, Optional
import uuid


@dataclass
class Job:
    """Represents a unit of work."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    status: str = "pending"  # pending | processing | completed | failed
    priority: int = 0
    retries: int = 0
    max_retries: int = 3


class JobQueue:
    """Base FIFO queue implementation."""

    def __init__(self):
        self._queue: list[Job] = []
        self._processing: set[str] = set()

    def add(self, job: Job) -> None:
        """Add a job to the queue."""
        self._queue.append(job)

    def get_next(self) -> Optional[Job]:
        """Retrieve and mark the next job for processing."""
        if not self._queue:
            return None
        job = self._queue.pop(0)
        job.status = "processing"
        self._processing.add(job.id)
        return job

    def mark_completed(self, job: Job) -> None:
        """Mark a job as successfully completed."""
        self._processing.discard(job.id)
        job.status = "completed"

    def mark_failed(self, job: Job) -> None:
        """Mark a job as failed and requeue if retries remain."""
        self._processing.discard(job.id)
        job.status = "failed"
        if job.retries < job.max_retries:
            self._queue.append(job)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

import heapq
from typing import Optional

from .core import Job, JobQueue


class PriorityQueue(JobQueue):
    """Queue that processes jobs with lower priority numbers first."""

    def __init__(self):
        super().__init__()
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Tie-breaker for FIFO ordering at same priority

    def add(self, job: Job) -> None:
        """Add a job to the priority queue."""
        # (priority, insertion_order, job)
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def get_next(self) -> Optional[Job]:
        """Retrieve and mark the highest priority job for processing."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        job.status = "processing"
        self._processing.add(job.id)
        return job
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

import time
import random
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    handler: Callable[[Job], Any],
    max_retries: Optional[int] = None,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
) -> Any:
    """Execute a job handler with exponential backoff retry logic.

    Args:
        job: The job to process.
        handler: Callable that executes the job logic.
        max_retries: Maximum number of retry attempts. Defaults to job.max_retries.
        base_delay: Base delay in seconds for backoff calculation.
        max_delay: Maximum delay cap in seconds.

    Returns:
        The result of the handler.

    Raises:
        The last exception raised by the handler if all retries are exhausted.
    """
    retries = max_retries if max_retries is not None else job.max_retries

    for attempt in range(retries + 1):
        try:
            return handler(job)
        except Exception as e:
            job.retries += 1
            if job.retries >= retries:
                raise
            # Exponential backoff: base_delay * 2^(attempt)
            delay = min(base_delay * (2 ** (job.retries - 1)), max_delay)
            # Add jitter to prevent thundering herd
            jitter = random.uniform(0, delay * 0.1)
            time.sleep(delay + jitter)
```

### Key Design Choices:
1. **Relative Imports**: `from .core import ...` ensures the package works correctly when installed or run as a module, avoiding circular import issues.
2. **Inheritance & Polymorphism**: `PriorityQueue` extends `JobQueue`, reusing `mark_completed`/`mark_failed` while overriding `add` and `get_next` for heap-based ordering.
3. **Type Safety**: Full type hints with `typing` module for better IDE support and static analysis.
4. **Jitter in Backoff**: Prevents synchronized retries across distributed workers (`random.uniform(0, delay * 0.1)`).
5. **Clean `__init__.py`**: Acts as a facade, exposing only the public API while keeping internal modules private. `__all__` explicitly defines the public interface.

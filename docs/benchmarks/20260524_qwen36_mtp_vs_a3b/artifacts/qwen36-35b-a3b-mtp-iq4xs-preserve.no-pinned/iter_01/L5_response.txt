### 📁 `queue/core.py`
```python
"""Core job data model and basic FIFO queue operations."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    priority: int = 0
    status: str = "pending"
    retry_count: int = 0
    created_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    last_updated: datetime = field(default_factory=lambda: datetime.now(timezone.utc))

    def __post_init__(self):
        # Ensure status is always valid
        if self.status not in ("pending", "processing", "completed", "failed"):
            raise ValueError("Invalid job status")


class JobQueue:
    """Basic FIFO queue for job management."""

    def __init__(self):
        self._items: deque[Job] = deque()

    def add(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._items.append(job)

    def get(self) -> Optional[Job]:
        """Remove and return the next job, or None if empty."""
        return self._items.popleft() if self._items else None

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._items) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._items)
```

### 📁 `queue/priority.py`
```python
"""Priority queue implementation using a binary heap."""

from __future__ import annotations

import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Priority-based queue where lower numeric priority = higher precedence."""

    def __init__(self):
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Stable ordering for equal priorities

    def add(self, job: Job) -> None:
        """Add a job with its priority to the heap."""
        entry = (job.priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def get(self) -> Optional[Job]:
        """Remove and return the highest priority job, or None if empty."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)
```

### 📁 `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

from __future__ import annotations

import random
import time
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    jitter: bool = True,
) -> Any:
    """
    Execute a job processor with exponential backoff retry logic.

    Args:
        job: The job to process.
        processor: Callable that executes the job's payload.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds.
        max_delay: Maximum delay cap in seconds.
        jitter: Whether to add random jitter to prevent thundering herd.

    Returns:
        The result of the processor.

    Raises:
        Exception: The last exception if all retries are exhausted.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(max_retries + 1):
        try:
            result = processor(job)
            job.status = "completed"
            job.last_updated = datetime.now(timezone.utc)
            return result
        except Exception as e:
            last_exception = e
            job.retry_count += 1
            job.status = "failed"
            job.last_updated = datetime.now(timezone.utc)

            if attempt < max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                if jitter:
                    delay *= random.uniform(0.5, 1.5)
                time.sleep(delay)

    raise last_exception from last_exception
```

### `queue/core.py`
```python
"""Core job data structures and basic FIFO queue operations."""
from dataclasses import dataclass, field
from typing import Any, Callable, Optional
import time


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    job_id: str
    task: Callable[..., Any]
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    result: Any = None
    error: Optional[Exception] = None


class JobQueue:
    """Basic FIFO job queue implementation."""
    
    def __init__(self) -> None:
        self._queue: list[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)
        job.status = "queued"

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job from the queue."""
        if self._queue:
            job = self._queue.pop(0)
            job.status = "running"
            return job
        return None

    def size(self) -> int:
        """Return the number of jobs waiting in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""Priority-based job queue using a min-heap."""
import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Job queue that processes higher-priority jobs first."""
    
    def __init__(self) -> None:
        self._queue: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures stable sorting for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority (lower number = higher priority)."""
        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._queue, entry)
        job.status = "queued"

    def dequeue(self) -> Job:
        """Remove and return the highest priority job."""
        if self._queue:
            _priority, _counter, job = heapq.heappop(self._queue)
            job.status = "running"
            return job
        raise IndexError("dequeue from an empty priority queue")

    def size(self) -> int:
        """Return the number of jobs waiting in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._queue) == 0
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff for job execution."""
import time
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    callback: Optional[Callable[[Job, int, Optional[Exception]], None]] = None
) -> Any:
    """Execute a job with exponential backoff retry logic.
    
    Args:
        job: The job to process.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds before first retry.
        max_delay: Maximum delay cap in seconds.
        callback: Optional hook called after each attempt with (job, attempt_num, error_or_None).
        
    Returns:
        The job result on success, or the last exception on failure.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(1, max_retries + 1):
        try:
            job.status = "running"
            result = job.task(*job.args, **job.kwargs)
            job.status = "completed"
            job.result = result
            if callback:
                callback(job, attempt, None)
            return result

        except Exception as e:
            last_exception = e
            job.error = e
            if callback:
                callback(job, attempt, e)
            
            if attempt == max_retries:
                job.status = "failed"
                break
                
            delay = min(base_delay * (2 ** (attempt - 1)), max_delay)
            time.sleep(delay)

    return last_exception
```

### Key Design Decisions:
1. **Single Responsibility**: `core.py` handles data & basic FIFO, `priority.py` handles heap-based ordering, `retry.py` handles execution resilience.
2. **Relative Imports**: All cross-module imports use `from .module import ...` to keep the package self-contained and avoid circular dependencies.
3. **Clean Public API**: `__init__.py` only exposes `Job` and `JobQueue` as requested. `PriorityQueue` and `process_with_retry` are available via explicit subpackage imports (`from queue.priority import PriorityQueue`).
4. **Type Safety & Documentation**: Added type hints, docstrings, and dataclass defaults for better IDE support and maintainability.
5. **Stable Priority Ordering**: The counter in `PriorityQueue` ensures FIFO behavior for jobs with identical priorities, preventing `heapq` comparison errors on `Job` objects.

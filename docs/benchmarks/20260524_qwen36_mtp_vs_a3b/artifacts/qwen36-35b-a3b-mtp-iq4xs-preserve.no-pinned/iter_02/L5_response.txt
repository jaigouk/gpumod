### `queue/core.py`
```python
"""Core job data model and basic FIFO queue operations."""
from __future__ import annotations
import time
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    job_id: str
    payload: Any
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    retries: int = 0


class JobQueue:
    """Basic FIFO job queue with O(1) enqueue/dequeue operations."""
    
    def __init__(self) -> None:
        self._jobs: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._jobs.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job, or None if empty."""
        return self._jobs.popleft() if self._jobs else None

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._jobs) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._jobs)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""
from __future__ import annotations
import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Priority-based job queue where lower numbers indicate higher priority.
    
    Uses a counter as a tie-breaker to maintain insertion order stability.
    """
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with an associated priority level."""
        entry = (priority, self._counter, job)
        heapq.heappush(self._heap, entry)
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job, or None if empty."""
        if self._heap:
            _priority, _counter, job = heapq.heappop(self._heap)
            return job
        return None

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""
from __future__ import annotations
import random
import time
from typing import Any, Callable, Optional, Tuple
from .core import Job


def process_with_retry(
    job: Job,
    func: Callable[[Any], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    jitter: bool = True
) -> Tuple[bool, Optional[Exception]]:
    """Execute a job with retry logic using exponential backoff.
    
    Args:
        job: The job to process
        func: Callable that accepts job.payload and processes it
        max_retries: Maximum number of retry attempts
        base_delay: Base delay in seconds for backoff calculation
        max_delay: Maximum delay cap in seconds
        jitter: Whether to add random jitter to prevent thundering herd
        
    Returns:
        Tuple of (success: bool, last_exception: Optional[Exception])
    """
    last_exception = None

    for attempt in range(max_retries + 1):
        try:
            func(job.payload)
            job.status = "completed"
            return True, None
        except Exception as e:
            last_exception = e
            job.retries += 1
            if attempt == max_retries:
                job.status = "failed"
                break

            # Calculate delay with exponential backoff
            delay = min(base_delay * (2 ** attempt), max_delay)
            if jitter:
                delay *= (0.5 + random.random())
            time.sleep(delay)

    return False, last_exception
```

### Key Refactoring Decisions:
1. **Relative Imports**: Used `from .core import Job` to avoid circular imports and keep the package self-contained.
2. **Single Responsibility**: 
   - `core.py` handles data modeling and basic FIFO operations
   - `priority.py` isolates heap-based ordering logic
   - `retry.py` encapsulates fault-tolerance strategy
   - `__init__.py` serves as the package facade
3. **Type Safety**: Added modern type hints and `__future__ annotations` for forward references.
4. **Performance**: Used `collections.deque` for O(1) queue operations instead of Python lists.
5. **Stable Priority Queue**: Added a monotonic counter to `PriorityQueue` to break ties deterministically (FIFO among same priorities).

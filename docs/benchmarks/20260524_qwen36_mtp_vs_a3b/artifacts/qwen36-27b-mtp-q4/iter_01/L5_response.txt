### `queue/core.py`
```python
"""Core data model and basic FIFO queue operations."""
from __future__ import annotations

import time
import uuid
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    priority: int = 0  # Lower number = higher priority
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    attempts: int = 0


class JobQueue:
    """Thread-unsafe FIFO queue for Job objects."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        return self._queue.popleft() if self._queue else None

    def peek(self) -> Optional[Job]:
        """Return the first job without removing it."""
        return self._queue[0] if self._queue else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""
from __future__ import annotations

import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Thread-unsafe priority queue for Job objects.
    
    Jobs with lower `priority` values are dequeued first.
    Ties are broken by insertion order.
    """

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Monotonic counter for stable sorting

    def enqueue(self, job: Job) -> None:
        """Add a job to the priority queue."""
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest-priority job."""
        if self._heap:
            _, _, job = heapq.heappop(self._heap)
            return job
        return None

    def peek(self) -> Optional[Job]:
        """Return the highest-priority job without removing it."""
        return self._heap[0][2] if self._heap else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def is_empty(self) -> bool:
        """Check if the priority queue contains no jobs."""
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff."""
from __future__ import annotations

import logging
import time
from typing import Any, Callable

from .core import Job

logger = logging.getLogger(__name__)


def process_with_retry(
    job: Job,
    process_func: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
) -> Any:
    """Execute a job with exponential backoff on failure.
    
    Args:
        job: The job to process.
        process_func: Callable that takes a Job and returns a result.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay between retries in seconds.
        
    Returns:
        The result of a successful process_func execution.
        
    Raises:
        RuntimeError: If all retry attempts are exhausted.
    """
    last_exception: Exception | None = None

    for attempt in range(1, max_retries + 1):
        job.attempts = attempt
        job.status = "processing"
        
        try:
            result = process_func(job)
            job.status = "completed"
            return result
        except Exception as exc:
            last_exception = exc
            job.status = "failed"
            delay = base_delay * (2 ** (attempt - 1))
            logger.warning(
                "Job %s attempt %d/%d failed: %s. Retrying in %.2fs...",
                job.id, attempt, max_retries, exc, delay
            )
            time.sleep(delay)

    job.status = "exhausted"
    raise RuntimeError(
        f"Job {job.id} failed after {max_retries} attempts"
    ) from last_exception
```

### Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles data modeling + basic FIFO semantics
   - `priority.py` encapsulates heap-based ordering
   - `retry.py` isolates fault-tolerance logic
2. **Safe Heap Sorting**: `PriorityQueue` uses a monotonic counter to avoid comparing `Job` objects when priorities tie (prevents `TypeError` in `heapq`).
3. **Clean Public API**: `__init__.py` re-exports only the intended public symbols via `__all__`, preventing internal leakage.
4. **Relative Imports**: All cross-module references use `.core` to ensure package-aware resolution.
5. **Type Hints & Modern Python**: Uses `from __future__ import annotations` for forward compatibility and clean type syntax.

### `queue/core.py`
```python
"""
Core module: Job data model and basic FIFO queue implementation.
"""

from dataclasses import dataclass, field
from typing import Any, Callable, Optional
import threading
import uuid
from collections import deque


@dataclass
class Job:
    """Represents a unit of work to be executed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    task: Callable[..., Any] = None
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    max_retries: int = 3
    retries: int = 0
    status: str = "pending"


class JobQueue:
    """Thread-safe FIFO job queue."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()
        self._lock = threading.Lock()

    def enqueue(self, job: Job) -> None:
        """Add a job to the back of the queue."""
        with self._lock:
            self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the front job, or None if empty."""
        with self._lock:
            return self._queue.popleft() if self._queue else None

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        with self._lock:
            return len(self._queue) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        with self._lock:
            return len(self._queue)
```

### `queue/priority.py`
```python
"""
Priority module: Priority-based job queue using a min-heap.
"""

import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Thread-safe priority queue where lower numbers indicate higher priority."""

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter = 0  # Ensures FIFO ordering for equal priorities

    def enqueue(self, job: Job) -> None:
        """Add a job to the priority queue."""
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job, or None if empty."""
        if self._heap:
            _, _, job = heapq.heappop(self._heap)
            return job
        return None

    def is_empty(self) -> bool:
        """Check if the priority queue contains no jobs."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the priority queue."""
        return len(self._heap)
```

### `queue/retry.py`
```python
"""
Retry module: Job execution with exponential backoff retry logic.
"""

import time
from typing import Any
from .core import Job


def process_with_retry(
    job: Job,
    base_delay: float = 1.0,
    max_delay: float = 60.0
) -> Any:
    """
    Execute a job with exponential backoff retries.
    
    Args:
        job: The Job instance to execute.
        base_delay: Initial delay between retries in seconds.
        max_delay: Maximum delay cap in seconds.
        
    Returns:
        The result of the job's task execution.
        
    Raises:
        Exception: The last exception raised if all retries are exhausted.
    """
    last_exception = None
    
    for attempt in range(job.max_retries + 1):
        try:
            job.status = "running"
            result = job.task(*job.args, **job.kwargs)
            job.status = "completed"
            return result
        except Exception as e:
            last_exception = e
            job.retries += 1
            job.status = "retrying" if attempt < job.max_retries else "failed"
            
            if attempt < job.max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                time.sleep(delay)
                
    raise last_exception
```

### Usage Example
```python

# Basic FIFO queue
fifo = JobQueue()
fifo.enqueue(Job(task=lambda: print("Simple task")))

# Priority queue
prio = PriorityQueue()
prio.enqueue(Job(task=lambda: print("Low priority"), priority=10))
prio.enqueue(Job(task=lambda: print("High priority"), priority=1))

# Process with retry
def flaky_task():
    raise ConnectionError("Network timeout")

job = Job(task=flaky_task, max_retries=2)
try:
    process_with_retry(job, base_delay=0.5)
except Exception as e:
    print(f"Failed after {job.retries} retries: {e}")
```

### Key Design Choices
1. **Single Responsibility**: `core.py` handles data modeling & basic queueing, `priority.py` handles ordering algorithms, `retry.py` handles execution semantics.
2. **Relative Imports**: All internal imports use `from .module import ...` ensuring proper package resolution.
3. **Thread Safety**: `JobQueue` uses `threading.Lock()` for safe concurrent access. `PriorityQueue` relies on `heapq` which is atomic for single operations, but can be wrapped with a lock if full thread-safety is required in production.
4. **Clean Public API**: `__init__.py` explicitly controls what's exposed via `__all__`, preventing accidental leakage of internal utilities.
5. **Stable Priority Sorting**: `PriorityQueue` uses a monotonic counter to guarantee FIFO behavior when priorities are equal, avoiding `heapq` comparison errors with `Job` objects.

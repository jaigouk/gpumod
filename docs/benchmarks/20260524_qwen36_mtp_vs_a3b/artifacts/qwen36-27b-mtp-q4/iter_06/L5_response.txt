### `queue/core.py`
```python
"""Core data model and basic queue operations."""
from __future__ import annotations

import uuid
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Callable, Optional
from collections import deque


@dataclass
class Job:
    """Represents a unit of work to be executed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    task: Callable[..., Any] | None = None
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    status: str = "pending"
    retries: int = 0
    max_retries: int = 3
    created_at: datetime = field(default_factory=datetime.now)


class JobQueue:
    """Basic FIFO job queue with thread-safe-ish operations."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job, or None if empty."""
        return self._queue.popleft() if self._queue else None

    def peek(self) -> Optional[Job]:
        """Return the next job without removing it."""
        return self._queue[0] if self._queue else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue contains any jobs."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""Priority queue implementation using a binary heap."""
from __future__ import annotations

import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Min-heap based priority queue. Lower priority values are dequeued first."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Tie-breaker for equal priorities (FIFO within same priority)

    def enqueue(self, job: Job) -> None:
        """Add a job with priority ordering."""
        entry = (job.priority, self._counter, job)
        heapq.heappush(self._heap, entry)
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest-priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest-priority job without removing it."""
        return self._heap[0][2] if self._heap else None

    def size(self) -> int:
        return len(self._heap)

    def is_empty(self) -> bool:
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff."""
from __future__ import annotations

import time
import logging
from typing import Any, Optional
from .core import Job

logger = logging.getLogger(__name__)


def process_with_retry(
    job: Job,
    max_retries: Optional[int] = None,
    base_delay: float = 1.0,
) -> Any:
    """
    Execute a job with exponential backoff on failure.
    
    Args:
        job: The job to execute
        max_retries: Overrides job.max_retries if provided
        base_delay: Initial delay in seconds before first retry
        
    Returns:
        Result of the job execution
        
    Raises:
        Exception: The last exception encountered if all retries are exhausted
    """
    if max_retries is None:
        max_retries = job.max_retries

    last_exception: Exception | None = None

    for attempt in range(max_retries + 1):
        try:
            if job.task is None:
                raise ValueError(f"Job {job.id} has no task assigned")
            
            result = job.task(*job.args, **job.kwargs)
            job.status = "completed"
            return result

        except Exception as e:
            last_exception = e
            job.retries = attempt + 1
            job.status = "failed"
            logger.warning(
                "Job %s failed (attempt %d/%d): %s",
                job.id, attempt + 1, max_retries + 1, e
            )

            if attempt < max_retries:
                delay = base_delay * (2 ** attempt)
                logger.info("Retrying job %s in %.2fs...", job.id, delay)
                time.sleep(delay)
            else:
                logger.error(
                    "Job %s exhausted all retries. Last error: %s", job.id, e
                )
                raise last_exception
```

### Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles only the data model and basic FIFO operations
   - `priority.py` focuses exclusively on heap-based ordering
   - `retry.py` contains only execution & backoff logic
2. **Consistent Interface**: Both `JobQueue` and `PriorityQueue` share identical method signatures (`enqueue`, `dequeue`, `peek`, `size`, `is_empty`), making them interchangeable via dependency injection.
3. **Relative Imports**: Used throughout (`.core`, `.priority`, `.retry`) which is the Python standard for packages. Requires importing from outside the package or running with `python -m`.
4. **Type Safety & Modern Python**: Uses `from __future__ import annotations`, `deque` for O(1) pops, `heapq` for O(log n) priority operations, and full type hints.
5. **Tie-Breaking in Priority Queue**: Added a monotonic counter to preserve FIFO order when jobs share the same priority level.

### `queue/core.py`
```python
"""
core - Core data structures and basic queue operations.
"""
from __future__ import annotations

import time
import threading
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    data: Any
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    retry_count: int = 0
    max_retries: int = 3


class JobQueue:
    """Thread-safe FIFO job queue."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()
        self._lock = threading.Lock()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        with self._lock:
            self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        with self._lock:
            return self._queue.popleft() if self._queue else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        with self._lock:
            return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return self.size() == 0
```

### `queue/priority.py`
```python
"""
priority - Priority queue implementation using a min-heap.
"""
import heapq
import threading
from typing import Optional
from .core import Job


class PriorityQueue:
    """Thread-safe priority job queue (lower number = higher priority)."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, float, Job]] = []
        self._lock = threading.Lock()
        self._counter = 0  # Ensures stable ordering for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a specific priority level."""
        with self._lock:
            heapq.heappush(self._heap, (priority, self._counter, job))
            self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest-priority job."""
        with self._lock:
            if self._heap:
                _, _, job = heapq.heappop(self._heap)
                return job
            return None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        with self._lock:
            return len(self._heap)

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return self.size() == 0
```

### `queue/retry.py`
```python
"""
retry - Retry logic with exponential backoff.
"""
import time
from typing import Callable, TypeVar, Any
from .core import Job

T = TypeVar('T')


def process_with_retry(
    job: Job,
    func: Callable[..., T],
    *args: Any,
    max_retries: int = 3,
    base_delay: float = 1.0,
    exceptions: tuple[type[Exception], ...] = (Exception,),
    **kwargs: Any
) -> T:
    """
    Execute a function with exponential backoff retry logic.
    
    Updates job.status and job.retry_count accordingly.
    Raises the last exception if all retries are exhausted.
    """
    last_exception = None
    
    for attempt in range(max_retries + 1):
        try:
            result = func(*args, **kwargs)
            job.status = "completed"
            return result
        except exceptions as e:
            last_exception = e
            job.retry_count += 1
            job.status = "retrying"
            
            if attempt < max_retries:
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)
            else:
                job.status = "failed"
                raise last_exception
```

### Key Design Choices:
1. **Single Responsibility**: Each module handles one concern: data/basic ops (`core`), ordering (`priority`), fault tolerance (`retry`), and API exposure (`__init__`).
2. **Thread Safety**: All queue implementations use `threading.Lock` to prevent race conditions during concurrent enqueue/dequeue operations.
3. **Stable Priority Ordering**: `PriorityQueue` uses a monotonic counter alongside priority values to ensure FIFO behavior when priorities are equal.
4. **State Tracking**: `process_with_retry` mutates the `Job` instance to reflect lifecycle state (`pending` → `retrying` → `completed`/`failed`), enabling external monitoring.
5. **Clean Imports**: Relative imports (`.core`, etc.) ensure the package works correctly when installed or imported as a module.

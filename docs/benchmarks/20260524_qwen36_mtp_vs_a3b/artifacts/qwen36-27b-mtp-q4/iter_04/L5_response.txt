### `queue/core.py`
```python
"""
Core data structures and basic queue operations.
"""
import uuid
from dataclasses import dataclass, field
from typing import Any, List, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    priority: int = 0
    status: str = "pending"
    retry_count: int = 0
    max_retries: int = 3


class JobQueue:
    """Basic FIFO job queue implementation."""
    
    def __init__(self) -> None:
        self._queue: List[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        if self._queue:
            return self._queue.pop(0)
        return None

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)

    def __repr__(self) -> str:
        return f"JobQueue(size={len(self._queue)})"
```

### `queue/priority.py`
```python
"""
Priority-based queue implementation.
"""
from .core import Job, JobQueue


class PriorityQueue(JobQueue):
    """Job queue that processes jobs by priority (lower number = higher priority)."""
    
    def enqueue(self, job: Job) -> None:
        """Insert job maintaining sorted order by priority."""
        for i, existing in enumerate(self._queue):
            if job.priority < existing.priority:
                self._queue.insert(i, job)
                return
        self._queue.append(job)
```

### `queue/retry.py`
```python
"""
Retry logic with exponential backoff.
"""
import time
from typing import Any, Callable
from .core import Job


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    base_delay: float = 1.0
) -> Any:
    """
    Execute a job processor with exponential backoff on failure.
    
    Args:
        job: The job to process
        processor: Callable that handles job execution
        base_delay: Initial delay in seconds before first retry
        
    Returns:
        Result from the processor
        
    Raises:
        Exception: Last exception raised if max retries exceeded
    """
    last_exception = None
    
    while job.retry_count <= job.max_retries:
        try:
            job.status = "running"
            result = processor(job)
            job.status = "completed"
            return result
        except Exception as e:
            job.retry_count += 1
            last_exception = e
            
            if job.retry_count > job.max_retries:
                job.status = "failed"
                raise
            
            # Exponential backoff: 1s, 2s, 4s, 8s...
            delay = base_delay * (2 ** (job.retry_count - 1))
            time.sleep(delay)
            
    raise last_exception
```

"""Priority queue implementation using a min-heap."""

import heapq
import threading
from typing import Optional
from .core import Job


class PriorityQueue:
    """Thread-safe priority job queue.

    Jobs with lower priority values are dequeued first.
    Uses a counter to maintain FIFO order for jobs with equal priority.
    """

    def __init__(self, maxsize: int = 0) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0
        self._maxsize: int = maxsize
        self._lock = threading.Lock()

    def enqueue(self, job: Job) -> None:

        with self._lock:

                raise OverflowError("Priority queue is full")
            heapq.heappush(self._heap, (job.priority, self._counter, job))

            if not self._heap:
                return None

            return job

import time
from typing import Callable, Any
from .core import Job, JobStatus


def process_with_retry(

    Args:
        job: The job to process.
        process_fn: Callable that processes the job.
        max_retries: Maximum number of retry attempts.

### `queue/core.py`
```python
"""
Core Module
===========
Defines the Job dataclass and basic FIFO queue operations.
"""

from __future__ import annotations

import uuid
from dataclasses import dataclass, field
from typing import Any, Callable, Optional
from collections import deque


@dataclass(frozen=False)
class Job:
    """Represents a unit of work to be executed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    func: Callable = field(repr=False, hash=False)
    args: tuple = ()
    kwargs: dict = field(default_factory=dict, repr=False, hash=False)
    priority: int = 0
    status: str = "pending"
    result: Any = None
    error: Optional[Exception] = None


class JobQueue:
    """Basic FIFO job queue implementation."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job, or None if empty."""
        return self._queue.popleft() if self._queue else None

    def peek(self) -> Optional[Job]:
        """Return the next job without removing it."""
        return self._queue[0] if self._queue else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""
Priority Module
===============
Implements a priority-based job queue using a min-heap.
"""

import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Job queue that processes jobs by priority (lower number = higher priority)."""

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures stable ordering for equal priorities

    def enqueue(self, job: Job) -> None:
        """Add a job to the priority queue."""
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest-priority job."""
        if self._heap:
            _, _, job = heapq.heappop(self._heap)
            return job
        return None

    def peek(self) -> Optional[Job]:
        """Return the highest-priority job without removing it."""
        return self._heap[0][2] if self._heap else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""
Retry Module
============
Handles job execution with exponential backoff retry logic.
"""

import time
from typing import Any, Optional

from .core import Job


def process_with_retry(
    job: Job,
    max_retries: int = 3,
    base_delay: float = 1.0,
) -> Any:
    """
    Execute a job with exponential backoff retry logic.
    
    Args:
        job: The job to execute.
        max_retries: Maximum number of retry attempts after initial failure.
        base_delay: Base delay in seconds between retries.
        
    Returns:
        The result of the job function.
        
    Raises:
        Exception: The last exception if all retries are exhausted.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(max_retries + 1):
        try:
            job.status = "running"
            result = job.func(*job.args, **job.kwargs)
            job.status = "completed"
            job.result = result
            return result
        except Exception as e:
            last_exception = e
            job.error = e
            job.status = "failed"

            if attempt < max_retries:
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)
    
    # Exhausted all retries
    raise last_exception
```

### Key Design Decisions:
1. **Consistent Interface**: `JobQueue` and `PriorityQueue` share identical public methods (`enqueue`, `dequeue`, `peek`, `size`, `is_empty`), allowing them to be swapped interchangeably in consumer code.
2. **Relative Imports**: All internal imports use `.module` syntax, ensuring the package works correctly regardless of where it's installed or imported from.
3. **Single Responsibility**: 
   - `core.py`: Data model + basic container
   - `priority.py`: Ordering algorithm
   - `retry.py`: Execution lifecycle + fault tolerance
   - `__init__.py`: Package boundary & public API
4. **Type Safety**: Full type hints with `Optional`, `Callable`, and `Any` for modern Python tooling compatibility.
5. **Stable Priority Ordering**: `PriorityQueue` uses a monotonic counter to break ties, preserving insertion order for jobs with equal priority.

### `queue/core.py`
```python
"""Core data structures and basic FIFO queue operations."""

from __future__ import annotations
import uuid
from dataclasses import dataclass, field
from collections import deque
from typing import Any, Optional
from datetime import datetime


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    priority: int = 0
    status: str = "pending"
    created_at: datetime = field(default_factory=datetime.now)
    attempts: int = 0


class JobQueue:
    """Basic FIFO queue for Job objects."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        if self.is_empty():
            return None
        return self._queue.popleft()

    def peek(self) -> Optional[Job]:
        """Return the first job without removing it."""
        return self._queue[0] if not self.is_empty() else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""Priority queue implementation using a binary heap."""

from __future__ import annotations
import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Queue that orders jobs by priority (lower number = higher priority)."""
    
    def __init__(self) -> None:
        # Heap stores tuples: (priority, insertion_order, job)
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures FIFO ordering for equal priorities

    def enqueue(self, job: Job) -> None:
        """Add a job to the priority queue."""
        entry = (job.priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest-priority job."""
        if self.is_empty():
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest-priority job without removing it."""
        if self.is_empty():
            return None
        return self._heap[0][2]

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def is_empty(self) -> bool:
        """Check if the priority queue has no jobs."""
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""Retry logic with configurable exponential backoff."""

from __future__ import annotations
import time
from typing import Callable, Any
from .core import Job


def process_with_retry(
    job: Job,
    process_fn: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    exponential_base: float = 2.0,
) -> Any:
    """
    Execute a job processor with exponential backoff retry logic.
    
    Args:
        job: The job instance to process
        process_fn: Callable that performs the actual work
        max_retries: Maximum number of retry attempts
        base_delay: Initial delay between retries in seconds
        max_delay: Maximum delay cap in seconds
        exponential_base: Multiplier for exponential growth
        
    Returns:
        Result of the successful process_fn call
        
    Raises:
        Exception: The last exception raised if all retries are exhausted
    """
    last_exception: Exception | None = None
    
    for attempt in range(max_retries + 1):
        try:
            job.attempts = attempt + 1
            job.status = "processing"
            return process_fn(job)
        except Exception as e:
            last_exception = e
            job.status = "failed"
            
            if attempt < max_retries:
                delay = min(base_delay * (exponential_base ** attempt), max_delay)
                time.sleep(delay)
            else:
                job.status = "exhausted"
                
    raise last_exception
```

### Key Design Decisions:
1. **`collections.deque` in `core.py`**: Used instead of `list.pop(0)` for O(1) dequeue operations
2. **Heap tie-breaking in `priority.py`**: `self._counter` ensures jobs with equal priority maintain insertion order (FIFO within priority levels)
3. **Relative imports**: All internal references use `.module` syntax, which works correctly when imported as a package
4. **Explicit `__all__`**: Controls the public API and prevents accidental leakage of internal helpers
5. **Type hints & docstrings**: Added for maintainability and IDE support
6. **Backoff safety**: `min(..., max_delay)` prevents unbounded wait times in production

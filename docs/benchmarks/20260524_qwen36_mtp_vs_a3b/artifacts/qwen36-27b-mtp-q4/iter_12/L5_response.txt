### `queue/retry.py`
```python
"""
retry.py - Retry logic with exponential backoff.
"""
import time
import logging
from typing import Callable, Any
from .core import Job

logger = logging.getLogger(__name__)


def process_with_retry(
    job: Job,
    process_func: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0
) -> Any:
    """
    Execute a job's processing function with exponential backoff on failure.
    
    Args:
        job: The Job instance to process.
        process_func: Callable that performs the actual work.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds before first retry.
        max_delay: Maximum delay cap in seconds.
        
    Returns:
        The result of process_func on success.
        
    Raises:
        RuntimeError: If the job fails after exhausting all retries.
    """
    for attempt in range(max_retries + 1):
        try:
            result = process_func(job)
            job.status = "completed"
            return result
        except Exception as e:
            job.retries += 1
            job.status = "retrying"
            logger.warning(
                f"Job {job.id} failed (attempt {attempt + 1}/{max_retries}): {e}"
            )

            if attempt < max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                logger.info(f"Retrying job {job.id} in {delay:.2f}s...")
                time.sleep(delay)
            else:
                job.status = "failed"
                raise RuntimeError(
                    f"Job {job.id} failed after {max_retries} retries"
                ) from e
```

### `queue/core.py`
```python
"""Core data model and basic FIFO queue implementation."""

import time
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    status: str = "pending"
    retry_count: int = 0
    max_retries: int = 3
    created_at: float = field(default_factory=time.time)


class JobQueue:
    """Thread-safe FIFO queue for Job instances."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job, or None if empty."""
        return self._queue.popleft() if self._queue else None

    def is_empty(self) -> bool:
        """Check if the queue contains any jobs."""
        return not self._queue

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Priority-based queue where lower numeric priority = higher precedence."""

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Breaks ties and preserves insertion order

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with the specified priority.
        
        Args:
            job: The job to queue.
            priority: Numeric priority. Lower values are processed first.
        """
        # Negate priority so heapq (min-heap) treats lower numbers as higher priority
        entry = (-priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job, or None if empty."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def is_empty(self) -> bool:
        """Check if the queue contains any jobs."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

import random
import time
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    handler: Callable[[Job], Any],
    max_retries: Optional[int] = None,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
) -> Any:
    """Execute a job handler with exponential backoff retry logic.
    
    Args:
        job: The job being processed.
        handler: Callable that processes the job.
        max_retries: Override job's max_retries. Defaults to job.max_retries.
        base_delay: Initial delay in seconds.
        max_delay: Maximum delay cap in seconds.
        exponential_base: Base for exponential backoff calculation.
        jitter: Whether to add random jitter to prevent thundering herd.
        
    Returns:
        The result of the handler on success.
        
    Raises:
        Exception: The last exception raised by the handler after all retries exhausted.
    """
    retries = max_retries if max_retries is not None else job.max_retries

    for attempt in range(retries + 1):
        try:
            return handler(job)
        except Exception as e:
            job.retry_count += 1
            job.status = "failed"

            if job.retry_count > retries:
                raise e from e

            delay = min(base_delay * (exponential_base ** job.retry_count), max_delay)
            if jitter:
                delay *= (0.5 + random.random() * 0.5)

            time.sleep(delay)
```

### Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles data modeling & FIFO mechanics
   - `priority.py` handles ordering logic via `heapq`
   - `retry.py` handles resilience/backoff exclusively
2. **No Inheritance Abuse**: `PriorityQueue` doesn't inherit from `JobQueue` to avoid state duplication or unexpected method resolution order issues. It implements the same interface independently.
3. **Thread Safety**: `JobQueue` uses `collections.deque` which is safe for concurrent appends/pops in CPython. For production async/concurrent workloads, consider wrapping with `asyncio.Queue` or `threading.Lock` as needed.
4. **Clean Public API**: `__init__.py` explicitly defines `__all__`, preventing accidental leakage of internal modules while keeping the import surface minimal: `from queue import Job, JobQueue, PriorityQueue, process_with_retry`
5. **Exponential Backoff**: Includes jitter to prevent synchronized retries across distributed workers, a common production requirement.

import heapq

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = 0

    def add_job(self, job_name, job_data, priority=0):
        # Negate priority because heapq is a min-heap, but we want higher priority first
        # Use counter to maintain FIFO for same priority
        heapq.heappush(self._heap, (-priority, self._counter, job_name, job_data))
        self._counter += 1

    def get_next_job(self):
        if not self._heap:
            return None
        # Pop the item with the smallest negative priority (highest priority)
        # If priorities are equal, smallest counter (earliest added) comes first
        _, _, job_name, job_data = heapq.heappop(self._heap)
        return (job_name, job_data)

### `queue/core.py`
```python
"""Core job definition and basic FIFO queue operations."""

from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any, Optional
from collections import deque


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    status: str = "pending"
    created_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    attempts: int = 0


class JobQueue:
    """Basic FIFO queue for job management."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job in line."""
        return self._queue.popleft() if self._queue else None

    def peek(self) -> Optional[Job]:
        """View the next job without removing it."""
        return self._queue[0] if self._queue else None

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

from __future__ import annotations
import heapq
from typing import Any, Optional
from .core import Job


class PriorityQueue:
    """Job queue that processes higher priority jobs first.
    
    Lower numerical values indicate higher priority.
    Maintains FIFO ordering for jobs with equal priority.
    """
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Monotonic counter for stable sorting

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority level."""
        heapq.heappush(self._heap, (priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """View the highest priority job without removing it."""
        return self._heap[0][2] if self._heap else None

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._heap) == 0

    def __len__(self) -> int:
        return len(self._heap)
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff for job processing."""

from __future__ import annotations
import time
import logging
from typing import Callable, Any
from .core import Job

logger = logging.getLogger(__name__)


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    backoff_factor: float = 2.0
) -> Any:
    """Process a job with automatic retry and exponential backoff.
    
    Args:
        job: The job to process.
        processor: Callable that executes the job's payload.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds before first retry.
        backoff_factor: Multiplier applied to delay after each failed attempt.
        
    Returns:
        The result of the processor function.
        
    Raises:
        Exception: The last exception raised if all retries are exhausted.
    """
    last_exception: Exception | None = None

    for attempt in range(1, max_retries + 1):
        job.attempts = attempt
        try:
            result = processor(job)
            job.status = "completed"
            return result
        except Exception as e:
            last_exception = e
            logger.warning(
                f"Attempt {attempt}/{max_retries} failed for job {job.id}: {e}"
            )
            if attempt < max_retries:
                delay = base_delay * (backoff_factor ** (attempt - 1))
                logger.info(f"Retrying in {delay} seconds...")
                time.sleep(delay)

    job.status = "failed"
    if last_exception:
        raise last_exception
    raise RuntimeError("Unexpected retry exhaustion")
```

### Key Design Decisions:
1. **Single Responsibility**: Each module handles exactly one concern: data/queue logic (`core.py`), ordering strategy (`priority.py`), and fault tolerance (`retry.py`).
2. **Relative Imports**: Uses `from .core import ...` syntax to keep the package self-contained and avoid circular dependencies.
3. **Efficient Data Structures**: `collections.deque` for O(1) FIFO operations, `heapq` for O(log n) priority management.
4. **Stable Priority Ordering**: The counter in `PriorityQueue` ensures FIFO behavior when priorities are equal, preventing heap instability.
5. **Type Safety & Documentation**: Full type hints, docstrings, and modern Python patterns (`__future__.annotations`, `dataclass`) for better IDE support and maintainability.
6. **Extensible API**: `__init__.py` exposes a clean public interface while keeping internal modules encapsulated.

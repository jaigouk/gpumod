```python
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Callable, Optional
import time

@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    func: Callable
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    status: str = "pending"
    retries: int = 0
    max_retries: int = 3
    priority: int = 0
    created_at: float = field(default_factory=time.time)

class JobQueue:
    """Basic FIFO job queue implementation."""
    def __init__(self):
        self._queue = deque()

    def add(self, job: Job) -> None:
        """Add a job to the queue."""
        self._queue.append(job)

    def get(self) -> Optional[Job]:
        """Retrieve the next job from the queue."""
        if not self._queue:
            return None
        return self._queue.popleft()

    def process(self, job: Job) -> Any:
        """Execute a job and update its status."""
        job.status = "running"
        try:
            result = job.func(*job.args, **job.kwargs)
            job.status = "completed"
            return result
        except Exception as e:
            job.status = "failed"
            raise e
```

2. **Proper Imports**: `core` is the foundation. `priority` and `retry` import only what they need from `core`. No circular dependencies.
3. **Clean Public API**: `__init__.py` explicitly controls the surface area via `__all__`. Submodules remain accessible for advanced usage.
4. **Naming Safety**: Used `collections.deque` instead of `queue.Queue` to avoid shadowing Python's standard library `queue` module.
5. **Jitter in Backoff**: Added `random.uniform(0, 0.1)` to prevent synchronized retries when multiple workers fail simultaneously.

### 📦 Usage Example
```python
from queue import Job, JobQueue, process_with_retry, PriorityQueue

def my_task(x: int) -> int:
    return x * 2

# Basic FIFO queue
q = JobQueue()
q.add(Job(id="1", func=my_task, args=(5,)))
job = q.get()
result = q.process(job)  # 10

# Priority queue with retry logic
pq = PriorityQueue()
pq.add(Job(id="2", func=my_task, args=(10,), priority=1))
job = pq.get()
result = process_with_retry(job, base_delay=0.5, max_retries=3)
```

```python
import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Priority-based job queue. Lower priority values are processed first."""

    def __init__(self):
        self._heap: list = []
        self._counter = 0  # Ensures stable ordering for equal priorities
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

import random
import time
from typing import Any, Callable, Optional, TypeVar

from .core import Job

T = TypeVar("T")


def process_with_retry(
    job: Job,
    processor: Callable[[Job], T],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
) -> T:
    """Process a job with exponential backoff retry logic.

    Args:
        job: The job to process.
        processor: Callable that accepts a Job and returns a result.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
        max_delay: Maximum delay in seconds to cap backoff.

    Returns:
        The result of the processor.

    Raises:
        Exception: The last exception raised if max retries are exceeded.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(max_retries + 1):
        job.attempts += 1
        try:
            return processor(job)
        except Exception as e:
            last_exception = e
            if attempt < max_retries:
                # Exponential backoff with jitter to prevent thundering herd
                delay = min(base_delay * (2 ** attempt) + random.uniform(0, 1), max_delay)
                time.sleep(delay)

    raise last_exception
```

   - `retry.py` isolates failure recovery and backoff strategy
   - `__init__.py` acts as the clean facade/public API
2. **Import Safety**: All cross-module imports use relative syntax (`from .core import Job`), preventing circular dependencies and making the package self-contained.
3. **Production Readiness**: 
   - Added type hints and docstrings
   - Used a counter tie-breaker in `PriorityQueue` for stable sorting
   - Added jitter to exponential backoff to prevent retry storms
   - `Job.attempts` is mutated in-place to track state across retries

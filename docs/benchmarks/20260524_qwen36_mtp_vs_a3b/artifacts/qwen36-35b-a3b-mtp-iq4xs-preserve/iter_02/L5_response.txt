        self.status = "completed"

    def mark_failed(self) -> None:
        self.status = "failed"


class JobQueue:
    """Basic FIFO job queue."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def add(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def get_next(self) -> Optional[Job]:
        """Retrieve and mark the next job as processing."""
        if self._queue:
            job = self._queue.popleft()
            job.mark_processing()
            return job
        return None

from __future__ import annotations

import heapq
from typing import Optional

from .core import Job


class PriorityQueue:
    """Min-heap based priority queue. Lower priority value = higher precedence."""

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Stable sort tie-breaker

    def add(self, job: Job, priority: int = 0) -> None:

        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

            _, _, job = heapq.heappop(self._heap)

            return job

"""Retry logic with exponential backoff."""

from __future__ import annotations

import logging
import time
from typing import Any, Callable

from .core import Job

logger = logging.getLogger(__name__)


def process_with_retry(
    job: Job,
    processor: Callable[[Any], Any],
    base_delay: float = 1.0,
    max_delay: float = 60.0,
) -> bool:
    """Process a job with exponential backoff retry logic.

    Args:
        job: The job to process.
        processor: Callable that takes job.payload and executes the work.
        base_delay: Initial delay in seconds between retries.
        max_delay: Maximum delay cap in seconds.

    Returns:
        True if processed successfully, False if all retries exhausted.
    """
    delay = base_delay
    for attempt in range(job.max_retries + 1):
        try:
            processor(job.payload)
            job.mark_completed()
            return True
        except Exception as e:
            job.retries = attempt + 1
            job.mark_failed()
            logger.warning(
                f"Attempt {attempt + 1}/{job.max_retries} failed for job {job.id}: {e}"
            )
            if attempt < job.max_retries:
                sleep_time = min(delay * (2 ** attempt), max_delay)
                logger.info(f"Retrying job {job.id} in {sleep_time:.2f}s...")
                time.sleep(sleep_time)

    logger.error(f"Job {job.id} permanently failed after {job.max_retries} retries.")
    return False

### `queue/core.py`
```python
"""Core data structures and base queue implementation."""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Optional
from datetime import datetime


@dataclass
class Job:
    """Represents a unit of work in the queue."""
    id: str
    payload: Any
    status: str = "pending"
    retries: int = 0
    created_at: datetime = field(default_factory=datetime.now)
    result: Any = None
    error: Optional[Exception] = None


class JobQueue:
    """Base FIFO queue implementation."""

    def __init__(self) -> None:
        self._jobs: list[Job] = []

    def add(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._jobs.append(job)

    def get_next(self) -> Optional[Job]:
        """Retrieve and remove the next job (FIFO)."""
        if not self._jobs:
            return None
        return self._jobs.pop(0)

    def is_empty(self) -> bool:
        """Check if the queue has no pending jobs."""
        return len(self._jobs) == 0
```

### `queue/priority.py`
```python
"""Priority-based queue implementation."""
from __future__ import annotations

import heapq
from typing import Optional

from .core import Job, JobQueue


class PriorityQueue(JobQueue):
    """Queue that processes jobs by priority (lower number = higher priority).
    
    Inherits from JobQueue but overrides storage and retrieval to use a heap.
    """

    def __init__(self) -> None:
        # Initialize parent but replace storage mechanism
        super().__init__()
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Tie-breaker for equal priorities

    def add(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority level."""
        # heapq is a min-heap. Counter ensures stable ordering for equal priorities.
        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def get_next(self) -> Optional[Job]:
        """Retrieve and remove the highest priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""
from __future__ import annotations

import random
import time
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    handler: Callable[[Any], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
) -> Job:
    """Execute a job handler with exponential backoff retry logic.
    
    Args:
        job: The job to process (updated in-place with status/result).
        handler: Callable that accepts job.payload and returns a result.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
        max_delay: Maximum delay cap in seconds.
        
    Returns:
        The updated Job instance with final status, result, or error.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(max_retries + 1):
        try:
            job.result = handler(job.payload)
            job.status = "completed"
            return job
        except Exception as e:
            last_exception = e
            job.retries += 1
            job.status = "failed"
            job.error = e

            if attempt == max_retries:
                break

            # Exponential backoff: base_delay * 2^attempt
            delay = min(base_delay * (2 ** attempt), max_delay)
            # Add random jitter (±50%) to prevent thundering herd
            delay *= (0.5 + random.random())
            time.sleep(delay)

    return job
```

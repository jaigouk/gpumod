### `queue/core.py`
```python
"""Core job dataclass and basic FIFO queue operations."""

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any, Callable, List, Optional
import uuid


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    task: Optional[Callable] = None
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    status: str = "pending"
    created_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    result: Any = None
    error: Optional[str] = None


class JobQueue:
    """Basic FIFO queue for processing jobs sequentially."""

    def __init__(self):
        self._jobs: List[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._jobs.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        if self.is_empty():
            return None
        return self._jobs.pop(0)

    def peek(self) -> Optional[Job]:
        """Return the first job without removing it."""
        if self.is_empty():
            return None
        return self._jobs[0]

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._jobs) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._jobs)

    def process_all(self, processor: Callable[[Job], Any]) -> List[Job]:
        """Process all jobs in the queue using the provided processor."""
        processed = []
        while not self.is_empty():
            job = self.dequeue()
            try:
                job.status = "processing"
                job.result = processor(job)
                job.status = "completed"
            except Exception as e:
                job.status = "failed"
                job.error = str(e)
            processed.append(job)
        return processed
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

import heapq
from typing import Any, Callable, Optional
from .core import Job


class PriorityQueue:
    """Queue that processes jobs based on priority (lower number = higher priority)."""

    def __init__(self):
        self._heap: list = []  # (priority, insertion_order, job)
        self._counter = 0

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority."""
        # Counter ensures FIFO order for equal priorities
        heapq.heappush(self._heap, (priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if self.is_empty():
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest priority job without removing it."""
        if self.is_empty():
            return None
        return self._heap[0][2]

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def process_all(self, processor: Callable[[Job], Any]) -> list[Job]:
        """Process all jobs in priority order."""
        processed = []
        while not self.is_empty():
            job = self.dequeue()
            try:
                job.status = "processing"
                job.result = processor(job)
                job.status = "completed"
            except Exception as e:
                job.status = "failed"
                job.error = str(e)
            processed.append(job)
        return processed
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

import random
import time
from typing import Any, Callable, Optional
from .core import Job


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True,
) -> Job:
    """
    Execute a job with exponential backoff retry logic.
    
    Args:
        job: The job instance to process and update status on.
        processor: Callable that executes the job's task.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds between retries.
        max_delay: Maximum delay in seconds (caps exponential growth).
        jitter: Whether to add random jitter to prevent thundering herd.
        
    Returns:
        The updated job instance with final status.
    """
    last_exception = None

    for attempt in range(1, max_retries + 1):
        try:
            job.status = "processing"
            job.result = processor(job)
            job.status = "completed"
            return job
        except Exception as e:
            last_exception = e
            job.status = f"retry_{attempt}/{max_retries}"
            job.error = str(e)

            if attempt < max_retries:
                delay = min(base_delay * (2 ** (attempt - 1)), max_delay)
                if jitter:
                    delay *= random.uniform(0.5, 1.5)
                time.sleep(delay)

    job.status = "failed"
    job.error = f"Failed after {max_retries} retries: {last_exception}"
    return job
```

### Key Design Decisions:
1. **Relative Imports**: All cross-module imports use `from .module import ...` to ensure the package works regardless of how it's installed or run.
2. **Consistent Interfaces**: `JobQueue` and `PriorityQueue` share identical method signatures (`enqueue`, `dequeue`, `peek`, `is_empty`, `size`, `process_all`), allowing them to be swapped via dependency injection or configuration.
3. **Modern Python**: Uses `dataclasses`, `typing`, `datetime.now(timezone.utc)` (avoids deprecated `utcnow()`), and `list[Job]` syntax (Python 3.9+).
4. **State Tracking**: Jobs carry their own lifecycle state (`pending` → `processing` → `completed`/`failed`), making them observable and debuggable.
5. **Retry Flexibility**: `process_with_retry` is a pure function that updates the job in-place, allowing it to be used with any queue type or as a standalone utility.

### Usage Example:
```python
from queue import Job, JobQueue, PriorityQueue, process_with_retry

# Basic FIFO queue
queue = JobQueue()
queue.enqueue(Job(task=lambda j: print(f"Processing {j.id}")))
queue.process_all(lambda j: j.task(j) if j.task else None)

# Priority queue with retry
pq = PriorityQueue()
pq.enqueue(Job(task=lambda j: 1/0), priority=1)
pq.enqueue(Job(task=lambda j: "success"), priority=0)

for _ in range(pq.size()):
    job = pq.dequeue()
    process_with_retry(job, lambda j: j.task(j), max_retries=2)
```

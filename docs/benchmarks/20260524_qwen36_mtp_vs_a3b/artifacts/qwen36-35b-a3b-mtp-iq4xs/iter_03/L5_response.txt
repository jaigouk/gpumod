### `queue/core.py`
```python
"""Core module: Job dataclass and basic FIFO queue operations."""
from dataclasses import dataclass
from typing import Any, Callable, List, Optional
import threading

@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    status: str = "pending"
    attempts: int = 0
    max_retries: int = 3

class JobQueue:
    """Thread-safe FIFO job queue."""
    def __init__(self):
        self._queue: List[Job] = []
        self._lock = threading.Lock()

    def add(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        with self._lock:
            self._queue.append(job)

    def pop(self) -> Optional[Job]:
        """Remove and return the next job from the front of the queue."""
        with self._lock:
            if self._queue:
                return self._queue.pop(0)
            return None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        with self._lock:
            return len(self._queue)

    def process(self, job: Job, handler: Callable[[Job], Any]) -> Any:
        """Execute a handler function for a given job with status tracking."""
        with self._lock:
            job.status = "processing"
        try:
            result = handler(job)
            with self._lock:
                job.status = "completed"
            return result
        except Exception as e:
            with self._lock:
                job.status = "failed"
            raise e
```

### `queue/retry.py`
```python
"""Retry module: Exponential backoff retry logic."""
import time
from typing import Any, Callable, Optional
from .core import Job

def process_with_retry(
    job: Job,
    handler: Callable[[Job], Any],
    base_delay: float = 1.0,
    max_retries: Optional[int] = None
) -> Any:
    """
    Execute a handler with exponential backoff retry logic.

    Args:
        job: The job instance to process.
        handler: Callable that takes a job and returns a result.
        base_delay: Base delay in seconds for backoff calculation.
        max_retries: Maximum number of retries. Defaults to job.max_retries.

    Returns:
        The result of the successful handler execution.

    Raises:
        The last exception raised by the handler if max retries are exceeded.
    """
    retries = max_retries if max_retries is not None else job.max_retries

    for attempt in range(retries + 1):
        try:
            return handler(job)
        except Exception as e:
            job.attempts = attempt + 1
            if attempt == retries:
                raise
            delay = base_delay * (2 ** attempt)
            time.sleep(delay)
```

### Key Design Decisions:
1. **Single Responsibility**: `core.py` handles data & basic FIFO operations, `priority.py` handles heap-based ordering, and `retry.py` isolates backoff logic. No module depends on more than one sibling module.
2. **Thread Safety**: All queue operations use `threading.Lock()` to prevent race conditions in concurrent environments.
3. **Priority Queue Design**: Instead of inheriting from `JobQueue`, `PriorityQueue` implements the same interface independently. This avoids lock inheritance conflicts and keeps the heap logic clean.
4. **Clean Public API**: `__init__.py` uses explicit relative imports and `__all__` to define exactly what's exposed when users do `from queue import Job, JobQueue`.
5. **Type Hints & Docstrings**: Added throughout for better IDE support and maintainability.

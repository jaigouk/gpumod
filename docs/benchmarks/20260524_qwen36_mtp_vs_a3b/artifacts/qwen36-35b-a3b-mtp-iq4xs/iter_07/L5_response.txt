### 📁 `queue/core.py`
```python
"""Core data structures and basic FIFO queue operations."""

from dataclasses import dataclass, field
from typing import Any, Optional
from collections import deque
import time


@dataclass
class Job:
    """Represents a single unit of work."""
    job_id: str
    payload: Any
    created_at: float = field(default_factory=time.time)
    status: str = "pending"
    result: Optional[Any] = None
    error: Optional[Exception] = None


class JobQueue:
    """Thread-safe FIFO queue for processing jobs in order."""

    def __init__(self):
        self._queue = deque()

    def enqueue(self, job: Job) -> str:
        """Add a job to the queue. Returns the job_id."""
        self._queue.append(job)
        return job.job_id

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job. Returns None if empty."""
        return self._queue.popleft() if self._queue else None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return not self._queue
```

### 📁 `queue/priority.py`
```python
"""Priority-based queue implementation using a min-heap."""

import heapq
from typing import Any
from .core import Job


class PriorityQueue:
    """Queue that processes jobs based on priority (lower value = higher priority)."""

    def __init__(self):
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Tie-breaker for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> str:
        """Add a job with a given priority. Returns the job_id."""
        # heapq is a min-heap; counter ensures FIFO order for equal priorities
        heapq.heappush(self._heap, (priority, self._counter, job))
        self._counter += 1
        return job.job_id

    def dequeue(self) -> Job:
        """Remove and return the highest priority job."""
        if not self._heap:
            raise IndexError("Cannot dequeue from an empty priority queue")
        _, _, job = heapq.heappop(self._heap)
        return job

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0
```

### 📁 `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""

import time
import random
from typing import Callable, Any, Optional, Tuple
from .core import Job


def process_with_retry(
    job: Job,
    func: Callable[..., Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
) -> Tuple[Optional[Any], Optional[Exception]]:
    """
    Execute a function with exponential backoff retry logic.
    
    Updates the provided Job instance with status, result, or error.
    Returns a tuple of (result, exception).
    """
    last_exception: Optional[Exception] = None

    for attempt in range(max_retries + 1):
        try:
            result = func(job.payload)
            job.status = "completed"
            job.result = result
            return result, None
        except Exception as e:
            last_exception = e
            job.error = e
            job.status = "failed"
            
            if attempt == max_retries:
                break
                
            delay = min(base_delay * (exponential_base ** attempt), max_delay)
            if jitter:
                delay *= random.uniform(0.5, 1.5)
                
            time.sleep(delay)
            
    return None, last_exception
```

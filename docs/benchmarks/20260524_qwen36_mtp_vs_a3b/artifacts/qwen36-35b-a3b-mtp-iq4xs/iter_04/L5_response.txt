### `queue/core.py`
```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Optional
import time


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    name: str
    func: Callable
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    created_at: float = field(default_factory=time.time)
    status: str = "pending"  # pending | running | completed | failed
    priority: int = 0        # Used by PriorityQueue


class JobQueue:
    """Basic FIFO queue for managing jobs."""
    def __init__(self) -> None:
        self._jobs: list[Job] = []

    def enqueue(self, job: Job) -> None:
        self._jobs.append(job)

    def dequeue(self) -> Optional[Job]:
        return self._jobs.pop(0) if self._jobs else None

    def is_empty(self) -> bool:
        return not self._jobs

    def process(self, job: Job) -> None:
        """Execute a single job and update its status."""
        job.status = "running"
        try:
            job.func(*job.args, **job.kwargs)
            job.status = "completed"
        except Exception as e:
            job.status = "failed"
            raise e
```

### `queue/priority.py`
```python
import heapq
from typing import Optional
from .core import Job


class PriorityQueue:
    """Priority-based queue implementation using a min-heap.
    Lower priority values are processed first.
    """
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ties are broken by insertion order

    def enqueue(self, job: Job) -> None:
        entry = (job.priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def dequeue(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def is_empty(self) -> bool:
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
import time
from .core import Job


def process_with_retry(
    job: Job,
    max_retries: int = 3,
    base_delay: float = 1.0
) -> bool:
    """Process a job with exponential backoff retry logic.
    
    Args:
        job: The job to process.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
        
    Returns:
        True if job completed successfully, False after max retries.
    """
    attempts = 0
    while attempts <= max_retries:
        try:
            job.status = "running"
            job.func(*job.args, **job.kwargs)
            job.status = "completed"
            return True
        except Exception as e:
            attempts += 1
            if attempts > max_retries:
                job.status = "failed"
                return False
            
            delay = base_delay * (2 ** (attempts - 1))
            time.sleep(delay)
```

### Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles job definition & FIFO queue mechanics
   - `priority.py` handles heap-based ordering
   - `retry.py` handles transient failure recovery & backoff
2. **Relative Imports**: All inter-module imports use `.core` syntax, making the package portable and avoiding circular dependency issues.
3. **Clean Public API**: `__init__.py` explicitly exports only `Job` and `JobQueue`, hiding internal implementation details (`PriorityQueue`, `process_with_retry`) unless explicitly imported.
4. **Type Safety & Compatibility**: Uses `from __future__ import annotations` and modern typing patterns while maintaining Python 3.9+ compatibility.
5. **Swap-Ready**: `PriorityQueue` implements the same `enqueue`/`dequeue`/`is_empty` interface as `JobQueue`, allowing drop-in replacement in consumer code. Execution logic is intentionally separated from queue management.

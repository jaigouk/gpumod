### `queue/core.py`
```python
from dataclasses import dataclass, field
from typing import Callable, Any, List, Optional

@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    func: Callable
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    status: str = "pending"
    priority: int = 0
    attempts: int = 0

class JobQueue:
    """Base queue implementation with FIFO ordering."""
    def __init__(self):
        self.jobs: List[Job] = []

    def add(self, job: Job) -> None:
        self.jobs.append(job)

    def get_next(self) -> Optional[Job]:
        if not self.jobs:
            return None
        return self.jobs.pop(0)

    def process(self, job: Job) -> Any:
        job.status = "running"
        job.attempts += 1
        try:
            result = job.func(*job.args, **job.kwargs)
            job.status = "completed"
            return result
        except Exception as e:
            job.status = "failed"
            raise e
```

### `queue/priority.py`
```python
from heapq import heappush, heappop
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """Priority-based queue that processes lower priority values first."""
    def __init__(self):
        super().__init__()
        self._heap = []
        self._counter = 0

    def add(self, job: Job) -> None:
        # Lower priority value = higher priority
        # Counter maintains FIFO order for equal priorities
        entry = (job.priority, self._counter, job)
        self._counter += 1
        heappush(self._heap, entry)

    def get_next(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, _, job = heappop(self._heap)
        return job
```

### `queue/retry.py`
```python
import time
from .core import Job
from typing import Optional, Any

def process_with_retry(
    job: Job,
    max_retries: int = 3,
    base_delay: float = 1.0
) -> Optional[Any]:
    """Process a job with exponential backoff retry logic."""
    for attempt in range(1, max_retries + 1):
        try:
            result = job.func(*job.args, **job.kwargs)
            job.status = "completed"
            return result
        except Exception as e:
            job.attempts = attempt
            if attempt == max_retries:
                job.status = "failed"
                raise e
            delay = base_delay * (2 ** (attempt - 1))

```

### Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles data modeling (`Job`) and base queue mechanics
   - `priority.py` isolates heap-based ordering logic
   - `retry.py` encapsulates backoff strategy without coupling to queue internals
2. **Relative Imports**: Uses `.core`, `.priority`, `.retry` for package cohesion and cleaner runtime resolution
3. **Type Safety**: Full type hints with `Optional`, `List`, `Callable`, and `Any` for better IDE support and static analysis
4. **Minimal Public API**: `__init__.py` only exposes `Job` and `JobQueue` as requested. Other components are importable directly when needed, avoiding namespace pollution
5. **Extensibility**: `PriorityQueue` inherits from `JobQueue` but overrides storage/ordering. `process_with_retry` works with any `Job` instance, making it composable with any queue implementation.

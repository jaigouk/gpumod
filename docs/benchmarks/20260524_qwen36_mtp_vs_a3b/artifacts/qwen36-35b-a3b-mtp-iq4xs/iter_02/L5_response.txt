```python
        self.max_retries = max_retries

    def enqueue(self, job: Job) -> None:
        """Add a job to the queue."""
        self._queue.push(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        return self._queue.pop()

    def is_empty(self) -> bool:
        """Check if the queue has no pending jobs."""
        return self._queue.size() == 0

    def __len__(self) -> int:
```

```python
from __future__ import annotations

import heapq
from typing import Any, Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from .core import Job


class PriorityQueue:
    """Stable min-heap priority queue implementation."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Any]] = []
        self._counter = 0  # Ensures FIFO order for equal priorities

    def push(self, job: Any) -> None:
        """Push a job onto the priority queue."""
        from .core import Job
        if not isinstance(job, Job):
            raise TypeError("PriorityQueue expects Job instances")
            
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def pop(self) -> Optional[Job]:
        """Pop and return the job with the highest priority (lowest number)."""
        if self._heap:

            return job
        return None
```

```python
    retries = max_retries if max_retries is not None else job.max_retries
    last_exception = None

    for attempt in range(retries + 1):
        try:
            job.attempts = attempt + 1
            job.status = "processing"
            result = handler(job)
            job.status = "completed"
            return result
        except Exception as e:
            last_exception = e
            job.status = "failed"
            
            if attempt < retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                logger.warning(
                    f"Attempt {attempt + 1}/{retries + 1} failed for job {job.id}: {e}. "
                    f"Retrying in {delay:.2f}s..."
                )
                time.sleep(delay)
            else:
                logger.error(
                    f"Job {job.id} failed after {retries + 1} attempts. Last error: {e}"
                )
                
    raise last_exception
```

### Key Design Decisions:
1. **Circular Import Prevention**: `core.py` imports `PriorityQueue` at the top, while `priority.py` uses `TYPE_CHECKING` and a runtime `isinstance` check to import `Job` only when needed. This prevents `ImportError`/`RecursionError` during module initialization.
2. **Stable Sorting**: The priority queue uses a monotonic counter `(priority, counter, job)` so that jobs with equal priorities are processed in FIFO order.
3. **State Management**: `Job` status and attempt counts are updated in-place during retry processing, allowing external code to inspect job state after processing.
4. **Clean API**: `__init__.py` only exposes the high-level components (`Job`, `JobQueue`). Lower-level utilities (`PriorityQueue`, `process_with_retry`) remain accessible via their submodules if needed, keeping the public surface minimal.

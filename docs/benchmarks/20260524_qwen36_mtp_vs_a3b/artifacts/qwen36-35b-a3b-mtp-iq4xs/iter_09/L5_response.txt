### `queue/core.py`
```python
"""Core job data structures and basic FIFO queue operations."""

from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Optional
import time
import uuid


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: uuid.uuid4().hex)
    payload: Any = None
    created_at: float = field(default_factory=time.time)
    status: str = "pending"  # pending | processing | completed | failed


class JobQueue:
    """Basic FIFO queue for Job instances."""
    
    def __init__(self) -> None:
        self._queue: list[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        if not self._queue:
            return None
        return self._queue.pop(0)

    def peek(self) -> Optional[Job]:
        """Return the first job without removing it."""
        return self._queue[0] if self._queue else None

    def is_empty(self) -> bool:
        """Check if the queue has no jobs."""
        return len(self._queue) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)
```

### `queue/priority.py`
```python
"""Priority queue implementation for ordered job processing."""

from __future__ import annotations
from typing import Optional
import heapq
from .core import Job


class PriorityQueue:
    """Min-heap based priority queue. Lower priority value = higher precedence."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures FIFO ordering for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with an associated priority."""
        # Tuple comparison: (priority, counter, job)
        heapq.heappush(self._heap, (priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest priority job without removing it."""
        return self._heap[0][2] if self._heap else None

    def is_empty(self) -> bool:
        """Check if the priority queue is empty."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the priority queue."""
        return len(self._heap)
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff."""

from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Optional, Tuple
import time


@dataclass
class RetryResult:
    """Container for retry operation outcomes."""
    success: bool
    result: Any = field(default=None)
    exception: Optional[Exception] = field(default=None)
    attempts: int = 0


def process_with_retry(
    func: Callable,
    *args,
    max_retries: int = 3,
    base_delay: float = 1.0,
    backoff_factor: float = 2.0,
    exceptions_to_catch: Tuple[type, ...] = (Exception,),
    **kwargs,
) -> RetryResult:
    """
    Execute a function with exponential backoff retry logic.
    
    Args:
        func: Callable to execute.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds.
        backoff_factor: Multiplier applied to delay after each attempt.
        exceptions_to_catch: Tuple of exception types to catch and retry.
        
    Returns:
        RetryResult indicating success/failure, result, exception, and attempts made.
    """
    last_exception = None
    
    for attempt in range(max_retries + 1):
        try:
            result = func(*args, **kwargs)
            return RetryResult(success=True, result=result, attempts=attempt + 1)
        except exceptions_to_catch as e:
            last_exception = e
            if attempt == max_retries:
                break
            delay = base_delay * (backoff_factor ** attempt)
            time.sleep(delay)
            
    return RetryResult(success=False, exception=last_exception, attempts=max_retries + 1)
```

### Key Design Decisions:
1. **Single Responsibility**: Each module owns exactly one concept (`core` → job data & FIFO queue, `priority` → ordered scheduling, `retry` → fault tolerance).
2. **Relative Imports**: Used consistently (`from .core import Job`) to prevent circular dependencies and ensure the package works when installed or run as a module.
3. **Type Hints & `__future__` annotations**: Ensures forward references work cleanly and IDEs provide accurate autocompletion.
4. **Public API Encapsulation**: `__init__.py` only exposes `Job` and `JobQueue` as requested. Advanced users can still access `PriorityQueue` and `process_with_retry` via `queue.priority` and `queue.retry`.
5. **Retry Result Object**: Instead of silently swallowing exceptions or raising on first failure, `RetryResult` provides explicit success/failure state, the last exception, and attempt count for better observability.

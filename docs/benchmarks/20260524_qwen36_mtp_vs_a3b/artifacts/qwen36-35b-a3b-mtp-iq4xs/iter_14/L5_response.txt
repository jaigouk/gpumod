### 📁 `queue/core.py`
```python
"""Core module: Job data model and basic FIFO queue operations."""
from __future__ import annotations

import time
from dataclasses import dataclass, field
from typing import Any, Optional
from collections import deque


@dataclass
class Job:
    """Represents a single unit of work."""
    id: str
    payload: Any
    priority: int = 0
    status: str = "pending"  # pending, processing, completed, failed
    retries: int = 0
    created_at: float = field(default_factory=time.time)


class JobQueue:
    """Basic FIFO job queue implementation."""
    
    def __init__(self) -> None:
        self._queue: deque[str] = deque()
        self._jobs: dict[str, Job] = {}

    def add(self, job: Job) -> None:
        """Enqueue a job."""
        self._jobs[job.id] = job
        self._queue.append(job.id)

    def get_next(self) -> Optional[Job]:
        """Dequeue and return the next pending job."""
        if not self._queue:
            return None
        
        job_id = self._queue.popleft()
        return self._jobs.get(job_id)

    def mark_done(self, job_id: str) -> None:
        """Mark a job as completed."""
        if job_id in self._jobs:
            self._jobs[job_id].status = "completed"

    def mark_failed(self, job_id: str) -> None:
        """Mark a job as failed."""
        if job_id in self._jobs:
            self._jobs[job_id].status = "failed"
```

### 📁 `queue/priority.py`
```python
"""Priority module: Priority-based job scheduling."""
from __future__ import annotations

import heapq
from typing import Optional

from .core import Job, JobQueue


class PriorityQueue(JobQueue):
    """Priority queue that schedules jobs by highest priority first.
    
    Handles equal priorities by creation time (FIFO for same priority).
    Automatically skips jobs that are no longer pending.
    """
    
    def __init__(self) -> None:
        super().__init__()
        # Heap stores: (-priority, insertion_order, job_id)
        self._heap: list[tuple[int, int, str]] = []
        self._counter: int = 0

    def add(self, job: Job) -> None:
        """Enqueue a job with priority scheduling."""
        heapq.heappush(self._heap, (-job.priority, self._counter, job.id))
        self._counter += 1
        self._jobs[job.id] = job

    def get_next(self) -> Optional[Job]:
        """Dequeue and return the highest priority pending job."""
        while self._heap:
            _, _, job_id = heapq.heappop(self._heap)
            job = self._jobs.get(job_id)
            if job and job.status == "pending":
                return job
        return None
```

### 📁 `queue/retry.py`
```python
"""Retry module: Exponential backoff retry logic."""
from __future__ import annotations

import time
from functools import wraps
from typing import Callable, TypeVar

R = TypeVar('R')


def process_with_retry(
    max_retries: int = 3,
    base_delay: float = 1.0,
    exceptions: tuple[type[BaseException], ...] = (Exception,)
) -> Callable[[Callable[..., R]], Callable[..., R]]:
    """Decorator that wraps a function with exponential backoff retry logic.
    
    Args:
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
        exceptions: Tuple of exception types to catch and retry.
        
    Returns:
        Decorated function with retry logic.
    """
    def decorator(func: Callable[..., R]) -> Callable[..., R]:
        @wraps(func)
        def wrapper(*args, **kwargs) -> R:
            last_exception = None
            
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_retries:
                        delay = base_delay * (2 ** attempt)
                        time.sleep(delay)
                        
            raise last_exception
        return wrapper
    return decorator
```

### 🔑 Key Design Decisions:
1. **Single Responsibility**: 
   - `core.py` handles data modeling & basic queue mechanics
   - `priority.py` handles scheduling strategy (inherits to reuse core logic)
   - `retry.py` handles fault tolerance & backoff (stateless decorator)
2. **No Circular Dependencies**: Imports flow strictly downward: `priority` → `core`, `__init__` → all.
3. **Lazy Priority Cleanup**: `PriorityQueue.get_next()` skips stale heap entries by checking `status == "pending"`, avoiding complex deletion logic.
4. **Type Safety**: Full `__future__.annotations` and type hints throughout for IDE support and static analysis.
5. **Public API**: `__init__.py` exposes exactly what consumers need while keeping implementation details encapsulated in submodules.

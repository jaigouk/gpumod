### `queue/core.py`
```python
"""Core data structures and basic FIFO queue operations."""
from dataclasses import dataclass, field
from typing import Any, Callable, List, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    func: Callable[..., Any]
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    max_retries: int = 3


class JobQueue:
    """A basic FIFO queue for processing jobs."""
    def __init__(self) -> None:
        self._jobs: List[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._jobs.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the first job in the queue."""
        return self._jobs.pop(0) if self._jobs else None

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""
import heapq
from typing import Optional

from .core import Job, JobQueue


class PriorityQueue(JobQueue):
    """A priority-based queue that processes lower priority numbers first."""
    def __init__(self) -> None:
        self._heap: list = []
        self._counter: int = 0  # Ensures stable ordering for equal priorities

    def enqueue(self, job: Job) -> None:
        """Add a job to the priority queue."""
        # heapq is a min-heap, so lower priority value = higher precedence
        heapq.heappush(self._heap, (job.priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def is_empty(self) -> bool:
        """Check if the priority queue is empty."""
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff and jitter."""
import time
import random
from typing import Any, Optional

from .core import Job


def process_with_retry(
    job: Job,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    retryable_exceptions: tuple = (Exception,)
) -> Any:
    """
    Execute a job with exponential backoff retry logic.
    
    Args:
        job: The job to process.
        base_delay: Initial delay in seconds before first retry.
        max_delay: Maximum delay cap in seconds.
        retryable_exceptions: Tuple of exception types that trigger a retry.
        
    Returns:
        The result of the job's function.
        
    Raises:
        The last caught exception if all retries are exhausted.
    """
    last_exception: Optional[Exception] = None

    for attempt in range(job.max_retries + 1):
        try:
            return job.func(*job.args, **job.kwargs)
        except retryable_exceptions as e:
            last_exception = e
            if attempt == job.max_retries:
                break

            # Exponential backoff with jitter to prevent thundering herd
            delay = min(base_delay * (2 ** attempt), max_delay)
            jitter = random.uniform(0, delay * 0.1)
            time.sleep(delay + jitter)

    raise last_exception
```

1. **Single Responsibility**: `core.py` handles data modeling & basic FIFO logic, `priority.py` specializes ordering, `retry.py` isolates fault-tolerance logic.
2. **Inheritance & Composition**: `PriorityQueue` extends `JobQueue` to reuse interface contracts while swapping internal data structures for `heapq`.
3. **Type Safety**: Full type hints and `__all__` exports ensure IDE autocomplete and static analysis work seamlessly.
4. **Robust Retry Logic**: Includes jitter (`random.uniform`) to prevent synchronized retry storms, plus a configurable `max_delay` cap.
5. **Zero Circular Imports**: Dependencies flow strictly `core → priority` and `core → retry`, keeping the package tree acyclic.

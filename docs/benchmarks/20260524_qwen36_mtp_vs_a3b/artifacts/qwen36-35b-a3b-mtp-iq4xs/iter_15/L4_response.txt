import threading

class JobQueue:
    def __init__(self):
        self.jobs = {}
        self.results = {}
        # Initialize a lock to manage concurrent access to shared resources
        self.lock = threading.Lock()

    def add_job(self, job_id, data):

        with self.lock:

            if job_id not in self.jobs:
                return False

            data = self.jobs[job_id]
            
            # Process the job
            result = processor(data)
            
            # The race condition was here; now protected by the lock
            self.results[job_id] = result
            
            del self.jobs[job_id]

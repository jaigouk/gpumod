### `queue/core.py`
```python
"""Core job data model and FIFO queue implementation."""

from __future__ import annotations
import uuid
from dataclasses import dataclass, field
from typing import Any, Callable, Optional
from collections import deque


@dataclass
class Job:
    """Represents a unit of work."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    status: str = "pending"
    retries: int = 0
    max_retries: int = 3


class JobQueue:
    """FIFO queue for managing jobs."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def add(self, job: Job) -> None:
        """Add a job to the queue."""
        if job.status != "pending":
            raise ValueError(f"Job {job.id} is not pending (status: {job.status})")
        self._queue.append(job)

    def get_next(self) -> Optional[Job]:
        """Retrieve and remove the next job in FIFO order."""
        return self._queue.popleft() if self._queue else None

    def process(self, job: Job, handler: Callable[[Any], Any]) -> Any:
        """Execute a job using the provided handler function."""
        if job.status != "pending":
            raise ValueError(f"Job {job.id} is not pending (status: {job.status})")
        try:
            result = handler(job.payload)
            job.status = "completed"
            return result
        except Exception as e:
            job.status = "failed"
            raise e

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._queue) == 0
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

import heapq
from typing import Any, Callable, Optional

from .core import Job


class PriorityQueue:
    """Priority-based queue where lower integers indicate higher priority."""

    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures stable ordering for equal priorities

    def add(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority."""
        if job.status != "pending":
            raise ValueError(f"Job {job.id} is not pending (status: {job.status})")
        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def get_next(self) -> Optional[Job]:
        """Retrieve and remove the highest priority job."""
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def process(self, job: Job, handler: Callable[[Any], Any]) -> Any:
        """Execute a job using the provided handler function."""
        if job.status != "pending":
            raise ValueError(f"Job {job.id} is not pending (status: {job.status})")
        try:
            result = handler(job.payload)
            job.status = "completed"
            return result
        except Exception as e:
            job.status = "failed"
            raise e

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff for job processing."""

import time
from typing import Any, Callable

from .core import Job


def process_with_retry(
    job: Job,
    handler: Callable[[Any], Any],
    base_delay: float = 1.0,
    max_delay: float = 60.0,
) -> Any:
    """
    Process a job with automatic retry and exponential backoff.
    
    Args:
        job: The job to process.
        handler: Callable that takes job.payload and returns the result.
        base_delay: Initial delay in seconds before the first retry.
        max_delay: Maximum delay in seconds between retries.
        
    Returns:
        The result of the handler on success.
        
    Raises:
        Exception: The last exception raised by the handler after all retries are exhausted.
    """
    last_exception = None
    
    for attempt in range(job.max_retries + 1):
        try:
            return handler(job.payload)
        except Exception as e:
            last_exception = e
            job.retries += 1
            job.status = "failed"
            
            if attempt == job.max_retries:
                break
                
            # Exponential backoff: base_delay * 2^attempt, capped at max_delay
            delay = min(base_delay * (2 ** attempt), max_delay)
            time.sleep(delay)
            
    raise last_exception
```

### Key Design Decisions:
1. **Single Responsibility**: `core.py` handles data modeling & FIFO logic, `priority.py` handles heap-based ordering, `retry.py` handles failure recovery & timing, and `__init__.py` acts as the facade.
2. **No Circular Imports**: All cross-module dependencies use relative imports pointing to `core.py`, which has no dependencies on other package modules.
3. **Type Safety & Modern Python**: Uses `dataclass`, `typing`, and Python 3.9+ syntax (`list[...]`, `tuple[...]`). Added `from __future__ import annotations` in `core.py` for forward compatibility.
4. **Stable Priority Queue**: Uses a monotonic counter as a tie-breaker in the heap tuple `(priority, counter, job)` to prevent `TypeError` when comparing `Job` objects with equal priorities.
5. **Clean Public API**: `__init__.py` explicitly defines `__all__` to prevent namespace pollution and clearly signals the intended public interface.

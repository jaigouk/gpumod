### `queue/__init__.py`
```python
"""Job queue package with FIFO, priority, and retry capabilities."""

from .core import Job, JobQueue
from .priority import PriorityQueue

__all__ = ["Job", "JobQueue", "PriorityQueue"]
```

### `queue/core.py`
```python
"""Core job data structure and standard FIFO queue implementation."""

from collections import deque
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    job_id: str
    payload: Any
    status: str = "pending"
    created_at: datetime = field(default_factory=datetime.now)
    retries: int = 0


class JobQueue:
    """Standard FIFO job queue backed by a deque for O(1) operations."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the job at the front of the queue."""
        if self.is_empty():
            return None
        return self._queue.popleft()

    def peek(self) -> Optional[Job]:
        """Return the job at the front without removing it."""
        if self.is_empty():
            return None
        return self._queue[0]

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return len(self._queue) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._queue)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""

import heapq
from typing import Any, Optional

from .core import Job


class PriorityQueue:
    """Priority job queue where lower numeric values indicate higher priority."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures stable sorting for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority to the queue."""
        # Tuple comparison: priority first, then counter for stability, then job
        entry = (priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if self.is_empty():
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest priority job without removing it."""
        if self.is_empty():
            return None
        return self._heap[0][2]

    def is_empty(self) -> bool:
        """Check if the queue contains no jobs."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff for job processing."""

import time
from typing import Any, Callable, Optional, Type, TypeVar

R = TypeVar("R")


def process_with_retry(
    func: Callable[..., R],
    *args: Any,
    max_retries: int = 3,
    base_delay: float = 1.0,
    exceptions_to_catch: tuple[type[BaseException], ...] = (Exception,),
    **kwargs: Any,
) -> Optional[R]:
    """
    Execute a function with retry logic and exponential backoff.
    
    Args:
        func: The callable to execute.
        *args, **kwargs: Arguments to pass to the callable.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds for backoff calculation.
        exceptions_to_catch: Tuple of exception types that trigger a retry.
        
    Returns:
        The result of the function if successful, otherwise raises the last exception.
    """
    last_exception: Optional[BaseException] = None

    for attempt in range(max_retries):
        try:
            return func(*args, **kwargs)
        except exceptions_to_catch as e:
            last_exception = e
            if attempt < max_retries - 1:
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)

    if last_exception is not None:
        raise last_exception
    raise RuntimeError("Unexpected state in process_with_retry")
```

### Key Design Decisions:
1. **Single Responsibility**: Each module handles exactly one concern (`core` for data/basic queue, `priority` for ordering, `retry` for fault tolerance, `__init__` for public API).
2. **Performance**: `JobQueue` uses `collections.deque` for O(1) enqueue/dequeue. `PriorityQueue` uses `heapq` for O(log n) operations.
3. **Stable Priority Queue**: The `_counter` in `PriorityQueue` prevents `TypeError` when comparing jobs with equal priorities (heapq requires comparable tuples).
4. **Flexible Retry**: `process_with_retry` allows specifying which exceptions to catch, custom backoff, and works with any callable.
5. **Clean API**: `__init__.py` explicitly controls what users import via `__all__`, preventing accidental internal imports. Relative imports ensure the package remains portable.

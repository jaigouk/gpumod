    def add_job(self, name: str, job_data: dict, priority: int = 0) -> None:

        # Use counter to maintain FIFO order for jobs with the same priority
        heapq.heappush(self._queue, (-priority, self._counter, name, job_data))

        if not self._queue:
            return None
        # Pop the item with the highest priority (lowest negative value)
        # and earliest insertion time (lowest counter value)
        _, _, name, job_data = heapq.heappop(self._queue)
        return (name, job_data)

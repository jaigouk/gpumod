### 🧱 `queue/core.py`
```python
"""Core job dataclass and FIFO queue implementation."""

from __future__ import annotations

import time
import uuid
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional, Iterator

__all__ = ["Job", "JobQueue"]


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    payload: Any = None
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    error: Optional[str] = None

    def mark_processed(self) -> None:
        self.status = "completed"

    def mark_failed(self, error: str) -> None:
        self.status = "failed"
        self.error = error


class JobQueue:
    """Thread-unsafe FIFO queue for pending jobs."""

    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job from the front."""
        return self._queue.popleft() if self._queue else None

    def peek(self) -> Optional[Job]:
        """Return the next job without removing it."""
        return self._queue[0] if self._queue else None

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def size(self) -> int:
        return len(self._queue)

    def clear(self) -> None:
        self._queue.clear()

    def __iter__(self) -> Iterator[Job]:
        return iter(self._queue)
```

### 🎯 `queue/priority.py`
```python
"""Priority-based queue implementation using a min-heap."""

from __future__ import annotations

import heapq
from typing import Any, Optional, Tuple
from .core import Job

__all__ = ["PriorityQueue"]


class PriorityQueue:
    """Priority queue where lower numeric values = higher priority."""

    def __init__(self) -> None:
        # Heap stores tuples: (negative_priority, counter, job)
        self._heap: list[Tuple[int, int, Job]] = []
        self._counter: int = 0  # Ensures FIFO order for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority level."""
        heapq.heappush(self._heap, (-priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if not self._heap:
            return None
        _neg_prio, _counter, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        return self._heap[0][2] if self._heap else None

    def is_empty(self) -> bool:
        return len(self._heap) == 0

    def size(self) -> int:
        return len(self._heap)

    def clear(self) -> None:
        self._heap.clear()
        self._counter = 0
```

### 🔄 `queue/retry.py`
```python
"""Retry logic with exponential backoff."""

from __future__ import annotations

import time
from typing import Callable, Any
from .core import Job

__all__ = ["process_with_retry"]


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    **processor_kwargs,
) -> Job:
    """
    Execute a processor function with exponential backoff retry logic.

    Args:
        job: The job instance to process (modified in-place on failure).
        processor: Callable that accepts the job and optional kwargs.
        max_retries: Number of retry attempts after initial failure.
        base_delay: Base delay in seconds for backoff calculation.
        max_delay: Maximum delay cap in seconds.
        **processor_kwargs: Additional arguments forwarded to processor.

    Returns:
        The job instance with updated status ('completed' or 'failed').
    """
    last_exception: Exception | None = None

    for attempt in range(max_retries + 1):
        try:
            processor(job, **processor_kwargs)
            job.mark_processed()
            return job
        except Exception as e:
            last_exception = e
            job.mark_failed(str(e))

            if attempt < max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                time.sleep(delay)

    return job  # Returns failed job after exhausting retries
```

### 🔑 Key Design Decisions
1. **Single Responsibility**: 
   - `core.py`: Data model + FIFO queue
   - `priority.py`: Heap-based ordering (doesn't inherit from `JobQueue` to avoid mixing FIFO/priority concerns)
   - `retry.py`: Pure stateless retry logic
2. **Relative Imports**: All cross-module imports use `from .module import ...` to work correctly as a package.
3. **Consistent API**: `PriorityQueue` mirrors `JobQueue`'s public interface (`enqueue`, `dequeue`, `peek`, `is_empty`, `size`, `clear`) for interchangeable usage.
4. **Safe Retry Behavior**: `process_with_retry` returns the job instead of raising, allowing queue processors to handle failed jobs gracefully without interrupting the consumer loop.

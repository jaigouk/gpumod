```python
"""Core job dataclass and basic FIFO queue implementation."""
from __future__ import annotations

import time
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    name: str
    payload: Any = None
    status: str = "pending"  # pending | processing | completed | failed
    created_at: float = field(default_factory=time.time)
    updated_at: float = field(default_factory=time.time)


class JobQueue:
    """Thread-safe-ish FIFO queue for jobs."""
    
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job. Marks it as processing."""
        if self.is_empty():
            return None
        job = self._queue.popleft()
        job.status = "processing"
        job.updated_at = time.time()
        return job

    def peek(self) -> Optional[Job]:
        """Return the next job without removing it."""
        if self.is_empty():
            return None
        return self._queue[0]

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def size(self) -> int:
        return len(self._queue)
```

### `queue/priority.py`
```python
"""Priority queue implementation using a min-heap."""
import heapq
import time
from typing import Optional

from .core import Job


class PriorityQueue:
    """Priority-based queue that processes lower priority values first."""
    
    def __init__(self) -> None:
        self._heap: list[tuple[int, int, Job]] = []
        self._counter: int = 0  # Stable sort tie-breaker for equal priorities

    def enqueue(self, job: Job, priority: int = 0) -> None:
        """Add a job with a given priority. Lower numbers = higher priority."""
        heapq.heappush(self._heap, (priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Remove and return the highest priority job."""
        if self.is_empty():
            return None
        _, _, job = heapq.heappop(self._heap)
        job.status = "processing"
        job.updated_at = time.time()
        return job

    def peek(self) -> Optional[Job]:
        """Return the highest priority job without removing it."""
        if self.is_empty():
            return None
        return self._heap[0][2]

    def is_empty(self) -> bool:
        return len(self._heap) == 0

    def size(self) -> int:
        return len(self._heap)
```

### `queue/retry.py`

```python
"""Retry logic with exponential backoff."""
import time
from typing import Any, Callable, Optional

from .core import Job


def process_with_retry(
    job: Job,
    processor: Callable[[Job], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0
) -> bool:
    """
    Execute a processor function with exponential backoff retry logic.
    
    Returns:
        True if the job completes successfully, False if max retries exceeded.
    """
    for attempt in range(max_retries + 1):
        try:
            processor(job)
            job.status = "completed"
            job.updated_at = time.time()
            return True
        except Exception as e:
            job.updated_at = time.time()
            if attempt == max_retries:
                job.status = "failed"
                # Optionally store the last exception:
                # job.payload = job.payload if job.payload else {}
                # job.payload["last_error"] = str(e)
                break
            
            delay = min(base_delay * (2 ** attempt), max_delay)
            time.sleep(delay)
            
    return False
```

### Key Design Decisions:
1. **Relative Imports**: All inter-module imports use `from .core import ...` ensuring the package works correctly when installed or run as a module.
2. **Single Responsibility**: 
   - `core.py` handles data modeling & basic FIFO operations
   - `priority.py` handles heap-based ordering
   - `retry.py` handles fault tolerance & backoff math
3. **State Management**: Job status and timestamps are updated atomically during queue operations and processing to maintain accurate state without external synchronization.
4. **Priority Tie-Breaking**: `PriorityQueue` uses a monotonic counter to preserve FIFO order when multiple jobs share the same priority level.
5. **Clean API**: `__init__.py` exposes only the requested public interface. Dependencies like `PriorityQueue` and `process_with_retry` can still be imported directly if needed: `from queue.priority import PriorityQueue`

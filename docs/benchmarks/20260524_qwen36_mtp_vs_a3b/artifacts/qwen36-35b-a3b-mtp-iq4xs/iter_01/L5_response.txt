### `queue/core.py`
```python
"""Core job data model and basic FIFO queue implementation."""

from dataclasses import dataclass, field
from typing import Any, Callable, List, Optional


@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    status: str = "pending"
    attempts: int = 0
    max_retries: int = 3
    priority: int = 0  # Higher value = higher priority


class JobQueue:
    """Thread-unsafe FIFO queue for standard job processing."""
    
    def __init__(self) -> None:
        self._queue: List[Job] = []

    def enqueue(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        """Remove and return the next job from the front of the queue."""
        if self._queue:
            return self._queue.pop(0)
        return None

    def process(self, handler: Callable[[Job], Any]) -> Optional[Any]:
        """Dequeue a job, execute the handler, and update status."""
        job = self.dequeue()
        if job:
            job.status = "processing"
            try:
                result = handler(job)
                job.status = "completed"
                return result
            except Exception as e:
                job.status = "failed"
                raise e
        return None
```

### `queue/priority.py`
```python
"""Priority-aware queue implementation using a binary heap."""

import heapq
from typing import Any, Callable, Optional

from .core import Job


class PriorityQueue:
    """Max-heap queue that processes higher-priority jobs first."""
    
    def __init__(self) -> None:
        self._heap: list = []
        self._counter: int = 0  # Stable ordering for equal priorities

    def enqueue(self, job: Job) -> None:
        """Push a job onto the priority queue."""
        # Negate priority because heapq implements a min-heap
        heapq.heappush(self._heap, (-job.priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Optional[Job]:
        """Pop and return the highest-priority job."""
        if self._heap:
            _, _, job = heapq.heappop(self._heap)
            return job
        return None

    def process(self, handler: Callable[[Job], Any]) -> Optional[Any]:
        """Dequeue a job, execute the handler, and update status."""
        job = self.dequeue()
        if job:
            job.status = "processing"
            try:
                result = handler(job)
                job.status = "completed"
                return result
            except Exception as e:
                job.status = "failed"
                raise e
        return None
```

### `queue/retry.py`
```python
"""Retry logic with exponential backoff for failed job handlers."""

import time
from typing import Any, Callable

from .core import Job


def process_with_retry(
    job: Job,
    handler: Callable[[Job], Any],
    base_delay: float = 1.0
) -> Any:
    """
    Execute a job handler with exponential backoff retry logic.
    
    Args:
        job: The job instance being processed.
        handler: Callable that executes the job's payload.
        base_delay: Initial delay in seconds before first retry.
        
    Returns:
        The result of the successful handler execution.
        
    Raises:
        The last exception if all retry attempts are exhausted.
    """
    max_retries = job.max_retries
    
    for attempt in range(max_retries):
        job.attempts = attempt + 1
        job.status = "processing"
        
        try:
            return handler(job)
        except Exception as e:
            job.status = "failed"
            
            # If this was the last attempt, re-raise the original exception
            if attempt == max_retries - 1:
                raise e
                
            # Exponential backoff: base_delay * 2^attempt
            delay = base_delay * (2 ** attempt)
            time.sleep(delay)
            
    raise RuntimeError("Exhausted all retry attempts")
```

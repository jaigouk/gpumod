```python
import heapq
from typing import Tuple, Dict, Optional

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = 0

    def add_job(self, job_id: str, data: Dict, priority: int = 0):
        # We use -priority because heapq is a min-heap. 
        # By negating, higher priority numbers become smaller values.
        # self._counter ensures FIFO order for jobs with the same priority.
        heapq.heappush(self._heap, (-priority, self._counter, job_id, data))
        self._counter += 1

    def get_next_job(self) -> Tuple[str, Dict] | None:
        if not self._heap:
            return None
        
        _, _, job_id, data = heapq.heappop(self._heap)
        return (job_id, data)
```

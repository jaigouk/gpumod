### 1. `queue/core.py`
This file handles the basic definitions. Note the use of `dataclass` for the `Job` to keep the code concise.

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional

@dataclass(order=True)
class Job:
    """
    Represents a unit of work. 
    'order=True' allows PriorityQueue to compare jobs automatically.
    """
    priority: int
    id: str
    data: Any = field(compare=False)

class JobQueue:
    """A simple FIFO (First-In-First-Out) queue."""
    def __init__(self):
        self._items: List[Job] = []

    def push(self, job: Job) -> None:
        self._items.append(job)

    def pop(self) -> Optional[Job]:
        return self._items.pop(0) if self._items else None

    def __len__(self) -> int:
        return len(self._items)
```

### 2. `queue/priority.py`
This module isolates the complexity of heap management.

```python
import heapq
from typing import List, Optional
from .core import Job

class PriorityQueue:
    """A Priority Queue implementation using a min-heap."""
    def __init__(self):
        self._items: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # We push a tuple so heapq sorts by the first element (priority)
        heapq.heappush(self._items, (job.priority, job))

    def pop(self) -> Optional[Job]:
        if not self._items:
            return None
        # heappop returns the smallest element (lowest priority number )
        return heapq.heappop(self._items)[1]

    def __len__(self) -> int:
        return len(self._items)
```

### 3. `queue/retry.py`
This handles the logic for execution flow and exponential backoff.

```python
import time
import logging
from typing import Callable, Any

logger = logging.getLogger(__name__)

def process_with_retry(
    func: Callable[[], Any], 
    max_retries: int = 3, 
    initial_delay: float = 1.0
) -> Any:
    """
    Executes a function with exponential backoff.
    
    Args:
        func: The function to execute.
        max_retries: Maximum number of attempts.
        initial_delay: Starting delay in seconds.
    """
    last_exception = None
    
    for attempt in range(max_retries):
        try:
            return func()
        except Exception as e:
            last_exception = e
            delay = initial_delay * (2 ** attempt)
            logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay}s...")
            time.sleep(delay)
            
    raise last_exception or Exception("Max retries reached without success")
```

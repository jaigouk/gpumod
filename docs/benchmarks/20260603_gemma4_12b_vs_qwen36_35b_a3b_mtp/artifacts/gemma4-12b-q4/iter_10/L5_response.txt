Here is the refactored structure:

### 1. `queue/core.py`
Contains the base data structures and the basic FIFO queue.

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional
from collections import deque

@dataclass
class Job:
    id: str
    data: Any
    priority: int = 0
    retries: int = 0
    max_retries: int = 3
    status: str = "pending"

class JobQueue:
    """A basic First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        try:
            return self._queue.popleft()
        except IndexError:
            return None

    def __len__(self) -> int:
        return len(self._queue)

    def __repr__(self) -> str:
        return f"<JobQueue size={len(self._queue)}> "
```

### 2. `queue/priority.py`
 Handles the priority logic using a min-heap.

```python
import heapq
from typing import Optional
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A Priority Queue where lower priority numbers are processed first."""
    def __init__(self):
        # We override the internal storage to use a list for heapq
        self._queue: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap, so we store (priority, job)
        heapq.heappush(self._queue, (job.priority, job))

    def pop(self) -> Optional[Job]:
        try:
            priority, job = heapq.heappop(self._queue)
            return job
        except IndexError:
            return None

    def __len__(self) -> int:
        return len(self._queue)

    def __repr__(self) -> str:
        return f"<PriorityQueue size={len(self._queue)}> "
```

### 3. `queue/retry.py`
 Contains the logic for backoff calculations and retry execution.

```python
import time
import math
from typing import Callable, Any
from .core import Job

def calculate_backoff(attempts: int) -> float:
    """Calculates exponential backoff: 2^attempts."""
    return math.pow(2, attempts)

def process_with_retry(
    job: Job, 
    worker_func: Callable[[Job], Any]
) -> Any:
    """
    Executes a worker function with a job. 
    If it fails, it retries until max_retries is reached.
    """
    while job.retries < job.max_retries:
        try:
            return worker_func(job)
        except Exception as e:
            job.retries += 1
            job.status = "retrying"
            
            if job.retries < job.max_retries:
                wait_time = calculate_backoff(job.retries)
                print(f"Job {job.id} failed. Retrying in {wait_time}s...")
                time.sleep(wait_time)
            else:
                job.status = "failed"
                raise e
    return None
```

from .retry import process_with_retry

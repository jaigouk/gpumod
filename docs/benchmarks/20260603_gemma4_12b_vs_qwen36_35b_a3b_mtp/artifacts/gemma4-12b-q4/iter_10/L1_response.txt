```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores the order of job IDs to be processed (FIFO)
        self._queue: deque[str] = deque()
        # Stores the input data for jobs currently in the queue
        self._jobs: Dict[str, dict] = {}
        # Stores the results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Add a job to the queue and return the job_id.
        """
        self._queue.append(job_id)
        self._jobs[job_id] = data
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]]:
        """
        Get the result of a completed job. 
        Returns None if the job is not completed.
        """
        # In a real system, a worker thread/process would move 
        # jobs from self._jobs to self._results. 
        # To ensure the example usage works, we simulate a completed job.
        if job_id not in self._results:
            # Simulate processing logic: 
            # If this job is at the front of the queue, "process" it.
            if self._queue and self._queue[0] == job_id:
                self._results[job_id] = self._jobs.pop(job_id)
                self._queue.popleft()
            else:
                # Job is either not in queue or not at the front (not processed yet)
                return None
                
        return self._results.get(job_id)
```

### 1. `queue/core.py`
This file contains the base definitions. It defines what a `Job` is and the base requirements for any `JobQueue`.

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional
from abc import ABC, abstractmethod

@dataclass(order=True)
class Job:
    """Data class representing a unit of work."""
    priority: int
    id: str
    task_name: str
    data: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue(ABC):
    """Abstract Base Class for Job Queue operations."""
    def __init__(self):
        self._items: List[Job] = []

    @abstractmethod
    def push(self, job: Job) -> None:
        """Add a job to the queue."""
        pass

    @abstractmethod
    def pop(self) -> Optional[Job]:
        """Remove and return the next job from the queue."""
        pass

    def __len__(self) -> int:
        return len(self._items)
```

### 2. `queue/priority.py`
This handles the specific implementation of a Priority Queue using a heap (binary tree) for $O(\log n)$ performance.

```python
import heapq
from typing import Optional
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority queue implementation using a min-heap."""
    def __init__(self):
        super().__init__()
        # We maintain a separate list for heapq operations
        self._heap: list[Job] = []

    def push(self, job: Job) -> None:
        """Push a job onto the heap."""
        heapq.heappush(self._heap, job)
        self._items = self._heap # Syncing for len() support

    def pop(self) -> Optional[Job]:
        """Pop the highest priority job (lowest priority number)."""
        if not self._heap:
            return None
        return heapq.heappop(self._heap)
```

### 3. `queue/retry.py`
This encapsulates the logic for handling failures and calculating backoff delays.

```python
import time
import random
from typing import Callable
from .core import Job

def calculate_backoff(retry_count: int) -> float:
    """Calculate exponential backoff with jitter."""
    base_delay = 1.0
    max_delay = 32.0
    delay = min(max_delay, base_delay * (2 ** retry_count))
    return delay + random.uniform(0, 0.1 * delay)

def process_with_retry(
    job: Job, 
    worker_func: Callable[[Job], bool], 
    max_retries: int = 3
) -> bool:
    """
    Executes a worker function against a job. 
    If it fails, it sleeps and retries until max_retries is reached.
    """
    while job.retries <= max_retries:
        try:
            success = worker_func(job)
            if success:
                return True
            else:
                raise Exception("Worker returned False")
        except Exception as e:
            job.retries += 1
            if job.retries > max_retries:
                print(f"Job {job.id} failed after {max_retries} retries.")
                return False
            
            delay = calculate_backoff(job.retries)
            print(f"Retry {job.retries} for job {job.id} in {delay:.2f}s...")
            time.sleep(delay)
            
    return False
```

### Summary of Changes & Design Decisions:

1.  **Separation of Concerns**: 
    *   `core.py` doesn't know how sorting works; it only knows what a `Job` looks like.
    *   `priority.py` doesn't know about retries; it only knows how to sort items.
    *   `retry.py` doesn't know about the internal structure of the queue; it only knows how to handle a single `Job` failure.
2.  **Relative Imports**: Used `from .core import ...` to ensure the package remains portable.
3.  **Complexity**: Used `heapq` in `priority.py` to ensure that pushing/popping remains efficient even as the queue grows.
4.  **Public API**: By including `__all__` in `__init__.py`, we ensure that `from queue import Job` works perfectly, keeping the implementation details hidden from the end-user.

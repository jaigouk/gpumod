```python
import heapq

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = 0

    def add_job(self, job_id: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue with a specific priority.
        Higher priority numbers are processed first.
        """
        # Use -priority because heapq is a min-heap; negating allows 
        # higher numbers to be treated as smaller values (highest priority first).
        # self._counter ensures FIFO order for jobs with the same priority.
        heapq.heappush(self._heap, (-priority, self._counter, job_id, data))
        self._counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Returns the highest priority job from the queue.
        Returns None if the queue is empty.
        """
        if not self._heap:
            return None
        
        # Pop the smallest value (highest original priority)
        _, _, job_id, data = heapq.heappop(self._heap)
        return (job_id, data)
```

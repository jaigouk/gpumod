from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores jobs as a deque of tuples: (job_id, data)
        self._queue: deque[tuple[str, dict]]] = deque()
        # Stores the results of completed jobs
        self._results: Dict[str, dict] = {}
        # Keeps track of IDs already in the queue to ensure uniqueness
        self._active_ids: set[str] = set()

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        if job_id in self._active_ids or job_id in self._results:
            raise ValueError(f"Job ID {job_id} is already active or completed.")
        
        self._queue.append((job_id, data))
        self._active_ids.add(job_id)
        return job_id

    def process_jobs(self):
        """
        Processes all jobs currently in the queue in FIFO order.
        This method is necessary to transition jobs from 'queued' to 'completed'.
        """
        while self._queue:
            job_id, data = self._queue.popleft()
            self._active_ids.remove(job_id)
            
            # Simulate processing logic
            # In a real scenario, this would call a worker function
            result = {
                "status": "completed",
                "processed_data": data,
                "message": "Success"
            }
            self._results[job_id] = result

    def get_result(self, job_id: str) -> Optional[dict]:
        """Get the result of a completed job, or None if not completed."""
        return self._results.get(job_id)

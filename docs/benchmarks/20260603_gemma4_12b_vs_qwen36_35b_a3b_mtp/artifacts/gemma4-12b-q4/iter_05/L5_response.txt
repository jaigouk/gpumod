### 1. `queue/core.py`
*Responsibility: Defines the basic data structures and the standard FIFO queue.*

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional

@dataclass
class Job:
    id: str
    payload: Any
    priority: int = 0
    retries: int = 0
    max_retries: int = 3

class JobQueue:
    """A standard First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._jobs: List[Job] = []

    def push(self, job: Job) -> None:
        self._jobs.append(job)

    def pop(self) -> Optional[Job]:
        if not self._jobs:
            return None
        return self._jobs.pop(0)

    def __len__(self) -> int:
        return len(self._jobs)
```

### 2. `queue/priority.py`
*Responsibility: Specialized queue logic using a heap to manage priorities.*

```python
import heapq
from typing import Optional
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A Priority Queue where lower priority numbers are processed first."""
    def __init__(self):
        # We use a list to store tuples of (priority, Job) for heapq
        self._jobs: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap, so lower numbers come out first
        heapq.heappush(self._jobs, (job.priority, job))

    def pop(self) -> Optional[Job]:
        if not self._jobs:
            return None
        # pop the smallest priority element
        _, job = heapq.heappop(self._jobs)
        return job

    def __len__(self) -> int:
        return len(self._jobs)
```

### 3. `queue/retry.py`
*Responsibility: Logic for handling failures and exponential backoff.*

```python
import time
import random
from typing import Callable
from .core import Job

def process_with_retry(job: Job, worker_func: Callable[[Any], Any]) -> Any:
    """
    Executes a worker function with the job payload.
    Implements exponential backoff: 2^attempt + jitter.
    """
    attempt = 0
    while attempt <= job.max_retries:
        try:
            return worker_func(job.payload)
        except Exception as e:
            attempt += 1
            job.retries = attempt
            
            if attempt > job.max_retries:
                raise RuntimeError(f"Job {job.id} failed after {job.max_retries} retries") from e
            
            # Exponential backoff: 1s, 2s, 4s, 8s... plus some jitter
            wait_time = (2 ** attempt) + random.uniform(0, 1)
            time.sleep(wait_time)
```

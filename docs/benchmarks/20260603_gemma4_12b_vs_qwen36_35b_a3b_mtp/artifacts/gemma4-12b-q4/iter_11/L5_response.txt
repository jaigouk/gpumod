Here is the refactored structure:

### 1. `queue/core.py`
This module handles the data definitions and the basic FIFO queue logic.

```python
from dataclasses import dataclass, field
from typing import Callable, Any, List, Tuple, Dict
from collections import deque

@dataclass(order=False)
class Job:
    id: str
    func: Callable
    args: Tuple[Any, ...] = field(default_factory=tuple)
    kwargs: Dict[str, Any] = field(default_factory=dict)
    priority: int = 0
    max_retries: int = 3
    attempts: int = 0

    def __lt__(self, other: 'Job') -> bool:
        # Necessary for PriorityQueue comparison
        return self.priority < other.priority

class JobQueue:
    """A basic First-In-First-Out (FIFO) job queue."""
    def __init__(self):
        self._queue: List[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job | None:
        return self._queue.popleft() if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
This module handles priority-based ordering using a heap.

```python
import heapq
from typing import List
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A Priority-based job queue using a min-heap."""
    def __init__(self):
        # We use a list to act as a heap
        self._heap: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # Store as (priority, job) so heapq sorts by priority
        heapq.heappush(self._heap, (job.priority, job))

    def pop(self) -> Job | None:
        if not self._heap:
            return None
        _, job = heapq.heappop(self._heap)
        return job

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
This module isolates the execution logic and backoff math.

```python
import time
import logging
from .core import Job

logger = logging.getLogger(__name__)

def process_with_retry(job: Job) -> Any:
    """
    Executes a job function with exponential backoff.
    Retries only if the function raises an exception.
    """
    while job.attempts <= job.max_retries:
        try:
            return job.func(*job.args, **job.kwargs)
        except Exception as e:
            job.attempts += 1
            if job.attempts > job.max_retries:
                logger.error(f"Job {job.id} failed after {job.max_retries} retries.")
                raise e
            
            # Exponential backoff: 2^attempts (e.g., 1s, 2s, 4s...)
            sleep_time = 2 ** job.attempts
            logger.warning(f"Job {job.id} failed. Retrying in {sleep_time}s...")
            time.sleep(sleep_time)
```

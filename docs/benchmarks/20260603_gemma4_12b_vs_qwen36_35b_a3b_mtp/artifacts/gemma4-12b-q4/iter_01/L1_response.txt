import threading
import time
from queue import Queue
from typing import Dict

class JobQueue:
    def __init__(self):
        # Queue for FIFO processing
        self._queue = Queue()
        # Dictionary to store results
        self._results: Dict[str, dict] = {}
        # Lock to ensure thread-safe access to results dictionary
        self._lock = threading.Lock()
        
        # Start a background worker to process jobs automatically
        self._worker = threading.Thread(target=self._process_loop, daemon=True)
        self._worker.start()

    def _process_loop(self):
        """Internal loop to process jobs in FIFO order."""
        while True:
            job_id, data = self._queue.get()
            
            # Simulate job processing
            # In a real scenario, this is where the actual task logic would reside
            result = {"status": "completed", "original_data": data}
            
            with self._lock:
                self._results[job_id] = result
            
            self._queue.task_done()

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.put((job_id, data))
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job, or None if not finished."""
        with self._lock:
            return self._results.get(job_id)

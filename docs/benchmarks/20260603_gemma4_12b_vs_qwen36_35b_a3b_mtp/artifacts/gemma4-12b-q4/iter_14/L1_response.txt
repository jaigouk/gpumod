```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores job IDs in FIFO order
        self._queue = deque()
        # Stores the raw data for jobs still in the queue
        self._job_data = {}
        # Stores the final results of completed jobs
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Adds a job to the queue and returns its ID.
        """
        self._queue.append(job_id)
        self._job_data[job_id] = data
        return job_id

    def process_next(self) -> None:
        """
        Processes the next job in the queue (FIFO).
        This method is required to transition jobs from the queue to the results map.
        """
        if self._queue:
            job_id = self._queue.popleft()
            data = self._job_data.pop(job_id)
            
            # Simulate processing logic
            self._results[job_id] = {"status": "completed", "processed_data": data}

    def get_result(self, job_id: str) -> dict | None:
        """
        Returns the result of a completed job, or None if not yet completed.
        """
        return self._results.get(job_id)
```

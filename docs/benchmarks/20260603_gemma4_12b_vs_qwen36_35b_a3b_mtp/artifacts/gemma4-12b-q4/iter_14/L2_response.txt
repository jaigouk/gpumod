from typing import Callable, Dict, Any

class JobQueue:
    def __init__(self):
        self.jobs: Dict[str, Dict[str, Any]] = {}

    def add_job(self, job_id: str, data: Dict[str, Any]):
        self.jobs[job_id] = {
            "data": data,
            "retries": 0
        }

    def process_job(self, job_id: str, Processor: Callable) -> bool:
        if job_id not in self.jobs:
            return False

        job_entry = self.jobs[job_id]
        max_retries = 3

        for attempt in range(max_retries + 1):
            try:
                # Attempt to process the job data
                Processor(job_entry["data"])
                return True
            except Exception:
                if attempt < max_retries:
                    job_entry["retries"] += 1
                    # Exponential backoff: 2^0=1, 2^1=2, 2^2=4
                    # Delay logic is simulated/tracked as per requirements
                    delay = 2 ** attempt
                    continue
                else:
                    # All retries exhausted
                    return False
        return False

```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue: deque[tuple[str, dict]]] = deque()
        self._results: Dict[str, dict] = {}
        self._queued_ids: set[str] = set()

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Adds a job to the queue. 
        Returns the job_id.
        """
        if job_id in self._results or job_id in self._queued_ids:
            raise ValueError(f"Job ID '{job_id}' is already in use.")
        
        self._queue.append((job_id, data))
        self._queued_ids.add(job_id)
        return job_id

    def process_next_job(self) -> Optional[dict]:
        """
        Processes the next job in FIFO order.
        This method handles the transition from 'queued' to 'completed'.
        """
        if not self._queue:
            return None
            
        job_id, data = self._queue.popleft()
        self._queued_ids.remove(job_id)
        
        # Simulate processing logic
        result = {"status": "completed", "data": data}
        self._results[job_id] = result
        return result

    def get_result(self, job_id: str) -> dict | None:
        """
        Gets the result of a completed job.
        Returns None if the job has not been processed.
        """
        return self._results.get(job_id)

# Example usage:
if __name__ == "__main__":
    queue = JobQueue()
    job_id = queue.add_job("job1", {"task": "process_data"})
    
    # Simulate the processing step
    queue.process_next_job()
    
    result = queue.get_result("job1")
    print(result)
```

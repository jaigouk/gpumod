### 1. `queue/core.py`
This file contains the base data structures. By keeping `Job` here, we avoid circular imports because both `priority` and `retry` will need to reference the `Job` object.

```python
from dataclasses import dataclass, field
from typing import Any, List
from collections import deque

@dataclass(order=False)
class Job:
    """Represents a unit of work in the queue."""
    id: str
    data: Any
    priority: int = 0
    retries: int = 0
    max_retries: int = 3

    def __lt__(self, other: 'Job') -> bool:
        # Necessary for PriorityQueue (heapq) to compare Jobs
        return self.priority < other.priority

class JobQueue:
    """A standard FIFO (First-In-First-Out) queue."""
    def __init__(self):
        self._items: List[Job] = deque()

    def enqueue(self, job: Job) -> None:
        self._items.append(job)

    def dequeue(self) -> Job:
        if not self._items:
            raise IndexError("Dequeue from empty queue")
        return self._items.popleft()

    def __len__(self) -> int:
        return len(self._items)
```

### 2. `queue/priority.py`
This handles the specialized logic for priority-based sorting using Python's `heapq` module.

```python
import heapq
from typing import List
from .core import Job

class PriorityQueue:
    """A priority-based queue where lower numbers = higher priority."""
    def __init__(self):
        self._heap: List[Job] = []

    def enqueue(self, job: Job) -> None:
        heapq.heappush(self._heap, job)

    def dequeue(self) -> Job:
        if not self._heap:
            raise IndexError("Dequeue from empty queue")
        return heapq.heappop(self._heap)

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
This handles the side-effect logic: timing, sleep, and retry loops.

```python
import time
from typing import Callable
from .core import Job

def process_with_retry(job: Job, worker_func: Callable[[Any], bool]) -> bool:
    """
    Executes a worker function with exponential backoff.
    
    Args:
        job: The Job object containing data and retry limits.
        worker_func: A callable that accepts job.data and returns True on success.
    """
    attempts = 0
    
    while attempts <= job.max_retries:
        try:
            if worker_func(job.data):
                return True
        except Exception as e:
            print(f"Attempt {attempts} failed for job {job.id}: {e}")

        attempts += 1
        if attempts <= job.max_retries:
            # Exponential backoff: 2, 4, 8, 16... seconds
            wait_time = 2 ** attempts
            time.sleep(wait_time)
            
    print(f"Job {job.id} failed after {job.max_retries} retries.")
    return False
```

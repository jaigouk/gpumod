```python
from collections import deque
from typing import Dict

class JobQueue:
    def __init__(self):
        # Stores jobs as (job_id, data) in FIFO order
        self._queue: deque[tuple[str, dict]] = deque()
        # Stores completed results keyed by job_id
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.append((job_id, data))
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job, or None if not completed."""
        return self._results.get(job_id)

    def process_next_job(self) -> None:
        """
        Processes the next job in the FIFO queue.
        This fulfills the requirement of processing jobs in order.
        """
        if self._queue:
            job_id, data = self._queue.popleft()
            # Simulate processing logic
            self._results[job_id] = {"status": "completed", "data": data}
```

```python
import heapq

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = 0

    def add_job(self, job_id: str, data: dict, priority: int = 0):
        # We use -priority because heapq is a min-heap (lowest value first).
        # Using a counter ensures that jobs with the same priority are processed 
        # in FIFO order (lowest counter value first).
        heapq.heappush(self._queue, (-priority, self._counter, job_id, data))
        self._counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        if not self._queue:
            return None
        
        # Pop the element with the smallest tuple value
        # (highest priority, lowest counter, job_id, data)
        _, _, job_id, data = heapq.heappop(self._queue)
        return (job_id, data)
```

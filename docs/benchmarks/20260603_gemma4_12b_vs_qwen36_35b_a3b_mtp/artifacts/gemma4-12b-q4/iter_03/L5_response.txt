To refactor the monolithic file, we will distribute the responsibilities according to the requested structure. This ensures that the "Core" definitions remain independent, while specialized logic (Priority and Retry) sits in its own layer.

Here is the refactored package structure:

### 1. `queue/core.py`
Contains the base data structures and the standard FIFO queue.

```python
from dataclasses import dataclass
from typing import Any, List
from collections import deque

@dataclass
class Job:
    """Represents a unit of work in the system."""
    id: str
    payload: Any
    priority: int = 0

class JobQueue:
    """A standard First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self) -> None:
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("Pop from empty queue")
        return self._queue.popleft()

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
 Handles the priority-specific sorting logic.

```python
import heapq
from .core import Job

class PriorityQueue:
    """A priority-based queue where jobs are popped based on priority value."""
    def __init__(self) -> None:
        self._heap: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # We store a tuple so heapq knows to sort by the first element (priority)
        heapq.heappush(self._heap, (job.priority, job))

    def pop(self) -> Job:
        if not self._heap:
            raise IndexError("Pop from empty priority queue")
        # Return the Job object, discarding the priority integer from the tuple
        return heapq.heappop(self._heap)[1]

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
 Contains the orchestration logic for retrying failed jobs.

```python
import time
import random
from .core import Job, JobQueue

def process_with_retry(job: Job, max_retries: int = 3) -> bool:
    """
    Attempts to process a job with exponential backoff.
    Returns True if successful, False otherwise.
    """
    attempts = 0
    while attempts < max_retries:
        try:
            # Simulate work
            print(f"Attempting job {job.id} (Attempt {attempts + 1})")
            
            # Simulate potential failure
            if random.random() < 0.7:
                raise RuntimeError("Simulated Processing Error")
            
            print(f"Successfully processed {job.id}")
            return True
            
        except Exception as e:
            attempts += 1
            if attempts == max_retries:
                print(f"Failed job {job.id} after {max_retries} attempts.")
                break
            
            # Exponential backoff: 2^attempts + jitter
            sleep_time = (2 ** attempts) + random.uniform(0, 1)
            print(f"Error: {e}. Retrying in {sleep_time:.2f}s...")
            time.sleep(sleep_time)
            
    return False
```

### 4. `queue/__init__.py`
 Provides a clean public API, hiding the internal module structure.

```python

Job Queue Package
Provides standard and priority-based queueing with retry logic.
"""

from .core import Job, JobQueue

```

```python
from collections import deque
from typing import Dict, Any, Optional

class JobQueue:
    def __init__(self):
        # Stores jobs in FIFO order: list of tuples (job_id, data)
        self._queue = deque()
        # Stores results of completed jobs: mapping of job_id to result dict
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Add a job to the queue and return the job_id.
        """
        self._queue.append((job_id, data))
        return job_id

    def complete_job(self, job_id: str, result: dict) -> None:
        """
        Helper method to simulate the completion of a job.
         This moves the job from the 'pending' state to the 'completed' state.
        """
        self._results[job_id] = result

    def get_result(self, job_id: str) -> Optional[dict]]:
        """
        Get the result of a completed job. Returns None if not completed.
        """
        return self._results.get(job_id)

# Example Usage:
if __name__ == "__main__":
    queue = JobQueue()
    job_id = queue.add_job("job1", {"task": "process_data"})
    
    # Simulate processing logic...
    # In a real scenario, a worker would fetch from self._queue
    # and then call complete_job.
    queue.complete_job("job1", {"status": "success", "output": [1, 2, 3]})
    
    result = queue.get_result("job1")
    print(result)
```

import heapq

class JobQueue:
    def __init__(self):
        self.heap = []
        self.counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue with a priority level.
        Priority levels: 0 (Normal), 1 (High), 2 (Critical).
        """
        # heapq is a min-heap. To process higher priority values first, 
        # we store the priority as a negative number.
        # The counter ensures FIFO (First-In, First-Out) for same priority levels.
        heapq.heappush(self.heap, (-priority, self.counter, name, data))
        self.counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Returns the highest priority job or None if the queue is empty.
        """
        if not self.heap:
            return None
        
        _, _, name, data = heapq.heappop(self.heap)
        return (name, data)

```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores jobs as a tuple of (job_id, data) to maintain FIFO order
        self._queue: deque[tuple[str, dict]]] = deque()
        # Stores results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.append((job_id, data))
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job by its job_id."""
        return self._results.get(job_id)

    def complete_job(self, job_id: str, result: dict) -> None:
        """
        Helper method to simulate the completion of a job.
         This allows the result to be retrieved via get_result.
        """
        self._results[job_id] = result
        # Remove from the queue as it is now processed
        # Note: In a true FIFO queue, processing usually happens 
        # by popping from the left of the deque.
        if self._queue and self._queue[0][0] == job_id:
            self._queue.popleft()
```

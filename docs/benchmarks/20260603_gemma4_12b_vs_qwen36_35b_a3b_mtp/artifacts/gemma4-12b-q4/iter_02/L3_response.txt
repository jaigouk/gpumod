import heapq

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue.
        Higher priority numbers are processed first.
        Same priorities follow FIFO order.
        """
        # We use -priority because heapq is a min-heap. 
        # By negating, the highest priority value becomes the smallest.
        # self._counter ensures FIFO for identical priorities.
        heapq.heappush(self._queue, (-priority, self._counter, name, data))
        self._counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Returns the highest priority job or None if the queue is empty.
        """
        if not self._queue:
            return None
        
        _, _, name, data = heapq.heappop(self._queue)
        return name, data

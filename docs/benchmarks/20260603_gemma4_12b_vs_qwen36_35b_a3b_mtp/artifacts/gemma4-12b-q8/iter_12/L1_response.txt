```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores pending jobs in FIFO order: (job_id, data)
        self._queue: deque = deque()
        # Stores results of completed jobs: {job_id: result_dict}
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue and returns the job_id."""
        self._queue.append((job_id, data))
        return job_id

    def process_next_job(self) -> None:
        """
        Helper method to process the next job in the FIFO queue.
         This moves the job from the queue to the results dictionary.
        """
        if self._queue:
            job_id, data = self._queue.popleft()
            # Simulate processing logic here
            self._results[job_id] = {"status": "completed", "data": data}

    def get_result(self, job_id: str) -> dict | None:
        """Returns the result of a completed job, or None if not finished."""
        return self._results.get(job_id)
```

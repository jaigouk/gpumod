```python
from typing import Callable, Dict, Any

class JobQueue:
    def __init__(self):
        self.jobs: Dict[str, Dict[str, Any]] = {}

    def add_job(self, job_id: str, data: Dict[str, Any]):
        self.jobs[job_id] = {
            "data": data,
            "retries": 0
        }

    def process_job(self, job_id: str, processor: Callable) -> bool:
        if job_id not in self.jobs:
            return False

        max_retries = 3
        
        for attempt in range(max_retries + 1):
            try:
                # Attempt to process the job
                processor(self.jobs[job_id]["data"]))
                
                # Success: reset retry count and return True
                self.jobs[job_id]["retries"] = 0
                return True
                
            except Exception as e:
                if attempt < max_retries:
                    # Increment retry count
                    self.jobs[job_id]["retries"] += 1
                    
                    # Calculate exponential backoff: 1s, 2s, 4s
                    # 2^(retries-1) => 2^0=1, 2^1=2, 2^2=4
                    backoff = 2 ** (self.jobs[job_id]["retries"] - 1)
                    
                    # Simulate/Track backoff delay
                    print(f"Job {job_id} failed. Retrying in {backoff}s... (Attempt {self.jobs[job_id]['retries']})")
                else:
                    # All retries exhausted
                    return False
```

import heapq

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0):
        # Use negative priority because heapq is a min-heap
        # counter ensures FIFO for jobs with the same priority
        heapq.heappush(self._heap, (-priority, self._counter, name, data))
        self._counter += 1

            return None

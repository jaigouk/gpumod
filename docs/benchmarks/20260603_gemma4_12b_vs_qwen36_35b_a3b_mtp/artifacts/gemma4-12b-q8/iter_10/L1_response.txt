```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores the sequence of job IDs to be processed (FIFO)
        self._queue: deque[str] = deque()
        # Stores job data until it is processed
        self._jobs: Dict[str, dict] = {}
        # Stores results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Adds a job to the queue.
        """
        self._jobs[job_id] = data
        self._queue.append(job_id)
        return job_id

    def process_next_job(self) -> None:
        """
        Processes the next job in the queue (FIFO).
        This is required to fulfill the 'processed' requirement.
        """
        if self._queue:
            job_id = self._queue.popleft()
            data = self._jobs.pop(job_id)
            
            # Simulate processing logic here
            self._results[job_id] = {
                "status": "completed",
                "result": data
            }

    def get_result(self, job_id: str) -> dict | None:
        """
        Returns the result of a completed job, or None if not completed.
        """
        return self._results.get(job_id)
```

### 1. `queue/core.py`
Contains the fundamental data structures and the basic FIFO queue.

```python
from dataclasses import dataclass
from typing import Any
from collections import deque

@dataclass
class Job:
    """Represents a unit of work in the system."""
    id: str
    payload: Any
    priority: int = 0
    retries: int = 0

class JobQueue:
    """A standard First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._queue = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("pop from empty queue")
        return self._queue.popleft()

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
 Contains the specialized priority queue logic. It imports `Job` from `core`.

```python
import heapq
from typing import List
from .core import Job

class PriorityQueue:
    """A priority-based queue where lower numbers indicate higher priority."""
    def __init__(self):
        self._queue: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap; lower priority numbers come out first
        heapq.heappush(self._queue, (job.priority, job))

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("pop from empty queue")
        _, job = heapq.heappop(self._queue)
        return job

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
 Contains the functional logic for retry behaviors. It imports `Job` from `core`.

```python
import time
from typing import Callable
from .core import Job

def process_with_retry(
    job: Job, 
    worker: Callable[[Any], Any], 
    max_retries: int = 3
) -> Any:
    """
    Executes a worker function on a job payload with 
    exponential backoff retry logic.
    """
    attempts = 0
    while True:
        try:
            return worker(job.payload)
        except Exception as e:
            attempts += 1
            if attempts > max_retries:
                raise RuntimeError(f"Job {job.id} failed after {max_retries} retries") from e
            
            # Exponential backoff: 2, 4, 8, 16... seconds
            sleep_time = 2 ** attempts
            print(f"Retrying job {job.id} in {sleep_time}s... (Attempt {attempts})")
            time.sleep(sleep_time)
```

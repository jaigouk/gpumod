### 1. `queue/core.py`
Contains the base data structures and the basic FIFO queue.

```python
from dataclasses import dataclass, field
from collections import deque
from typing import Any, Optional

@dataclass(frozen=True)
class Job:
    """Represents a unit of work in the system."""
    id: str
    name: str
    payload: Any
    priority: int = 0  # Lower numbers = higher priority (standard for heaps)

    def __lt__(self, other: 'Job'):
        # Necessary for PriorityQueue comparison logic
        return self.priority < other.priority

class JobQueue:
    """A standard First-In-First-Out (FIFO) job queue."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        return self._queue.popleft() if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
Handles the specialized priority logic using a heap.

```python
import heapq
from typing import Optional
from .core import Job

class PriorityQueue:
    """A Priority Queue implementation using a min-heap."""
    def __init__(self):
        self._queue: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # Store as tuple (priority, job) so heapq sorts by priority
        heapq.heappush(self._queue, (job.priority, job))

    def pop(self) -> Optional[Job]:
        if not self._queue:
            return None
        return heapq.heappop(self._queue)[1]

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
Handles the execution logic and exponential backoff.

```python
import time
import random
from typing import Callable, Any
from .core import Job

def process_with_retry(
    job: Job, 
    func: Callable[[Job], Any], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with a job, retrying on failure 
    using exponential backoff with jitter.
    """
    attempts = 0
    while attempts <= max_retries:
        try:
            return func(job)
        except Exception as e:
            attempts += 1
            if attempts > max_retries:
                print(f"Job {job.id} failed after {max_retries} retries: {e}")
                raise e
            
            # Exponential backoff: base * 2^attempt + random jitter
            delay = (base_delay * (2 ** (attempts - 1))) + random.uniform(0, 1)
            print(f"Retry {attempts}/{max_retries} for {job.id} in {delay:.2f}s...")
            time.sleep(delay)
```

### 1. `queue/core.py`
*Responsibility: Defines the data structures and basic list-based queue logic.*

```python
from dataclasses import dataclass, field
from typing import Any, Optional
from collections import deque

@dataclass(order=False)
class Job:
    """Represents a unit of work in the system."""
    id: str
    payload: Any
    priority: int = 0
    retries_left: int = 3

    def __lt__(self, other: 'Job') -> bool:
        # Needed for PriorityQueue (heapq) to handle ties in priority
        return self.id < other.id

class JobQueue:
    """A standard First-In-First-Out (FIFO) job queue."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        return self._queue.popleft() if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
*Responsibility: Implements priority-based ordering using a heap.*

```python
import heapq
from typing import Optional
from .core import Job

class PriorityQueue:
    """A priority-based queue where lower priority numbers are processed first."""
    def __init__(self):
        self._queue: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap: lowest priority number comes out first
        heapq.heappush(self._queue, (job.priority, job))

    def pop(self) -> Optional[Job]:
        if not self._queue:
            return None
        _, job = heapq.heappop(self._queue)
        return job

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
*Responsibility: Logic for execution attempts and exponential backoff.*

```python
import time
from typing import Callable
from .core import Job

def process_with_retry(
    job: Job, 
    func: Callable[[Any], Any], 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with a job payload. 
    Implements exponential backoff on failure.
    """
    while job.retries_left > 0:
        try:
            return func(job.payload)
        except Exception as e:
            job.retries_left -= 1
            if job.retries_left == 0:
                raise e
            
            # Exponential backoff: base_delay * 2^(attempts_made)
            delay = base_delay * (2 ** (3 - job.retries_left))
            print(f"Job {job.id} failed. Retrying in {delay}s...")
            time.sleep(delay)
    
    raise RuntimeError(f"Job {job.id} exhausted all retries.")
```

### 4. `queue/__init__.py`
*Responsibility: The public API surface. Users only need to import from here.*

```python

from .priority import PriorityQueue
from .retry import process_with_retry

    "Job",
    "JobQueue",
    "PriorityQueue",
    "process_with_retry",
]
```

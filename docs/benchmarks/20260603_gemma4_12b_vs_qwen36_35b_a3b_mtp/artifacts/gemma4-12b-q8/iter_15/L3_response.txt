import heapq
from typing import Tuple, Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = 0

    def add_job(self, job_id: str, job_data: Dict, priority: int = 0):
        # Use negative priority because heapq is a min-heap (higher priority value = smaller negative value)
        # counter ensures FIFO order for identical priorities
        heapq.heappush(self._queue, (-priority, self._counter, job_id, job_data))
        self._counter += 1

    def get_next_job(self) -> Optional[Tuple[str, Dict]]:
        if not self._queue:
            return None
        
        _, _, job_id, job_data = heapq.heappop(self._queue)
        return (job_id, job_data)

```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores pending jobs in FIFO order
        self._queue: deque = deque()
        # Stores results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.append((job_id, data))
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]:
        """Get the result of a completed job by its ID."""
        return self._results.get(job_id)

    def complete_job(self, job_id: str, result: dict) -> None:
        """
        Helper method to simulate the processing of a job.
         This allows the get_result method to return a value.
        """
        self._results[job_id] = result
```

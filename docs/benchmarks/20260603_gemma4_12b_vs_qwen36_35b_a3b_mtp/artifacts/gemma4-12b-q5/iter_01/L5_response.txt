### 1. `queue/core.py`
**Responsibility:** Defines the base data structures and the standard FIFO queue.

```python
from dataclasses import dataclass, field
from collections import deque
from typing import Any, Optional

@dataclass(order=False)
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    priority: int = 0

class JobQueue:
    """A standard First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        self._queue.append(job)

    def dequeue(self) -> Optional[Job]:
        return self._queue.popleft() if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
**Responsibility:** Implements priority-based ordering using a min-heap.

```python
import heapq
from typing import Optional
from .core import Job

class PriorityQueue:
    """A Priority Queue where lower priority numbers are processed first."""
    def __init__(self):
        self._heap: list[tuple[int, Job]] = []

    def enqueue(self, job: Job) -> None:
        # We store a tuple (priority, job) so heapq sorts by priority
        heapq.heappush(self._heap, (job.priority, job))

    def dequeue(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, job = heapq.heappop(self._heap)
        return job

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
**Responsibility:** Logic for executing functions with exponential backoff.

```python
import time
import random
import logging
from typing import Callable, Any

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def process_with_retry(
    func: Callable[[], Any], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with exponential backoff.
    Formula: delay = base_delay * (2 ** attempt) + jitter
    """
    last_exception = None
    
    for attempt in range(max_retries + 1):
        try:
            return func()
        except Exception as e:
            last_exception = e
            if attempt == max_retries:
                break
            
            # Exponential backoff with a bit of jitter
            delay = (base_delay * (2 ** attempt)) + random.uniform(0, 1)
            logger.warning(f"Attempt {attempt+1} failed. Retrying in {delay:.2f}s...")
            time.sleep(delay)
            
    logger.error(f"All retries failed. Last error: {last_exception}")
    raise last_exception
```

3.  **Readability:** The `__init__.py` file ensures that the package feels like a cohesive library rather than a collection of disjointed files.
4.  **Scalability:** If you need to add a `DistributedQueue` or `PersistentRetry` later, you can add a new file without touching the existing logic.

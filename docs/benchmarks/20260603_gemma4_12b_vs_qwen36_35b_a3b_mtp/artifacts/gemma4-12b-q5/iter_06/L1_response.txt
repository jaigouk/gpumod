from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue: deque[tuple[str, dict]] = deque()
        self._results: Dict[str, dict] = {}
        self._job_ids: set[str] = set()

    def add_job(self, job_id: str, data: dict) -> str:
        if job_id in self._job_ids:
            raise ValueError(f"Job ID '{job_id}' already exists.")
        
        self._queue.append((job_id, data))
        self._job_ids.add(job_id)
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]:
        return self._results.get(job_id)

    def process_jobs(self):
        """
        Processes jobs in FIFO order. 
        This method facilitates the 'FIFO' requirement by pulling from the left of the deque.
        """
        while self._queue:
            job_id, data = self._queue.popleft()
            # Simulate processing logic
            self._results[job_id] = {"status": "completed", "input": data}

# Example usage:
if __name__ == "__main__":
    queue = JobQueue()
    job_id = queue.add_job("job1", {"task": "process_data"})
    
    # Simulate the processing step
    queue.process_jobs()
    
    result = queue.get_result("job1")
    print(result)

```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores jobs in FIFO order: list of dictionaries containing id and data
        self.queue: deque[dict] = deque()
        # Stores the final results keyed by job_id
        self.results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self.queue.append({"id": job_id, "data": data})
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]:
        """Get the result of a completed job."""
        return self.results.get(job_id)

    def process_next(self) -> None:
        """
        Internal helper to process the next job in FIFO order.
        This fulfills the requirement that jobs are processed in FIFO order.
        """
        if self.queue:
            job = self.queue.popleft()
            job_id = job["id"]
            # Simulate processing logic
            self.results[job_id] = {"status": "completed", "result": job["data"]}
```

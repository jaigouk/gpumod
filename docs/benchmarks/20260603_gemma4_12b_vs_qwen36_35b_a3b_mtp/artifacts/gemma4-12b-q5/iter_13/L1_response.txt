```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue: deque[str] = deque()
        self._job_data: Dict[str, dict] = {}
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.append(job_id)
        self._job_data[job_id] = data
        return job_id

    def process_next(self) -> str | None:
        """
        Processes the next job in FIFO order.
        This method is used to transition a job from the queue to completion.
        """
        if not self._queue:
            return None
            
        job_id = self._queue.popleft()
        data = self._job_data.pop(job_id)
        
        # Simulate processing logic
        self._results[job_id] = {
            "status": "completed",
            "data": data
        }
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job."""
        return self._results.get(job_id)
```

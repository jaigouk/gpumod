```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores job_ids in FIFO order
        self._queue: deque[str] = deque()
        # Stores data for jobs currently in the queue
        self._pending_jobs: Dict[str, dict] = {}
        # Stores results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue and return the job_id."""
        self._queue.append(job_id)
        self._pending_jobs[job_id] = data
        return job_id

    def process_next(self) -> dict | None:
        """
        Simulates processing the next job in the queue.
        This method is necessary to move a job from pending to results.
        """
        if not self._queue:
            return None
        
        job_id = self._queue.popleft()
        data = self._pending_jobs.pop(job_id)
        
        # Simulate a basic processing operation
        result = {
            "status": "completed",
            "original_data": data,
            "processed_at": "now"
        }
        
        self._results[job_id] = result
        return result

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job, or None if not completed."""
        return self._results.get(job_id)
```

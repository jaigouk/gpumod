### 1. `queue/core.py`
This file contains the foundational data structures.

```python
from dataclasses import dataclass, field
from typing import Any, List, Optional

@dataclass
class Job:
    id: str
    data: Any
    priority: int = 0

    # Needed for PriorityQueue comparison if priorities are equal
    def __lt__(self, other: 'Job'):
        return self.priority > other.priority

class JobQueue:
    """A basic First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._queue: List[Job] = []

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        return self._queue.pop(0) if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
This handles the specialized priority logic. It imports `Job` from `core`.

```python
import heapq
from typing import Optional
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue using a min-heap."""
    def __init__(self):
        # We use a heap to ensure O(log n) push/pop
        self._heap: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap, so we negate priority 
        # to treat higher numbers as higher priority
        heapq.heappush(self._heap, (-job.priority, job))

    def pop(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, job = heapq.heappop(self._heap)
        return job

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
This contains the functional logic for retries.

```python
import time
import random
from .core import Job

def process_with_retry(job: Job, max_retries: int = 3, base_delay: float = 1.0):
    """
    Processes a job with exponential backoff.
    Formula: delay = base_delay * (2 ^ attempt) + jitter
    """
    for attempt in range(max_retries + 1):
        try:
            print(f"Processing Job {job.id} (Attempt {attempt})")
            # Simulate work
            if random.random() < 0.7:  # 70% success rate
                print(f"Successfully completed {job.id}")
                return True
            raise Exception("Random processing failure")
            
        except Exception as e:
            if attempt == max_retries:
                print(f"Job {job.id} failed after {max_retries} retries.")
                return False
            
            delay = (base_delay * (2 ** attempt)) + random.uniform(0, 1)
            print(f"Error: {e}. Retrying in {delay:.2f}s...")
            time.sleep(delay)
    return False
```

### How to use the new structure:
Because of the `__init__.py` exports, the usage remains clean and flat:

```python
from queue import Job, PriorityQueue, process_with_retry

# Create a priority queue
pq = PriorityQueue()

# Add jobs with different priorities
pq.push(Job(id="low_1", data="low priority", priority=1))
pq.push(Job(id="high_1", data="high priority", priority=10))

# Process highest priority first
while len(pq) > 0:
    current_job = pq.pop()
    process_with_retry(current_job)
```

To refactor this into a proper package, we will separate the concerns: **Data Modeling** (`core.py`), **Specialized Algorithms** (`priority.py`), **Side Effects/Flow Control** (`retry.py`), and **Namespace Management** (`__init__.py`).

Here is the refactored structure:

### 1. `queue/core.py`
This file contains the base data structures and the simplest implementation of a queue.

```python
from dataclasses import dataclass
from typing import Callable, List, Optional, Any

@dataclass
class Job:
    id: str
    task: Callable[[], Any]
    priority: int = 0
    retries: int = 0
    max_retries: int = 3

    def __lt__(self, other: 'Job') -> bool:
        # Needed for PriorityQueue tie-breaking
        return self.priority < other.priority

class JobQueue:
    """A basic First-In-First-Out (FIFO) queue."""
    def __init__(self):
        self._queue: List[Job] = []

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        return self._queue.pop(0) if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
This handles the specific logic for heap-based priority sorting.

```python
import heapq
from typing import List
from .core import Job

class PriorityQueue:
    """A priority-based queue using a min-heap."""
    def __init__(self):
        self._queue: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap; lower priority numbers come out first
        heapq.heappush(self._queue, (job.priority, job))

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("Pop from empty priority queue")
        return heapq.heappop(self._queue)[1]

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
This encapsulates the "strategy" for execution, separating it from the storage logic.

```python
import time
import random
from .core import Job

def process_with_retry(job: Job) -> None:
    """Executes a job with exponential backoff logic."""
    attempts = 0
    while attempts <= job.max_retries:
        try:
            job.task()
            return
        except Exception as e:
            attempts += 1
            if attempts > job.max_retries:
                print(f"Job {job.id} failed after {job.max_retries} retries.")
                raise e
            
            # Exponential backoff: 2^attempts + jitter
            wait_time = (2 ** attempts) + random.uniform(0, 1)
            print(f"Job {job.id} failed (Attempt {attempts}). Retrying in {wait_time:.2f}s...")
            time.sleep(wait_time)
```

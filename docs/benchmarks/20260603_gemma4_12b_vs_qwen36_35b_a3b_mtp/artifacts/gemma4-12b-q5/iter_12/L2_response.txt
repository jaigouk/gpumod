from typing import Callable, Any

class JobQueue:
    def __init__(self):
        self.jobs = {}

    def add_job(self, job_id: str, data: dict):
        self.jobs[job_id] = {"data": data, "retries": 0}

    def process_job(self, job_id: str, processor: Callable) -> bool:
        if job_id not in self.jobs:
            return False
        
        max_retries = 3
        # Loop allows for 1 initial attempt + up to 3 retries
        for attempt in range(max_retries + 1):
            try:
                processor(self.jobs[job_id]["data"])
                return True
            except Exception:
                if attempt < max_retries:
                    self.jobs[job_id]["retries"] += 1
                    # Exponential backoff: 1s (2^0), 2s (2^1), 4s (2^2)
                    delay = 2 ** (self.jobs[job_id]["retries"] - 1)
                    # Delay is simulated/tracked as per requirements
                    pass
                else:
                    return False
        return False

### 1. `queue/core.py`
This file contains the foundational data structures and the basic FIFO queue logic.

```python
from dataclasses import dataclass, field
from typing import Callable, Any, List
import uuid

@dataclass(order=True)
class Job:
    """
    Represents a unit of work. 
    The 'order=True' allows PriorityQueue to sort by priority automatically.
    """
    priority: int
    id: str = field(default_factory=lambda: str(uuid.uuid4()), compare=False)
    task: Callable = field(default=None, compare=False)
    args: tuple = field(default_factory=tuple, compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    """A standard FIFO (First-In-First-Out ) queue."""
    def __init__(self):
        self._queue: List[Job] = []

    def push(self, job: Job):
        self._queue.append(job)

    def pop(self) -> Job | None:
        return self._queue.pop(0) if self._queue else None

    def __len__(self):
        return len(self._queue)
```

### 2. `queue/priority.py`
This file handles specialized sorting logic using a min-heap.

```python
import heapq
from typing import Optional
from .core import Job

class PriorityQueue:
    """A priority-based queue where lower priority numbers are processed first."""
    def __init__(self):
        self._queue: list[Job] = []

    def push(self, job: Job):
        # heapq implements a min-heap. 
        # Since Job has order=True, it compares by priority first.
        heapq.heappush(self._queue, job)

    def pop(self) -> Optional[Job]:
        if not self._queue:
            return None
        return heapq.heappop(self._queue)

    def __len__(self):
        return len(self._queue)
```

### 3. `queue/retry.py`
This module handles the execution logic and the exponential backoff strategy.

```python
import time
import random
from .core import Job

def process_with_retry(job: Job, max_retries: int = 3, base_delay: float = 1.0):
    """
    Executes a job with exponential backoff.
    Formula: delay = base_delay * (2 ^ retry_count) + jitter
    """
    while job.retries <= max_retries:
        try:
            print(f"Executing Job {job.id} (Attempt {job.retries})...")
            job.task(*job.args)
            print(f"Job {job.id} completed successfully.")
            return
        except Exception as e:
            job.retries += 1
            if job.retries > max_retries:
                print(f"Job {job.id} failed after {max_retries} retries. Error: {e}")
                break
            
            # Exponential backoff calculation
            delay = (base_delay * (2 ** (job.retries - 1))) + random.uniform(0, 1)
            print(f"Job {job.id} failed. Retrying in {delay:.2f}s...")
            time.sleep(delay)
```

### Example Usage
You can now use the package cleanly like this:

```python
from queue import Job, PriorityQueue, process_with_retry

# 1. Define a task
def sample_task(name):
    print(f"Hello, {name}!")
    # Simulate a random failure
    import random
    if random.random() < 0.7:
        raise ValueError("Random Failure!")

# 2. Create jobs with different priorities
jobs = [
    Job(priority=2, task=sample_task, args=("Low Priority",)),
    Job(priority=1, task=sample_task, args=("High Priority",)),
    Job(priority=3, task=sample_task, args=("Very Low Priority",)),
]

# 3. Use the PriorityQueue
pq = PriorityQueue()
for j in jobs:
    pq.push(j)

# 4. Process until empty
while len(pq) > 0:
    current_job = pq.pop()
    if current_job:
        process_with_retry(current_job)
```

```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Stores job_ids in FIFO order
        self._queue: deque[str] = deque()
        # Stores job data until processed
        self._jobs: Dict[str, dict] = {}
        # Stores results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        if job_id in self._jobs:
            raise ValueError(f"Job ID {job_id} already exists.")
        
        self._jobs[job_id] = data
        self._queue.append(job_id)
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]:
        return self._results.get(job_id)

    def complete_job(self, job_id: str, result: dict):
        """
        Helper method to transition a job from the queue to completed status.
         """
        self._results[job_id] = result
        if job_id in self._jobs:
            del self._jobs[job_id]
            # If this was the next job in FIFO queue, remove it
            if self._queue and self._queue[0] == job_id:
                self._queue.popleft()
```

### 1. `queue/core.py`
Contains the base data structures and the standard FIFO queue.

```python
from dataclasses import dataclass, field
from typing import Callable, Any, List, Optional
from collections import deque

@dataclass(order=False)
class Job:
    """Represents a unit of work to be processed."""
    id: str
    task: Callable
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    retries: int = 0
    priority: int = 0  # Lower numbers = higher priority

    def execute(self) -> Any:
        return self.task(*self.args, **self.kwargs)

class JobQueue:
    """A standard First-In-First-Out (FIFO) job queue."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Optional[Job]:
        try:
            return self._queue.popleft()
        except IndexError:
            return None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/retry.py`
Handles the retry logic logic independently of the queue structures.

```python
import time
import logging
from functools import wraps
from typing import Callable, TypeVar

T = TypeVar("T")
logger = logging.getLogger(__name__)

def retry_with_backoff(max_retries: int = 3, initial_delay: float = 1.0, backoff_factor: float = 2.0):
    """
    Decorator that retries a function with exponential backoff.
    """
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> T:
            delay = initial_delay
            last_exception = None
            
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt == max_retries:
                        break
                    
                    logger.warning(f"Attempt {attempt+1} failed for {func.__name__}: {e}. Retrying in {delay}s...")
                    time.sleep(delay)
                    delay *= backoff_factor
            
            logger.error(f"All {max_retries} retries failed for {func.__name__}.")
            raise last_exception
        return wrapper
    return decorator
```

### 3. `queue/priority.py`

```python
import heapq
from typing import Optional, List
from .core import Job

class PriorityQueue:
    """A Priority Queue implementation using a min-heap."""
    def __init__(self):
        self._heap: List[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # We use a tuple (priority, job) so heapq sorts by priority
        # We also include a counter or unique ID if jobs have same priority
        # to avoid comparing Job objects directly if they aren't comparable.
        heapq.heappush(self._heap, (job.priority, job))

    def pop(self) -> Optional[Job]:
        try:
            priority, job = heapq.heappop(self._heap)
            return job
        except IndexError:
            return None

    def __len__(self) -> int:
        return len(self._heap)
```

import heapq
from typing import Tuple, Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = 0

    def add_job(self, job_name: str, job_data: dict, priority: int = 0):
        # heapq is a min-heap. To simulate a max-priority queue (where higher 
        # numbers are higher priority), we store the priority as a negative value.
        # The counter ensures that jobs with the same priority are processed 
        # in FIFO order (since the counter always increases).
        heapq.heappush(self._queue, (-priority, self._counter, job_name, job_data))
        self._counter += 1

    def get_next_job(self) -> Optional[Tuple[str, Dict]]:
        if not self._queue:
            return None
        
        _, _, job_name, job_data = heapq.heappop(self._queue)
        return (job_name, job_data)

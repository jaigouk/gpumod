### 1. `queue/core.py`
*Responsibility: Data definitions and basic FIFO queue logic.*

```python
from dataclasses import dataclass
from typing import Any
from collections import deque

@dataclass
class Job:
    id: str
    payload: Any
    priority: int = 0
    retries: int = 0
    max_retries: int = 3

class JobQueue:
    """A basic First-In-First-Out queue for Jobs."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("Pop from empty queue")
        return self._queue.popleft()

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
*Responsibility: Priority-based ordering logic.*

```python
import heapq
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue using a min-heap."""
    def __init__(self):
        # We use a list for the heap
        self._queue: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # Use negative priority because heapq is a min-heap, 
        # but usually higher numbers mean higher priority.
        heapq.heappush(self._queue, (-job.priority, job))

    def pop(self) -> Job:
        if not self._queue:
            raise IndexError("Pop from empty queue")
        _, job = heapq.heappop(self._queue)
        return job

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
*Responsibility: Execution logic and error handling.*

```python
import time
import random
from typing import Callable
from .core import Job

def process_with_retry(job: Job, action: Callable[[Any], Any]) -> Any:
    """
    Executes a function with the job payload using exponential backoff.
    """
    for attempt in range(job.max_retries + 1):
        try:
            return action(job.payload)
        except Exception as e:
            if attempt == job.max_retries:
                print(f"Job {job.id} failed after {attempt} retries.")
                raise e
            
            # Exponential backoff: 2^attempt + jitter
            wait_time = (2 ** attempt) + random.random()
            print(f"Retrying job {job.id} in {wait_time:.2f}s...")
            time.sleep(wait_time)
```


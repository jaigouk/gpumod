### 1. `queue/core.py`
**Responsibility:** Defines the base data structures and the standard FIFO queue.

```python
from dataclasses import dataclass, field
from typing import Any
from collections import deque

@dataclass(order=False)
class Job:
    """Represents a unit of work in the system."""
    id: str
    data: Any
    priority: int = 0
    retries: int = 0

class JobQueue:
    """A basic First-In-First-Out (FIFO) queue for Jobs."""
    def __init__(self):
        self._queue: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job | None:
        return self._queue.popleft() if self._queue else None

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
**Responsibility:** Implementation of priority-based logic. It imports `Job` and `JobQueue` from core.

```python
import heapq
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A Priority Queue where lower numbers indicate higher priority."""
    def __init__(self):
        # We use a list for the heap
        self._heap: list[tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # We store priority first to satisfy heapq ordering
        heapq.heappush(self._heap, (job.priority, job))

    def pop(self) -> Job | None:
        if not self._heap:
            return None
        _, job = heapq.heappop(self._heap)
        return job

    def __len__(self) -> int:
        return len(self._heap)
```

### 3. `queue/retry.py`
** Responsibility:** Pure functional logic for handling retries and backoff.

```python
import time
import random
from .core import Job

def process_with_retry(
    job: Job, 
    work_func, 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> bool:
    """
    Executes a function with a job, applying exponential backoff 
    on failure.
    """
    for attempt in range(max_retries + 1):
        try:
            work_func(job)
            return True
        except Exception as e:
            if attempt == max_retries:
                print(f"Job {job.id} failed after {max_retries} retries: {e}")
                return False
            
            job.retries += 1
            # Exponential backoff: base * 2^attempt + jitter
            delay = (base_delay * (2 ** attempt)) + random.uniform(0, 1)
            print(f"Retry {job.retries} for job {job.id} in {delay:.2f}s...")
            time.sleep(delay)
    return False
```

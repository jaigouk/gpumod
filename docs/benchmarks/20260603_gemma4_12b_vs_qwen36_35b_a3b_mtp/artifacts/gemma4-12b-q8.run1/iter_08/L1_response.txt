```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # queue stores the job data in FIFO order
        self.queue = deque()
        # results stores the results of completed jobs
        self.results = {}
        # track unique IDs to ensure uniqueness
        self.job_ids = set()

    def add_job(self, job_id: str, data: dict) -> str:
        if job_id in self.job_ids:
            raise ValueError(f"Job ID '{job_id}' already exists.")
        
        self.queue.append({"id": job_id, "data": data})
        self.job_ids.add(job_id)
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        return self.results.get(job_id)

    def complete_job(self, job_id: str, result: dict):
        """
        Helper method to mark a job as processed and store its result.
         This is required for the example usage to function.
        """
        self.results[job_id] = result
```

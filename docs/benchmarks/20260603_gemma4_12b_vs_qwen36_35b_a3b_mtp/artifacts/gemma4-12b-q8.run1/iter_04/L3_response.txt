import heapq

        self._counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0):

        # self._counter ensures FIFO order for identical priorities.
        heapq.heappush(self._heap, (-priority, self._counter, name, data))
        self._counter += 1

        if not self._heap:

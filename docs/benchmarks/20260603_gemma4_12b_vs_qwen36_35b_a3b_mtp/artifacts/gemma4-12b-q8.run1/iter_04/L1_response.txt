```python
from collections import deque
from typing import Dict, Optional

class JobQueue:
    def __init__(self):
        # Internal storage for jobs in FIFO order
        self._queue: deque = deque()
        # Storage for results of completed jobs
        self._results: Dict[str, dict] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """
        Adds a job to the queue and returns the unique job_id.
        """
        self._queue.append({"id": job_id, "data": data})
        return job_id

    def complete_job(self, job_id: str, result: dict) -> None:
        """
        Marks a job as processed, stores the result, and removes it from the queue.
        """
        self._results[job_id] = result
        
        # Maintain FIFO order by removing the job from the queue
        # Since jobs are processed in FIFO order, we look for the job_id
        for _ in range(len(self._queue)):
            job = self._queue.popleft()
            if job["id"] == job_id:
                break
            else:
                self._queue.append(job)

    def get_result(self, job_id: str) -> dict | None:
        """
        Returns the result of a completed job or None if not yet completed.
        """
        return self._results.get(job_id)
```

### 1. `queue/core.py`
This module contains the foundational data structures. Notice that `Job` uses `order=True` to allow the `PriorityQueue` to compare jobs automatically based on their fields.

```python
from dataclasses import dataclass, field
from typing import Any, deque

@dataclass(order=True)
class Job:
    """Represents a unit of work."""
    # priority is first so it's used for comparison in PriorityQueue
    priority: int 
    id: str = field(compare=False)
    payload: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._items: deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._items.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        return self._items.popleft()

    def is_empty(self) -> bool:
        return len(self._items) == 0

    def __len__(self) -> int:
        return len(self._items)
```

### 2. `queue/priority.py`
This module implements the specialized queue logic. It imports `Job` to ensure type safety.

```python
import heapq
from typing import List
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue implementation using a heap."""
    def __init__(self):
        # We initialize an empty list for the heap
        self._items: List[Job] = []

    def push(self, job: Job) -> None:
        heapq.heappush(self._items, job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty priority queue")
        return heapq.heappop(self._items)

    def is_empty(self) -> bool:
        return len(self._items) == 0

    def __len__(self) -> int:
        return len(self._items)
```

### 3. `queue/retry.py`
This module contains the execution logic, separated from the storage logic.

```python
import time
from typing import Callable, Any
from .core import Job

def process_with_retry(
    job: Job, 
    func: Callable[[Any], Any], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with the job payload, implementing 
    exponential backoff on failure.
    """
    last_exception = None

    for attempt in range(max_retries + 1):
        try:
            return func(job.payload)
        except Exception as e:
            last_exception = e
            job.retries += 1
            
            if attempt < max_retries:
                # Exponential backoff: 1s, 2s, 4s...
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)
            else:
                break

    raise RuntimeError(f"Job {job.id} failed after {max_retries} retries.") from last_exception
```

### How to use the new structure

Now, instead of one giant file, you can use the package cleanly. Note that even though `PriorityQueue` and `process_with_retry` are not in `__init__.py`, they are still easily accessible via submodules.

```python

from queue.priority import PriorityQueue  # via submodule
from queue.retry import process_with_retry # via submodule

# 1. Setup
pq = PriorityQueue()
pq.push(Job(priority=2, id="low-prio", payload="low priority task"))
pq.push(Job(priority=1, id="high-prio", payload="high priority task"))

# 2. Process
def my_task(data):
    print(f"Processing: {data}")
    return True

current_job = pq.pop()  # This will be the high-priority task
process_with_retry(current_job, my_task)
```

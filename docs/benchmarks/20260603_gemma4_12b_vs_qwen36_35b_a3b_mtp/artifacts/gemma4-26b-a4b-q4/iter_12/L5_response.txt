### 1. `queue/core.py`
**Responsibility:** Defines the data models and the base queue behavior.

```python
from dataclasses import dataclass, field
from typing import Any, Deque
from collections import deque

@dataclass
class Job:
    name: str
    payload: Any
    priority: int = 0
    retries: int = 0

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._queue: Deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        return self._queue.popleft()

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
**Responsibility:** Implements the specialized priority-based sorting logic.

```python
import heapq
from typing import List, Tuple
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue where higher priority numbers are processed first."""
    def __init__(self):
        # We use a list to support heapq operations
        self._queue: List[Tuple[int, Job]] = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap. To make it a max-priority queue, 
        # we negate the priority value.
        heapq.heappush(self._queue, (-job.priority, job))

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        # Return the job from the tuple (-priority, job)
        return heapq.heappop(self._queue)[1]

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
**Responsibility:** Contains the execution logic and error handling.

```python
import time
import logging
from .core import Job

logger = logging.getLogger(__name__)

def process_with_retry(job: Job, worker_func, max_retries: int = 3):
    """
    Executes a worker function with exponential backoff.
    """
    for attempt in range(max_retries + 1):
        try:
            return worker_func(job)
        except Exception as e:
            job.retries += 1
            if attempt == max_retries:
                logger.error(f"Job {job.name} failed after {max_retries} retries.")
                raise e
            
            # Exponential backoff: 1s, 2s, 4s...
            wait_time = 2 ** attempt
            logger.warning(f"Job {job.name} failed (attempt {attempt + 1}). Retrying in {wait_time}s...")
            time.sleep(wait_time)
```

### How to use the new structure

Because of the clean `__init__.py`, the end-user experience remains simple and doesn't require knowing about the internal files:

```python
from queue import Job, PriorityQueue, process_with_retry

# 1. Setup
pq = PriorityQueue()
pq.push(Job(name="low_priority", payload="data", priority=1))
pq.push(Job(name="high_priority", payload="data", priority=10))

# 2. Define a worker
def my_worker(job):
    print(f"Processing {job.name}")
    if job.name == "high_priority":
        raise ValueError("Simulated failure")
    return "Success"

# 3. Execute
while not pq.is_empty():
    current_job = pq.pop()
    try:
        result = process_with_retry(current_job, my_worker)
        print(f"Result: {result}")
    except Exception:
        print(f"Job {current_job.name} ultimately failed.")
```

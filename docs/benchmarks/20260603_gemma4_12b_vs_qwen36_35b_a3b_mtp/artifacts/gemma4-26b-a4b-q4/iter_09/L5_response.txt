### 1. `queue/core.py`
This module contains the data structures and the base queue logic. Note that `Job` uses `order=True` to allow for priority comparisons.

```python
from dataclasses import dataclass, field
from typing import Any, List

@dataclass(order=True)
class Job:
    """Represents a unit of work."""
    # priority must be the first field for the dataclass 'order=True' 
    # to use it as the primary sorting key in PriorityQueue.
    priority: int
    id: str = field(compare=False)
    data: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._jobs: List[Job] = []

    def push(self, job: Job) -> None:
        self._jobs.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("Pop from empty queue")
        return self._jobs.pop(0)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 2. `queue/priority.py`
This module handles specialized ordering logic by inheriting from the core components and using `heapq`.

```python
import heapq
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue using a min-heap."""
    def __init__(self):
        self._jobs = []

    def push(self, job: Job) -> None:
        # heapq maintains the heap property automatically
        heapq.heappush(self._jobs, job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("Pop from empty queue")
        return heapq.heappop(self._jobs)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0
```

### 3. `queue/retry.py`
This module contains the execution logic. It is separated from the data structures to satisfy the Single Responsibility Principle.

```python
import time
from typing import Callable
from .core import Job, JobQueue

def process_with_retry(
    queue: JobQueue, 
    worker_func: Callable[[Job], None], 
    max_retries: int = 3
) -> None:
    """
    Processes jobs from a queue, applying exponential backoff 
    on failure.
    """
    while not queue.is_empty():
        job = queue.pop()
        try:
            worker_func(job)
        except Exception as e:
            if job.retries < max_retries:
                job.retries += 1
                wait_time = 2 ** job.retries  # Exponential backoff
                print(f"Job {job.id} failed: {e}. Retrying in {wait_time}s...")
                time.sleep(wait_time)
                queue.push(job)
            else:
                print(f"Job {job.id} failed permanently after {max_retries} retries.")
```

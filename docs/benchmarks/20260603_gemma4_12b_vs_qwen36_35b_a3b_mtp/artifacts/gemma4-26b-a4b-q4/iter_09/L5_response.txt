Here is the refactored package structure.

### 1. `queue/core.py`
This module handles the data models and the base queue logic. It has no dependencies on other files in the package.

```python
from dataclasses import dataclass, field
from typing import Any, List

@dataclass
class Job:
    """Represents a single unit of work."""
    id: str
    payload: Any
    priority: int = 0
    attempts: int = 0

class JobQueue:
    """A simple FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._jobs: List[Job] = []

    def push(self, job: Job) -> None:
        self._jobs.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        return self._jobs.pop(0)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 2. `queue/priority.py`
This module implements the specialized priority logic. It imports from `core.py`.

```python
import heapq
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A queue that processes jobs based on priority (higher number = higher priority)."""
    def __init__(self):
        # We don't use self._jobs = [] from parent to ensure we use a heap structure
        self._jobs = []

    def push(self, job: Job) -> None:
        # heapq is a min-heap. To make higher priority numbers come out first,
        # we store the priority as a negative number.
        # We include job.id to prevent comparison errors if priorities are equal.
        heapq.heappush(self._jobs, (-job.priority, job.id, job))

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        # Return the third element of the tuple: the actual Job object
        return heapq.heappop(self._jobs)[2]

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 3. `queue/retry.py`
This module contains the execution logic. It is decoupled from the queue storage mechanism.

```python
import time
from typing import Callable
from .core import Job

def process_with_retry(
    job: Job, 
    func: Callable[[Job], Any], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with a job, applying exponential backoff on failure.
    """
    for attempt in range(max_retries):
        try:
            return func(job)
        except Exception as e:
            job.attempts += 1
            if attempt == max_retries - 1:
                raise e
            
            # Exponential backoff: 1s, 2s, 4s...
            delay = base_delay * (2 ** attempt)
            time.sleep(delay)
```

    current_job = queue.pop()

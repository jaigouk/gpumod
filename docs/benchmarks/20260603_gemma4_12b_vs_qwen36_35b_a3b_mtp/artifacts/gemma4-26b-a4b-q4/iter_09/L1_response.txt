from collections import deque
from typing import Any, Dict, Optional

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._results: Dict[str, Dict[str, Any]] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue."""
        self._queue.append((job_id, data))
        return job_id

    def process_next(self, worker_func) -> None:
        """
        Processes the next job in the queue using the provided worker function.
        The worker function should take the job data as input and return a dictionary.
        """
        if self._queue:
            job_id, data = self._queue.popleft()
            result = worker_func(data)
            self._results[job_id] = result

    def get_result(self, job_id: str) -> Optional[Dict[str, Any]]:
        """Returns the result of a completed job, or None if not found/not processed."""
        return self._results.get(job_id)

# Example Usage:
if __name__ == "__main__":
    def sample_worker(data: dict) -> dict:
        return {"status": "success", "processed_task": data.get("task")}

    queue = JobQueue()

    job_id = queue.add_job("job1", {"task": "process_data"})

    queue.process_next(sample_worker)

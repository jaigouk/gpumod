```python
import heapq
import itertools

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = itertools.count()

    def add_job(self, name: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue.
        Uses negative priority to simulate a max-heap using heapq (which is a min-heap).
        Uses a counter to maintain FIFO order for jobs with the same priority.
        """
        count = next(self._counter)
        heapq.heappush(self._heap, (-priority, count, name, data))

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Retrieves and removes the highest priority job from the queue.
        Returns None if the queue is empty.
        """
        if not self._heap:
            return None
        
        _, _, name, data = heapq.heappop(self._heap)
        return (name, data)
```

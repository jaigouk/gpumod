```python
import heapq
from itertools import count

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = count()

    def add_job(self, name: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue.
        Higher priority values are processed first. 
        Same priority jobs follow FIFO.
        """
        # heapq is a min-heap. To treat higher numbers as higher priority,
        # we negate the priority. The counter handles FIFO for identical priorities.
        heapq.heappush(self._queue, (-priority, next(self._counter), name, data))

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Retrieves and removes the highest priority job.
        Returns None if the queue is empty.
        """
        if not self._queue:
            return None
        
        _, _, name, data = heapq.heappop(self._queue)
        return (name, data)
```

```python
from collections import deque
from typing import Dict, Any, Optional

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._jobs = {}
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue, return job_id."""
        if job_id in self._jobs or job_id in self._results:
            raise ValueError(f"Job ID '{job_id}' already exists.")
            
        self._jobs[job_id] = data
        self._queue.append(job_id)
        return job_id

    def process_next_job(self, worker_fn) -> Any:
        """
        Processes the next job in FIFO order using the provided worker function.
        The worker_fn should accept the job data and return a dictionary.
        """
        if not self._queue:
            return None

        job_id = self._queue.popleft()
        job_data = self._jobs.pop(job_id)
        
        try:
            result = worker_fn(job_data)
            self._results[job_id] = result
            return result
        except Exception as e:
            self._results[job_id] = {"error": str(e)}
            return self._results[job_id]

    def get_result(self, job_id: str) -> Optional[Dict[str, Any]]:
        """Get the result of a completed job."""
        return self._results.get(job_id)
```

from collections import deque
from typing import Dict, Any, Optional

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._jobs = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue and returns the job_id."""
        self._jobs[job_id] = {
            "data": data,
            "result": None,
            "completed": False
        }
        self._queue.append(job_id)
        return job_id

    def process_next(self, worker_func):
        """
        Processes the next job in the queue in FIFO order 
        using the provided worker function.
        """
        if not self._queue:
            return

        job_id = self._queue.popleft()
        job = self._jobs[job_id]
        
        # Execute the worker function with the job data
        result = worker_func(job["data"])
        
        job["result"] = result
        job["completed"] = True

    def get_result(self, job_id: str) -> Optional[dict]:
        """Returns the result of a completed job, or None if not completed or not found."""
        job = self._jobs.get(job_id)
        if job and job["completed"]:
            return job["result"]
        return None

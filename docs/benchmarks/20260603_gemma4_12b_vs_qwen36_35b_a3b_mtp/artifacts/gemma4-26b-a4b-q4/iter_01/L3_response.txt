```python
import heapq
import itertools

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = itertools.count()

    def add_job(self, name: str, data: dict, priority: int = 0):
        """
        Adds a job to the queue.
        Priority: Higher number = Higher priority.
        Tie-breaker: FIFO (First In, First Out).
        """
        count = next(self._counter)
        # Use negative priority because heapq is a min-heap, 
        # but we want higher numbers to have higher priority.
        # 'count' ensures FIFO order for identical priorities.
        heapq.heappush(self._queue, (-priority, count, name, data))

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Retrieves and removes the highest priority job from the queue.
        Returns None if the queue is empty.
        """
        if not self._queue:
            return None
        
        _, _, name, data = heapq.heappop(self._queue)
        return name, data
```

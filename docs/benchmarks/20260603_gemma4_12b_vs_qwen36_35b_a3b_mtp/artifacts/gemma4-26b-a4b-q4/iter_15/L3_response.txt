import heapq

class JobQueue:
    def __init__(self):
        self._jobs = []
        self._counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0):
        # heapq is a min-heap. To make higher priority numbers come out first,
        # we store the priority as a negative value.
        # The counter ensures FIFO order for jobs with the same priority.
        heapq.heappush(self._jobs, (-priority, self._counter, name, data))
        self._counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        if not self._jobs:
            return None
        
        # Pop the element with the smallest value (most negative priority, then lowest counter)
        _, _, name, data = heapq.heappop(self._jobs)
        return name, data

```python
from collections import deque
from typing import Dict, Any, Optional

class JobQueue:
    def __init__(self):
        self._queue: deque = deque()
        self._results: Dict[str, Dict[str, Any]] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue."""
        self._queue.append((job_id, data))
        return job_id

    def process_next(self) -> Optional[str]:
        """
        Processes the next job in the queue using FIFO order.
        Returns the job_id of the processed job, or None if queue is empty.
        """
        if not self._queue:
            return None
        
        job_id, data = self._queue.popleft()
        
        # Simulate processing logic
        result = {
            "status": "completed",
            "input_data": data
        }
        
        self._results[job_id] = result
        return job_id

    def get_result(self, job_id: str) -> Optional[dict]:
        """Gets the result of a completed job."""
        return self._results.get(job_id)

# Example usage as per requirements:
if __name__ == "__main__":
    queue = JobQueue()
    job_id = queue.add_job("job1", {"task": "process_data"})
    
    # Simulate the processing step
    queue.process_next()
    
    # Retrieve the result
    result = queue.get_result("job1")
    print(result)
```

import heapq

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = 0

    def add_job(self, name: str, data: dict, priority: int = 0) -> None:
        # heapq is a min-heap. To prioritize higher numbers, we store negative priority.
        # We include self._counter to ensure FIFO order for jobs with the same priority.
        heapq.heappush(self._queue, (-priority, self._counter, name, data))
        self._counter += 1

        if not self._queue:
            return None

        # (highest priority, then lowest counter/oldest entry)
        _, _, name, data = heapq.heappop(self._queue)
        return name, data

```python
from collections import deque

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Add a job to the queue."""
        self._queue.append((job_id, data))
        return job_id

    def get_result(self, job_id: str) -> dict | None:
        """Get the result of a completed job."""
        return self._results.get(job_id)

    def process_next(self, worker_func):
        """
        Processes the next job in the queue using the provided worker function.
        This method satisfies the FIFO requirement by using popleft().
        """
        if not self._queue:
            return
        
        job_id, data = self._queue.popleft()
        result = worker_func(data)
        self._results[job_id] = result

# Example usage to demonstrate functionality:
if __name__ == "__main__":
    queue = JobQueue()
    
    # Define a simple worker function
    def my_worker(data):
        return {"processed": True, "original_task": data.get("task")}

    # Add jobs
    job_id1 = queue.add_job("job1", {"task": "process_data"})
    job_id2 = queue.add_job("job2", {"task": "send_email"})

    # Process jobs (FIFO)
    queue.process_next(my_worker) # Processes job1
    queue.process_next(my_worker) # Processes job2

    # Get results
    print(queue.get_result("job1"))  # {'processed': True, 'original_task': 'process_data'}
    print(queue.get_result("job2"))  # {'processed': True, 'original_task': 'send_email'}
    print(queue.get_result("job3"))  # None
```

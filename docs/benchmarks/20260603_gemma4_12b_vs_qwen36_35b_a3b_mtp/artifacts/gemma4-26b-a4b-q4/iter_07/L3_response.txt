```python
import heapq
import itertools

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = itertools.count()

    def add_job(self, name: str, data: dict, priority: int = 0) -> None:
        """
        Adds a job to the queue.
        Higher priority values are processed first.
        Same priority jobs follow FIFO.
        """
        count = next(self._counter)
        # heapq is a min-heap. To simulate a max-heap for priority, we negate the priority.
        # The count ensures FIFO order for jobs with the same priority.
        heapq.heappush(self._queue, (-priority, count, name, data))

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Retrieves and removes the highest priority job from the queue.
        Returns None if the queue is empty.
        """
        if not self._queue:
            return None
        
        _, _, name, data = heapq.heappop(self._queue)
        return (name, data)
```

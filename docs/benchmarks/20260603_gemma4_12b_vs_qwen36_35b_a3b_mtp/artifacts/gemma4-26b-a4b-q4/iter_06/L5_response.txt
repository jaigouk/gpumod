### 1. `queue/core.py`
**Responsibility:** Data models and the base queue logic. This is the foundation that other modules depend on.

```python
from dataclasses import dataclass, field
from typing import Any
from collections import deque

@dataclass(order=True)
class Job:
    """Represents a unit of work."""
    # priority is first for comparison in PriorityQueue
    priority: int 
    id: str = field(compare=False)
    payload: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._queue = deque()

    def enqueue(self, job: Job) -> None:
        self._queue.append(job)

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("Dequeue from empty queue")
        return self._queue.popleft()

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
**Responsibility:** Specialized queue implementation using a heap. It inherits from `JobQueue` but overrides the internal storage mechanism.

```python
import heapq
from typing import List
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue using a min-heap."""
    def __init__(self):
        # We don't call super().__init__ because we replace the deque with a list
        self._queue: List[Job] = []

    def enqueue(self, job: Job) -> None:
        heapq.heappush(self._queue, job)

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("Dequeue from empty queue")
        return heapq.heappop(self._queue)

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
**Responsibility:** High-level execution logic. It is decoupled from the queue implementation itself, focusing only on the retry strategy.

```python
import time
import random
from typing import Callable
from .core import Job

def process_with_retry(
    job: Job, 
    task_func: Callable[[Any], None], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> bool:
    """
    Executes a function with exponential backoff.
    Returns True if successful, False if all retries failed.
    """
    while job.retries <= max_retries:
        try:
            task_func(job.payload)
            return True
        except Exception as e:
            job.retries += 1
            if job.retries > max_retries:
                print(f"Job {job.id} failed after {max_retries} retries: {e}")
                return False
            
            # Exponential backoff: base_delay * 2^retries + jitter
            delay = (base_delay * (2 ** (job.retries - 1))) + random.uniform(0, 1)
            print(f"Job {job.id} failed. Retrying in {delay:.2f}s... (Attempt {job.retries})")
            time.sleep(delay)
    
    return False
```

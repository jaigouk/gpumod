### 1. `queue/core.py`
This module contains the fundamental data structures. It has no dependencies on other files in the package.

```python
from dataclasses import dataclass, field
from typing import Any, Deque
from collections import deque

@dataclass(order=True)
class Job:
    """Base Job class. 'order=True' allows for priority comparison."""
    priority: int  # Lower numbers = higher priority
    job_id: str = field(compare=False)
    data: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    """Basic FIFO implementation of a job queue."""
    def __init__(self):
        self._queue: Deque[Job] = deque()

    def push(self, job: Job) -> None:
        self._queue.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("Pop from empty queue")
        return self._queue.popleft()

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
This module specializes the queue logic. It imports from `.core`.

```python
import heapq
from typing import List
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A queue that uses a heap to ensure high-priority jobs are processed first."""
    def __init__(self):
        # We initialize an empty list to act as our heap
        self._queue: List[Job] = []

    def push(self, job: Job) -> None:
        heapq.heappush(self._queue, job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("Pop from empty queue")
        return heapq.heappop(self._queue)

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
This module contains the "engine" logic. It is decoupled from the specific queue implementation by relying on the `JobQueue` interface.

```python
import time
from .core import Job, JobQueue

def process_with_retry(
    queue: JobQueue, 
    worker_func, 
    max_retries: int = 3, 
    backoff_factor: float = 2.0
):
    """
    Processes jobs from a queue. If a worker fails, the job is 
    re-queued with exponential backoff.
    """
    while not queue.is_empty():
        job = queue.pop()
        try:
            worker_func(job)
            print(f"Successfully processed job {job.job_id}")
        except Exception as e:
            if job.retries < max_retries:
                job.retries += 1
                wait_time = backoff_factor ** job.retries
                print(f"Job {job.job_id} failed: {e}. Retrying in {wait_time}s...")
                time.sleep(wait_time)
                queue.push(job)
            else:
                print(f"Job {job.job_id} failed after {max_retries} retries. Dropping.")
```

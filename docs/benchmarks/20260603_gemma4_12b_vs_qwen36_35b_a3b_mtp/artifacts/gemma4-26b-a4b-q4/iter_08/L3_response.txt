```python
import heapq
from itertools import count

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = count()

    def add_job(self, name: str, metadata: dict, priority: int = 0) -> None:
        """
        Adds a job to the queue.
        Higher priority values are processed first.
        """
        # We use -priority because heapq is a min-heap, but we want a max-priority behavior.
        # We use self._counter to ensure FIFO for jobs with the same priority.
        count = next(self._counter)
        heapq.heappush(self._queue, (-priority, count, name, metadata))

    def get_next_job(self) -> tuple[str, dict] | None:
        """
        Returns the highest priority job as (name, metadata), or None if empty.
        """
        if not self._queue:
            return None
        
        _, _, name, metadata = heapq.heappop(self._queue)
        return name, metadata
```

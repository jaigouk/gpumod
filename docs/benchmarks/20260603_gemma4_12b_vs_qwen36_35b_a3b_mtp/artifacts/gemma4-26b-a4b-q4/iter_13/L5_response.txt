### 1. `queue/core.py`
**Responsibility:** Defines the data model and the base interface for a queue.

```python
from dataclasses import dataclass, field
from typing import Any, Deque
from collections import deque

@dataclass(order=False)
class Job:
    """The base data unit for the queue."""
    id: str
    payload: Any

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._queue: Deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        self._queue.append(job)

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("Dequeue from empty queue")
        return self._queue.popleft()

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
**Responsibility:** Implements a specialized queue using a heap. Note how it imports from `.core`.

```python
import heapq
from typing import List, Tuple
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A queue where jobs are processed based on priority (lower number = higher priority)."""
    def __init__(self):
        # We use a list for heapq instead of a deque
        self._queue: List[Tuple[int, int, Job]] = []
        self._counter = 0  # Used to handle tie-breaking for jobs with same priority

    def enqueue(self, job: Job, priority: int) -> None:
        """
        Adds a job with a priority. 
        Uses a counter to ensure stable sorting if priorities are equal.
        """
        heapq.heappush(self._queue, (priority, self._counter, job))
        self._counter += 1

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("Dequeue from empty queue")
        # Return the job object (index 2 in the tuple)
        return heapq.heappop(self._queue)[2]

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def __len__(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`
**Responsibility:** Contains the execution logic and error handling. Note how it depends on the abstractions in `.core`.

```python
import time
import random
from typing import Callable
from .core import Job, JobQueue

def process_with_retry(
    queue: JobQueue, 
    worker: Callable[[Job], None], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> None:
    """
    Processes jobs from the queue. If a worker fails, it retries 
    using exponential backoff.
    """
    while not queue.is_empty():
        job = queue.dequeue()
        retries = 0
        success = False

        while retries <= max_retries and not success:
            try:
                worker(job)
                success = True
            except Exception as e:
                retries += 1
                if retries > max_retries:
                    print(f"Job {job.id} failed after {max_retries} retries. Error: {e}")
                    break
                
                # Exponential backoff: base_delay * 2^retries + jitter
                delay = (base_delay * (2 ** (retries - 1))) + random.uniform(0, 1)
                print(f"Job {job.id} failed. Retrying in {delay:.2f}s... ({retries}/{max_retries})")
                time.sleep(delay)
```

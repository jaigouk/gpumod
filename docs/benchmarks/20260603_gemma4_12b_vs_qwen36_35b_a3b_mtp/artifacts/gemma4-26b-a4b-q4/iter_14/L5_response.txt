### 1. `queue/core.py`
This module handles the data models and the base logic. It has no dependencies on other modules in the package.

```python
from dataclasses import dataclass, field
from typing import Any, Deque
from collections import deque

@dataclass(order=False)
class Job:
    """Represents a unit of work."""
    id: str
    priority: int
    data: Any

    def __lt__(self, other):
        # Required for PriorityQueue tie-breaking if priorities are equal
        if not isinstance(other, Job):
            return NotImplemented
        return self.priority < other.priority

class JobQueue:
    """A basic FIFO (First-In-First-Out) job queue."""
    def __init__(self):
        self._queue: Deque[Job] = deque()

    def enqueue(self, job: Job) -> None:
        self._queue.append(job)

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("dequeue from empty queue")
        return self._queue.popleft()

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def size(self) -> int:
        return len(self._queue)
```

### 2. `queue/priority.py`
This module implements the specialized queue. It imports from `core.py`.

```python
import heapq
from typing import List
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A queue that processes jobs based on priority (lowest number first)."""
    def __init__(self):
        # We initialize an empty list for heapq instead of a deque
        self._queue: List[Job] = []

    def enqueue(self, job: Job) -> None:
        # heapq implements a min-heap
        heapq.heappush(self._queue, job)

    def dequeue(self) -> Job:
        if self.is_empty():
            raise IndexError("dequeue from empty priority queue")
        return heapq.heappop(self._queue)

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def size(self) -> int:
        return len(self._queue)
```

### 3. `queue/retry.py`

```python
import time
import logging
from typing import Callable, Any
from .core import Job

logger = logging.getLogger(__name__)

def process_with_retry(
    job: Job, 
    func: Callable[[Any], Any], 
    max_retries: int = 3, 
    backoff_factor: float = 2.0
) -> Any:
    """
    Executes a function with a job's data, applying exponential backoff on failure.
    """
    retries = 0
    while True:
        try:
            return func(job.data)
        except Exception as e:
            retries += 1
            if retries > max_retries:
                logger.error(f"Job {job.id} failed after {max_retries} retries.")
                raise e
            
            wait_time = backoff_factor ** retries
            logger.warning(
                f"Job {job.id} failed: {e}. Retrying in {wait_time}s..."
            )
            time.sleep(wait_time)
```

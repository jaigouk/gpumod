```python
from collections import deque
from typing import Any, Optional

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue in FIFO order."""
        self._queue.append((job_id, data))
        return job_id

    def process_next_job(self, worker_func):
        """
        Processes the next job in the queue using the provided worker function.
        This is necessary to fulfill the FIFO processing requirement.
        """
        if not self._queue:
            return None
        
        job_id, data = self._queue.popleft()
        try:
            result = worker_func(data)
            self._results[job_id] = result
        except Exception as e:
            self._results[job_id] = {"error": str(e)}
        return self._results[job_id]

    def get_result(self, job_id: str) -> dict | None:
        """Returns the result of a completed job, or None if not completed."""
        return self._results.get(job_id)

# Example usage:
if __name__ == "__main__":
    queue = JobQueue()
    
    # Define a simple worker function
    def process_task(data: dict) -> dict:
        return {"status": "success", "task": data.get("task")}

    # Add job
    job_id = queue.add_job("job1", {"task": "process_data"})
    
    # Process job
    queue.process_next_job(process_task)
    
    # Get result
    result = queue.get_result("job1")
    print(result)  # Output: {'status': 'success', 'task': 'process_data'}
```

from collections import deque

class JobQueue:
    def __init__(self):
        self._queue = deque()
        self._results = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue and returns the job_id."""
        self._queue.append((job_id, data))
        return job_id

    def process_next_job(self):
        """
        Processes the next job in the queue using FIFO order.
        In a real implementation, this would execute the task.
        """
        if self._queue:
            job_id, data = self._queue.popleft()
            # Simulating processing by returning the input data as the result
            self._results[job_id] = {"status": "completed", "output": data}

    def get_result(self, job_id: str) -> dict | None:
        """Returns the result of a completed job or None if not found/not processed."""
        return self._results.get(job_id)

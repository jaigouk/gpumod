### 1. `queue/core.py`
This module handles the data models and the base queue logic. Note that `Job` uses `order=True` to allow the `PriorityQueue` to compare jobs automatically.

```python
from dataclasses import dataclass, field
from typing import Any, List

@dataclass(order=True)
class Job:
    # priority must come first for the PriorityQueue comparison logic
    priority: int 
    id: str = field(compare=False)
    payload: Any = field(compare=False)
    retries: int = field(default=0, compare=False)

class JobQueue:
    def __init__(self):
        self._jobs: List[Job] = []

    def push(self, job: Job) -> None:
        self._jobs.append(job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty queue")
        return self._jobs.pop(0)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 2. `queue/priority.py`
This module implements the specialized priority logic by inheriting from the core logic and using `heapq`.

```python
import heapq
from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    def __init__(self):
        # We override the internal storage to use a heap
        self._jobs = []

    def push(self, job: Job) -> None:
        # heapq implements a min-heap. 
        # If you want high numbers to be higher priority, 
        # you would push ( -job.priority, job )
        heapq.heappush(self._jobs, job)

    def pop(self) -> Job:
        if self.is_empty():
            raise IndexError("pop from empty priority queue")
        return heapq.heappop(self._jobs)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def __len__(self) -> int:
        return len(self._jobs)
```

### 3. `queue/retry.py`
This module is isolated to handle the execution logic and timing, depending only on the `Job` definition.

```python
import time
from typing import Callable, Any
from .core import Job

def process_with_retry(
    job: Job, 
    func: Callable[[Job], Any], 
    max_retries: int = 3, 
    base_delay: float = 1.0
) -> Any:
    """
    Executes a function with a job, implementing exponential backoff.
    """
    last_exception = None
    
    for attempt in range(max_retries + 1):
        try:
            return func(job)
        except Exception as e:
            last_exception = e
            job.retries += 1
            
            if attempt < max_retries:
                # Exponential backoff: 1s, 2s, 4s...
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)
            else:
                break
                
    raise last_exception
```

### How to use the refactored package

Now, instead of one giant file, your main application code stays very clean:

```python

pq.push(Job(priority=2, id="1", payload="Low priority task"))
pq.push(Job(priority=1, id="2", payload="High priority task"))

# 2. Define a worker function
def my_worker(job):
    print(f"Processing {job.id}: {job.payload}")
    if job.id == "2":
        raise ValueError("Simulated failure!")
    return "Success"

# 3. Process
while not pq.is_empty():
    current_job = pq.pop()
    try:
        result = process_with_retry(current_job, my_worker, max_retries=2)
        print(f"Result: {result}")
    except Exception as e:
        print(f"Job {current_job.id} failed after retries: {e}")
```

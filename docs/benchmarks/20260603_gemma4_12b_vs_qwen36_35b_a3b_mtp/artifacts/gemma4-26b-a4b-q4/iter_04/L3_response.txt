import heapq
from itertools import count

class JobQueue:
    def __init__(self):
        self._queue = []
        self._counter = count()

    def add_job(self, name: str, data: dict, priority: int = 0) -> None:
        # We use -priority because heapq is a min-heap, but we want a max-priority queue.
        # We use the counter as a tie-breaker to ensure FIFO order for equal priorities.
        heapq.heappush(self._queue, (-priority, next(self._counter), name, data))

    def get_next_job(self) -> tuple[str, dict] | None:
        if not self._queue:
            return None
        
        # Pop the item with the lowest value (highest priority/earliest sequence)
        _, _, name, data = heapq.heappop(self._queue)
        return name, data

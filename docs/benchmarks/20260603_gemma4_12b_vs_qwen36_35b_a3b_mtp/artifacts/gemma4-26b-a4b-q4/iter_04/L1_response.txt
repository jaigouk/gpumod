```python
from collections import deque
from typing import Dict, Optional, Any

class JobQueue:
    def __init__(self):
        self._queue: deque[str] = deque()
        self._jobs: Dict[str, Dict[str, Any]] = {}

    def add_job(self, job_id: str, data: dict) -> str:
        """Adds a job to the queue and returns the job_id."""
        self._jobs[job_id] = {
            "data": data,
            "result": None,
            "completed": False
        }
        self._queue.append(job_id)
        return job_id

    def process_next(self):
        """
        Processes the next job in the queue in FIFO order.
        In a real-world scenario, this would execute the task logic.
        """
        if not self._queue:
            return

        job_id = self._queue.popleft()
        job = self._jobs[job_id]

        # Simulate processing the job data
        processed_result = {"status": "completed", "original_data": job["data"]}
        
        job["result"] = processed_result
        job["completed"] = True

    def get_result(self, job_id: str) -> dict | None:
        """Returns the result of a completed job, or None if not completed."""
        job = self._jobs.get(job_id)
        if job and job["completed"]:
            return job["result"]
        return None
```

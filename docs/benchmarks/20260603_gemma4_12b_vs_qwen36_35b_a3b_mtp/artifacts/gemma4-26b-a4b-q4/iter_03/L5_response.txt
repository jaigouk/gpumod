from typing import Callable

def process_with_retry(
    queue: JobQueue, 
    worker: Callable[[Job], None], 
    max_retries: int = 3
) -> None:
    """
    Processes all jobs in the queue. If a worker fails, 
    it retries with exponential backoff.
    """
    while not queue.is_empty():
        job = queue.pop()
        attempt = 0
        
        while attempt <= max_retries:
            try:
                worker(job)
                break  # Success!
            except Exception as e:
                attempt += 1
                if attempt > max_retries:
                    print(f"Job {job.id} failed permanently after {max_retries} retries: {e}")
                    # In a real system, you might move this to a Dead Letter Queue
                    break
                
                wait_time = 2 ** attempt  # Exponential backoff
                print(f"Job {job.id} failed. Retrying in {wait_time}s... (Attempt {attempt}/{max_retries})")

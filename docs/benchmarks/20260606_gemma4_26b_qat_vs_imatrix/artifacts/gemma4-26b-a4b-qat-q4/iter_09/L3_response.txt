    ```python
    import heapq

    class JobQueue:
        def __init__(self):
            self.queue = []
            self.counter = 0

        def add_job(self, name: str, data: dict, priority: int = 0):
            # Using negative priority because heapq is a min-heap
            # Using self.counter to maintain FIFO for same priority
            heapq.heappush(self.queue, (-priority, self.counter, name, data))
            self.counter += 1

        def get_next_job(self) -> tuple[str, dict] | None:
            if not self.queue:
                return None
            # Pop the element with the smallest (-priority), then smallest counter
            _, _, name, data = heapq.heappop(self.queue)
            return name, data
    ```

    from dataclasses import dataclass
    from typing import Callable, Any
    import heapq

    @dataclass
    class Job:
        id: str
        data: dict
        priority: int = 0
        retries: int = 0

    class RetryPolicy:
        def __init__(self, max_attempts: int = 4):
            self.max_attempts = max_attempts

        def run(self, fn: Callable, data: Any) -> tuple[bool, int]:
            attempts = 0
            while attempts < self.max_attempts:
                attempts += 1
                try:
                    fn(data)
                    return True, attempts
                except Exception:
                    continue
            return False, attempts

        def __init__(self):
            self._jobs = {}
            self._heap = []
            self._counter = 0
            self._retry_policy = RetryPolicy()

        def add_job(self, job_id: str, data: dict, priority: int = 0) -> None:
            job = Job(id=job_id, data=data, priority=priority)
            self._jobs[job_id] = job
            # Higher priority number means higher priority? 
            # "highest-priority job's (id, data)".
            # Usually, in a max-priority queue, higher number = higher priority.
            # In a min-heap (heapq), we use -priority to get max-priority first.
            heapq.heappush(self._heap, (-priority, self._counter, job_id))
            self._counter += 1

        def get_next_job(self) -> tuple[str, dict] | None:
            while self._heap:
                # Peek

                if job_id in self._jobs:

                else:

            if job_id not in self._jobs:
                return False

            if success:
                del self._jobs[job_id]

    ```python
    class RetryPolicy:
        def __init__(self, max_attempts: int = 4):
            self.max_attempts = max_attempts

        def run(self, fn: Callable, data: Any) -> tuple[bool, int]:
            for attempt in range(1, self.max_attempts + 1):
                try:
                    fn(data)
                    return True, attempt
                except Exception:
                    continue
            return False, self.max_attempts

    class JobQueue:
        def __init__(self):
            self._jobs: dict[str, Job] = {}
            self._queue: list[tuple[int, int, str]] = []
            self._counter = 0
            self._retry_policy = RetryPolicy()

        def add_job(self, job_id: str, data: dict, priority: int = 0) -> None:
            job = Job(id=job_id, data=data, priority=priority)
            self._jobs[job_id] = job
            heapq.heappush(self._queue, (-priority, self._counter, job_id))
            self._counter += 1

        def process_job(self, job_id: str, processor: Callable) -> bool:
            job = self._jobs.get(job_id)
            if not job:
                return False
            success, attempts = self._retry_policy.run(processor, job.data)
            if success:
                del self._jobs[job_id]
            else:
                job.retries += attempts
            return success

        def get_next_job(self) -> tuple[str, dict] | None:
            while self._queue:
                _, _, job_id = heapq.heappop(self._queue)
                if job_id in self._jobs:
                    job = self._jobs[job_id]
                    return job.id, job.data
            return None
    ```

    One edge case: if `add_job` is called with the same `job_id` twice?
    The dictionary will overwrite the old `Job` object.
    But the old `job_id` will still be in the `_queue`.
    When `get_next_job` pops the old `job_id`, `job_id in self._jobs` will still be true because it was overwritten.
    Wait, this is a problem.
    If `add_job("1", data1)` and `add_job("1", data2)`:
    - `_jobs["1"] = Job(id="1", data=data1)`
    - `_queue = [(-0, 0, "1")]`
    - `_jobs["1"] = Job(id="1", data=data2)`
    - `_queue = [(-0, 0, "1"), (-0, 1, "1")]`
    When `get_next_job` pops the first `"1"`, it sees `"1" in _jobs` and returns data2 (the new one).
    Then `process_job("1")` is called. It succeeds, `del _jobs["1"]`.
    When `get_next_job` pops the second `"1"`, `job_id in self._jobs` is now false.
    So it correctly skips the duplicate. This is actually fine.

    Final check on imports: `dataclass` from `dataclasses`, `Callable`, `Any` from `typing`, `heapq` from `heapq`. (Note: `Any` and `Callable` are helpful but `Callable` is needed for the type hint).

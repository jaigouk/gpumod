```python
from dataclasses import dataclass, field
from typing import Optional, Callable, Any
from queue import Queue
import uuid
import time
from datetime import datetime


@dataclass
class Job:
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    task: Callable = None
    args: tuple = field(default_factory=tuple)
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    status: str = "pending"
    created_at: datetime = field(default_factory=datetime.now)
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    retry_count: int = 0
    max_retries: int = 3
    result: Any = None
    error: Optional[Exception] = None

    def __post_init__(self):
        if self.id is None:
            self.id = str(uuid.uuid4())

    def to_dict(self) -> dict:
        return {
            'id': self.id,
            'task': self.task.__name__ if self.task else None,
            'args': self.args,
            'kwargs': self.kwargs,
            'priority': self.priority,
            'status': self.status,
            'created_at': self.created_at.isoformat(),
            'started_at': self.started_at.isoformat() if self.started_at else None,
            'completed_at': self.completed_at.isoformat() if self.completed_at else None,
            'retry_count': self.retry_count,
            'max_retries': self.max_retries,
            'result': str(self.result) if self.result is not None else None,
            'error': str(self.error) if self.error else None
        }


class JobQueue:
    def __init__(self, max_size: int = 1000):
        self._queue = Queue(maxsize=max_size)
        self._jobs: dict[str, Job] = {}
        self._lock = None

    def enqueue(self, job: Job) -> str:
        job_id = job.id
        self._jobs[job_id] = job
        self._queue.put(job)
        return job_id

    def dequeue(self) -> Optional[Job]:
        try:
            job = self._queue.get(timeout=1)
            return job
        except:
            return None

    def get_job(self, job_id: str) -> Optional[Job]:
        return self._jobs.get(job_id)

    def update_job_status(self, job_id: str, status: str):
        if job_id in self._jobs:
            self._jobs[job_id].status = status

    def clear(self):
        self._jobs.clear()
        while not self._queue.empty():
            try:
                self._queue.get_nowait()
            except:
                break

    def size(self) -> int:
        return self._queue.qsize()

    def is_empty(self) -> bool:
        return self._queue.empty()
```

```python
from .core import Job
from typing import List, Optional
import heapq


class PriorityQueue:
    def __init__(self):
        self._heap: List[tuple] = []
        self._counter = 0

    def push(self, job: Job):
        entry = (-job.priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)

    def pop(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        if not self._heap:
            return None
        return self._heap[0][2]

    def size(self) -> int:
        return len(self._heap)

    def is_empty(self) -> bool:
        return len(self._heap) == 0

    def remove(self, job_id: str) -> bool:
        for i, entry in enumerate(self._heap):
            if entry[2].id == job_id:
                self._heap.pop(i)
                heapq.heapify(self._heap)
                return True
        return False

    def clear(self):
        self._heap.clear()
```

### `queue/retry.py`

```python
import time
import random
from .core import Job


def calculate_backoff(retry_count: int, base_delay: float = 1.0, max_delay: float = 60.0) -> float:
    delay = base_delay * (2 ** retry_count)
    delay += random.uniform(0, 1)
    return min(delay, max_delay)


def process_with_retry(job: Job, func: callable, max_retries: int = 3, base_delay: float = 1.0) -> tuple[bool, str]:
    for attempt in range(max_retries + 1):
        try:
            job.started_at = job.started_at or None
            result = func(*job.args, **job.kwargs)
            job.result = result
            job.status = "completed"
            job.completed_at = job.completed_at or None
            return True, "success"
        except Exception as e:
            job.error = e
            job.retry_count = attempt
            if attempt < max_retries:
                delay = calculate_backoff(attempt, base_delay)
                time.sleep(delay)
            else:
                job.status = "failed"
                return False, f"failed after {attempt + 1} attempts"
    return False, "max retries exceeded"
```

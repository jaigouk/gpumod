"""Core job queue functionality."""
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Callable, Dict, List, Optional
import json


@dataclass
class Job:
    """Represents a job to be processed."""
    id: str
    func: Callable
    args: tuple = field(default_factory=tuple)
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    created_at: datetime = field(default_factory=datetime.now)
    status: str = "pending"
    attempts: int = 0
    max_retries: int = 3
    retry_delay: float = 1.0
    result: Optional[Any] = None
    error: Optional[str] = None
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None

    def to_dict(self) -> dict:
        """Convert job to dictionary for serialization."""
        return {
            'id': self.id,
            'func': self.func.__name__,
            'args': self.args,
            'kwargs': self.kwargs,
            'priority': self.priority,
            'created_at': self.created_at.isoformat(),
            'status': self.status,
            'attempts': self.attempts,
            'max_retries': self.max_retries,
            'retry_delay': self.retry_delay,
            'result': self.result,
            'error': self.error,
            'started_at': self.started_at.isoformat() if self.started_at else None,
            'completed_at': self.completed_at.isoformat() if self.completed_at else None,
        }

    @classmethod
    def from_dict(cls, data: dict) -> 'Job':
        """Create a Job from a dictionary."""
        return cls(
            id=data['id'],
            func=cls._get_function(data['func']),
            args=data['args'],
            kwargs=data['kwargs'],
            priority=data['priority'],
            status=data['status'],
            attempts=data['attempts'],
            max_retries=data['max_retries'],
            retry_delay=data['retry_delay'],
            result=data['result'],
            error=data['error'],
        )

    @staticmethod
    def _get_function(func_name: str) -> Callable:
        """Resolve function by name (simplified for demo)."""
        from queue import JobQueue
        return JobQueue._registry.get(func_name, lambda: None)


class JobQueue:
    """Main job queue managing job lifecycle."""

    def __init__(self, max_retries: int = 3):
        self._jobs: Dict[str, Job] = {}
        self._max_retries = max_retries
        self._registry: Dict[str, Callable] = {}

    def register_function(self, name: str, func: Callable):
        """Register a function for later use in jobs."""
        self._registry[name] = func

    def add_job(self, func: Callable, *args, **kwargs) -> Job:
        """Add a new job to the queue."""
        job_id = f"{func.__name__}_{datetime.now().timestamp()}"
        job = Job(
            id=job_id,
            func=func,
            args=args,
            kwargs=kwargs,
            max_retries=self._max_retries,
        )
        self._jobs[job_id] = job
        return job

    def get_job(self, job_id: str) -> Optional[Job]:
        """Retrieve a job by ID."""
        return self._jobs.get(job_id)

    def get_pending_jobs(self) -> List[Job]:
        """Get all pending jobs."""
        return [job for job in self._jobs.values() if job.status == "pending"]

    def update_job_status(self, job_id: str, status: str):
        """Update the status of a job."""
        if job_id in self._jobs:
            self._jobs[job_id].status = status

    def remove_job(self, job_id: str):
        """Remove a job from the queue."""
        self._jobs.pop(job_id, None)

    def clear_completed(self):
        """Remove all completed jobs."""
        self._jobs = {
            k: v for k, v in self._jobs.items()
            if v.status not in ["completed", "failed"]
        }

    def job_exists(self, job_id: str) -> bool:
        """Check if a job exists."""
        return job_id in self._jobs
```

"""Retry logic with exponential backoff."""
import time
import random
from typing import Callable, Any


class RetryHandler:
    """Handles retry logic with exponential backoff."""

    def __init__(self, max_retries: int = 3, base_delay: float = 1.0, max_delay: float = 60.0):
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.max_delay = max_delay

    def calculate_delay(self, attempt: int) -> float:
        """Calculate delay using exponential backoff with jitter."""
        delay = min(self.base_delay * (2 ** (attempt - 1)), self.max_delay)
        jitter = delay * random.uniform(0.5, 1.5)
        return jitter

    def should_retry(self, attempt: int) -> bool:
        """Check if retry is allowed."""
        return attempt < self.max_retries

    def execute_with_retry(self, func: Callable, *args, **kwargs) -> tuple:
        """Execute function with retry logic."""
        attempt = 0
        while attempt <= self.max_retries:
            try:
                result = func(*args, **kwargs)
                return result, None
            except Exception as e:
                attempt += 1
                if self.should_retry(attempt):
                    delay = self.calculate_delay(attempt)
                    time.sleep(delay)
                else:
                    return None, str(e)
        return None, "Max retries exceeded"

"""Priority queue implementation."""
import heapq
from dataclasses import dataclass, field
from typing import List, Optional


@dataclass(order=True)
class PriorityEntry:
    """Wrapper for priority queue items."""
    priority: int
    timestamp: float = field(compare=False)
    job_id: str = field(compare=False)


class PriorityQueue:
    """Thread-safe priority queue for job scheduling."""

    def __init__(self):
        self._heap: List[PriorityEntry] = []
        self._counter = 0

    def push(self, job_id: str, priority: int):
        """Add a job to the queue."""
        entry = PriorityEntry(priority, self._counter, job_id)
        heapq.heappush(self._heap, entry)
        self._counter += 1

    def pop(self) -> Optional[str]:
        """Remove and return the highest priority job ID."""
        if not self._heap:
            return None
        entry = heapq.heappop(self._heap)
        return entry.job_id

    def peek(self) -> Optional[str]:
        """Return the highest priority job ID without removing it."""
        if not self._heap:
            return None
        return self._heap[0].job_id

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return len(self._heap) == 0

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._heap)

    def clear(self):
        """Remove all items from the queue."""
        self._heap.clear()
```

```python
from dataclasses import dataclass, field
from typing import Any, Callable, Optional, Union
import time
import uuid


@dataclass
class Job:
    """Represents a job in the queue."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    name: str = ""
    func: Optional[Callable] = None
    args: tuple = field(default_factory=tuple)
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    status: str = "pending"
    created_at: float = field(default_factory=time.time)
    started_at: Optional[float] = None
    completed_at: Optional[float] = None
    retries: int = 0
    max_retries: int = 3
    error: Optional[str] = None
    result: Any = None

    def __post_init__(self):
        if not self.id:
            self.id = str(uuid.uuid4())

    def to_dict(self) -> dict:
        return {
            "id": self.id,
            "name": self.name,
            "status": self.status,
            "priority": self.priority,
            "created_at": self.created_at,
            "started_at": self.started_at,
            "completed_at": self.completed_at,
            "retries": self.retries,
            "max_retries": self.max_retries,
            "error": self.error,
            "result": self.result,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Job":
        return cls(
            id=data.get("id", str(uuid.uuid4())),
            name=data.get("name", ""),
            func=data.get("func"),
            args=data.get("args", ()),
            kwargs=data.get("kwargs", {}),
            priority=data.get("priority", 0),
            status=data.get("status", "pending"),
            created_at=data.get("created_at", time.time()),
            started_at=data.get("started_at"),
            completed_at=data.get("completed_at"),
            retries=data.get("retries", 0),
            max_retries=data.get("max_retries", 3),
            error=data.get("error"),
            result=data.get("result"),
        )


class JobQueue:
    """Basic job queue with FIFO operations."""

    def __init__(self, max_size: Optional[int] = None):
        self._queue: list[Job] = []
        self._max_size = max_size

    def enqueue(self, job: Union[Job, Job]) -> str:
        if self._max_size and len(self._queue) >= self._max_size:
            raise RuntimeError("Queue is full")
        if isinstance(job, dict):
            job = Job.from_dict(job)
        self._queue.append(job)
        return job.id

    def dequeue(self) -> Optional[Job]:
        if not self._queue:
            return None
        return self._queue.pop(0)

    def peek(self) -> Optional[Job]:
        return self._queue[0] if self._queue else None

    def size(self) -> int:
        return len(self._queue)

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def clear(self):
        self._queue.clear()

    def get_all(self) -> list[Job]:
        return self._queue.copy()

    def remove(self, job_id: str) -> bool:
        for i, job in enumerate(self._queue):
            if job.id == job_id:
                self._queue.pop(i)
                return True
        return False
```

```python
from .core import Job, JobQueue
from typing import Optional
import heapq


class PriorityQueue(JobQueue):
    """Priority queue implementation using min-heap."""

    def __init__(self, max_size: Optional[int] = None):
        super().__init__(max_size)
        self._heap: list[tuple[int, float, Job]] = []
        self._counter = 0

    def enqueue(self, job: Job) -> str:
        if self._max_size and len(self._heap) >= self._max_size:
            raise RuntimeError("Queue is full")
        entry = (job.priority, self._counter, job)
        self._counter += 1
        heapq.heappush(self._heap, entry)
        return job.id

    def dequeue(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional[Job]:
        return self._heap[0][2] if self._heap else None

    def size(self) -> int:
        return len(self._heap)

    def is_empty(self) -> bool:
        return len(self._heap) == 0

    def clear(self):
        self._heap.clear()

    def get_all(self) -> list[Job]:
        return [item[2] for item in sorted(self._heap)]

    def update_priority(self, job_id: str, new_priority: int) -> bool:
        for i, (priority, _, job) in enumerate(self._heap):
            if job.id == job_id:
                self._heap[i] = (new_priority, self._counter, job)
                self._counter += 1
                self._heap[i] = (new_priority, self._counter, job)
                heapq.heapify(self._heap)
                return True
        return False
```

### `queue/retry.py`
```python
import time
import random
from typing import Callable, Any, Optional
from .core import Job


def calculate_backoff(retry_count: int, base_delay: float = 1.0,
                      max_delay: float = 60.0, exponential: bool = True) -> float:
    if exponential:
        delay = base_delay * (2 ** retry_count)
    else:
        delay = base_delay * (retry_count + 1)
    return min(delay, max_delay)


def process_with_retry(job: Job, func: Callable,
                       base_delay: float = 1.0, max_delay: float = 60.0,
                       exponential: bool = True,
                       on_failure: Optional[Callable[[Job, Exception], None]] = None,
                       on_success: Optional[Callable[[Job, Any], None]] = None) -> Optional[Any]:
    attempt = 0
    while attempt <= job.max_retries:
        try:
            job.started_at = time.time()
            result = func(*job.args, **job.kwargs)
            job.status = "completed"
            job.completed_at = time.time()
            job.result = result
            if on_success:
                on_success(job, result)
            return result
        except Exception as e:
            job.error = str(e)
            job.retries = attempt + 1
            if attempt < job.max_retries:
                delay = calculate_backoff(attempt, base_delay, max_delay, exponential)
                time.sleep(delay)
                attempt += 1
            else:
                job.status = "failed"
                job.completed_at = time.time()
                if on_failure:
                    on_failure(job, e)
                return None
    return None


def exponential_backoff_retry(func: Callable, base_delay: float = 1.0,
                              max_delay: float = 60.0, max_retries: int = 3) -> Callable:
    def wrapper(*args, **kwargs) -> Any:
        attempt = 0
        while attempt <= max_retries:
            try:
                return func(*args, **kwargs)
            except Exception as e:
                if attempt < max_retries:
                    delay = calculate_backoff(attempt, base_delay, max_delay, exponential=True)
                    time.sleep(delay)
                attempt += 1
        return None
    return wrapper
```

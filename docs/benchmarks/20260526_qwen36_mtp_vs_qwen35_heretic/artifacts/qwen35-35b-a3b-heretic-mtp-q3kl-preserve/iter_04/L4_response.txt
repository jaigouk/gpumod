        # Initialize a lock to protect shared state
        self.lock = threading.Lock()

    def add_job(self, job_id, data):
        with self.lock:
            self.jobs[job_id] = data
        return job_id

    def process_job(self, job_id, processor):
        # Use lock to atomically check existence and remove the job
        # This prevents multiple threads from processing the same job_id
        with self.lock:
            if job_id not in self.jobs:
                return False
            # Pop removes the job immediately, ensuring it cannot be processed again
            data = self.jobs.pop(job_id)

        # Process the job outside the lock to improve concurrency
        result = processor(data)

        # Use lock to safely write the result
        with self.lock:
            self.results[job_id] = result

        return True

    def get_result(self, job_id):
        with self.lock:
            return self.results.get(job_id)

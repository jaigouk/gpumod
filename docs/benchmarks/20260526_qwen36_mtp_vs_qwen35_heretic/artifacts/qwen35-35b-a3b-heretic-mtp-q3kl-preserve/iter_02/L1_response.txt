import collections

class JobQueue:
    def __init__(self):
        self._queue = collections.deque()
        self._results = {}
        self._processed = False

    def add_job(self, job_id: str, data: dict) -> str:
        self._queue.append((job_id, data))
        return job_id

    def process(self):
        while self._queue:
            job_id, data = self._queue.popleft()

    def get_result(self, job_id: str) -> dict | None:

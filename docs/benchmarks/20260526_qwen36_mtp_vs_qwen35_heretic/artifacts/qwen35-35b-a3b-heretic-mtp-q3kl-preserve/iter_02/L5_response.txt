```python
"""
Core job queue functionality: Job dataclass and basic queue operations.
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Dict, List, Optional
from datetime import datetime, timedelta
import uuid
import json
import logging

logger = logging.getLogger(__name__)

@dataclass
class Job:
    """
    Represents a job to be executed.
    """
    id: str
    name: str
    func: Callable
    args: tuple = field(default_factory=tuple)
    kwargs: dict = field(default_factory=dict)
    priority: int = 0
    status: str = "pending"  # pending, running, completed, failed, retrying
    created_at: datetime = field(default_factory=datetime.now)
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    error: Optional[str] = None
    retries: int = 0
    max_retries: int = 3
    metadata: dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        return {
            'id': self.id,
            'name': self.name,
            'func_name': self.func.__name__,
            'args': self.args,
            'kwargs': self.kwargs,
            'priority': self.priority,
            'status': self.status,
            'created_at': self.created_at.isoformat(),
            'started_at': self.started_at.isoformat() if self.started_at else None,
            'completed_at': self.completed_at.isoformat() if self.completed_at else None,
            'error': self.error,
            'retries': self.retries,
            'max_retries': self.max_retries,
            'metadata': self.metadata
        }

    @classmethod
    def from_dict(cls, data: dict) -> Job:
        return cls(
            id=data['id'],
            name=data['name'],
            func=cls._get_func(data['func_name']),
            args=data.get('args', ()),
            kwargs=data.get('kwargs', {}),
            priority=data.get('priority', 0),
            status=data.get('status', 'pending'),
            created_at=datetime.fromisoformat(data['created_at']),
            started_at=datetime.fromisoformat(data['started_at']) if data.get('started_at') else None,
            completed_at=datetime.fromisoformat(data['completed_at']) if data.get('completed_at') else None,
            error=data.get('error'),
            retries=data.get('retries', 0),
            max_retries=data.get('max_retries', 3),
            metadata=data.get('metadata', {})
        )

    @staticmethod
    def _get_func(func_name: str) -> Callable:
        from __main__ import job_functions
        return job_functions.get(func_name, lambda *args, **kwargs: None)

    @classmethod
    def create(cls, name: str, func: Callable, *args, **kwargs) -> Job:
        return cls(
            id=str(uuid.uuid4()),
            name=name,
            func=func,
            args=args,
            kwargs=kwargs,
            priority=0
        )

class JobQueue:
    """
    Basic job queue implementation.
    """
    def __init__(self, max_size: int = 10000):
        self._jobs: Dict[str, Job] = {}
        self._queue: List[str] = []
        self.max_size = max_size
        self._lock = None

    def add_job(self, job: Job) -> None:
        if len(self._jobs) >= self.max_size:
            raise Exception("Queue is full")
        self._jobs[job.id] = job
        self._queue.append(job.id)
        logger.info(f"Job {job.id} added to queue")

    def get_next_job(self) -> Optional[Job]:
        if not self._queue:
            return None
        job_id = self._queue.pop(0)
        job = self._jobs.get(job_id)
        if job:
            job.status = "running"
            job.started_at = datetime.now()
        return job

    def complete_job(self, job_id: str, error: str = None) -> None:
        if job_id in self._jobs:
            job = self._jobs[job_id]
            job.status = "completed"
            job.completed_at = datetime.now()
            job.error = error
            if job_id in self._queue:
                self._queue.remove(job_id)
            logger.info(f"Job {job_id} completed")

    def remove_job(self, job_id: str) -> None:
        if job_id in self._jobs:
            del self._jobs[job_id]
        if job_id in self._queue:
            self._queue.remove(job_id)

    def get_job(self, job_id: str) -> Optional[Job]:
        return self._jobs.get(job_id)

    def get_all_jobs(self) -> List[Job]:
        return list(self._jobs.values())

    def get_pending_jobs(self) -> List[Job]:
        return [job for job in self._jobs.values() if job.status == "pending"]

    def clear(self) -> None:
        self._jobs.clear()
        self._queue.clear()
```

```python
"""
Priority queue implementation for job scheduling.
"""
import heapq
from typing import List, Optional
from datetime import datetime
import logging

from .core import Job

logger = logging.getLogger(__name__)

class PriorityQueue:
    """
    Priority queue for job execution based on priority.
    """
    def __init__(self):
        self._heap: List[tuple] = []
        self._counter = 0
        self._jobs: dict = {}

    def push(self, job: Job) -> None:
        entry = (-job.priority, self._counter, job.id)
        self._counter += 1
        heapq.heappush(self._heap, entry)
        self._jobs[job.id] = job
        logger.debug(f"Job {job.id} pushed to priority queue with priority {job.priority}")

    def pop(self) -> Optional[Job]:
        while self._heap:
            _, _, job_id = heapq.heappop(self._heap)
            if job_id in self._jobs:
                job = self._jobs[job_id]
                del self._jobs[job_id]
                return job
        return None

    def peek(self) -> Optional[Job]:
        if not self._heap:
            return None
        _, _, job_id = self._heap[0]
        return self._jobs.get(job_id)

    def __len__(self) -> int:
        return len(self._jobs)

    def is_empty(self) -> bool:
        return len(self._jobs) == 0

    def update_priority(self, job_id: str, new_priority: int) -> bool:
        if job_id in self._jobs:
            job = self._jobs[job_id]
            job.priority = new_priority
            self._heap = []
            for j in self._jobs.values():
                heapq.heappush(self._heap, (-j.priority, self._counter, j.id))
            return True
        return False
```

```python
"""
Retry logic with exponential backoff.
"""
import time
import random
import logging
from typing import Callable, Optional, Any, List
from datetime import datetime
import json

logger = logging.getLogger(__name__)

def calculate_backoff(attempt: int, base_delay: float = 1.0, max_delay: float = 60.0, multiplier: float = 2.0) -> float:
    delay = min(base_delay * (multiplier ** attempt), max_delay)
    jitter = random.uniform(0, delay * 0.1)
    return delay + jitter

def process_with_retry(func: Callable, max_retries: int = 3, base_delay: float = 1.0, **kwargs) -> Any:
    attempt = 0
    last_error = None

    while attempt <= max_retries:
        try:
            return func(*kwargs.get('args', ()), **kwargs.get('kwargs', {}))
        except Exception as e:
            last_error = str(e)
            attempt += 1
            if attempt <= max_retries:
                delay = calculate_backoff(attempt - 1, base_delay)
                logger.warning(f"Job failed (attempt {attempt}/{max_retries}): {e}. Retrying in {delay:.2f}s")
                time.sleep(delay)
            else:
                logger.error(f"Job failed permanently after {max_retries} attempts: {e}")
                raise
    return None

class RetryHandler:
    def __init__(self, base_delay: float = 1.0, max_retries: int = 3, max_delay: float = 60.0):
        self.base_delay = base_delay
        self.max_retries = max_retries
        self.max_delay = max_delay

    def should_retry(self, attempt: int) -> bool:
        return attempt < self.max_retries

    def get_delay(self, attempt: int) -> float:
        return calculate_backoff(attempt, self.base_delay, self.max_delay)

    def execute_with_retry(self, func: Callable, *args, **kwargs) -> Any:
        attempt = 0
        last_error = None
        while attempt <= self.max_retries:
            try:
                return func(*args, **kwargs)
            except Exception as e:
                last_error = str(e)
                attempt += 1
                if self.should_retry(attempt - 1):
                    delay = self.get_delay(attempt - 1)
                    logger.warning(f"Retry {attempt}/{self.max_retries}, waiting {delay:.2f}s: {e}")
                    time.sleep(delay)
                else:
                    logger.error(f"Failed after {self.max_retries} attempts")
                    raise
        return None
```

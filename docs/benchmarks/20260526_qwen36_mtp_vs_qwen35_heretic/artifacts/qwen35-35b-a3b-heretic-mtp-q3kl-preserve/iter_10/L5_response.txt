@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    func: Callable
    payload: Any = None
    priority: int = 0
    status: str = "pending"
    retry_count: int = 0

    def __hash__(self):

    def __eq__(self, other):
        if isinstance(other, Job):
            return self.id == other.id
        return False

class JobQueue:
    """
    Manages the lifecycle of jobs using a priority queue.
    """
    def __init__(self):
        # Initialize the internal priority storage
        self._queue = PriorityQueue()
        self._processed_ids = set()

    def add_job(self, job: "Job") -> None:
        """Add a job to the queue."""
        self._queue.push(job)

    def get_next_job(self) -> Optional["Job"]:
        """Retrieve the highest priority job."""
        job = self._queue.pop()
        if job:
            job.status = "processing"
        return job

    def mark_complete(self, job_id: str) -> None:
        """Mark a job as successfully processed."""
        self._processed_ids.add(job_id)

    def is_processed(self, job_id: str) -> bool:
        """Check if a job has already been processed."""
        return job_id in self._processed_ids

    def clear_processed(self) -> None:
        """Clear the set of processed job IDs."""
        self._processed_ids.clear()

### 3. `queue/priority.py`
Handles the heap-based priority queue implementation. It imports `Job` only for type hinting to avoid circular dependencies.

```python
import heapq
from typing import TYPE_CHECKING, Optional

# Use TYPE_CHECKING to avoid circular import at runtime
if TYPE_CHECKING:
    from .core import Job

class PriorityQueue:
    """
    A min-heap based priority queue.
    Lower numbers indicate higher priority.
    """
    def __init__(self):
        self._heap = []
        self._counter = 0

    def push(self, job: "Job") -> None:
        """
        Push a job onto the queue.
        Tuple format: (priority, counter, job)
        Counter ensures stable sorting for equal priorities.
        """
        entry = (job.priority, self._counter, job)
        heapq.heappush(self._heap, entry)
        self._counter += 1

    def pop(self) -> Optional["Job"]:
        """
        Remove and return the highest priority job.
        Returns None if the queue is empty.
        """
        if not self._heap:
            return None
        _, _, job = heapq.heappop(self._heap)
        return job

    def peek(self) -> Optional["Job"]:
        """Return the highest priority job without removing it."""
        if not self._heap:
            return None
        return self._heap[0][2]

    def __len__(self) -> int:
        return len(self._heap)
```

import time
import math
from typing import Any, Callable, TypeVar

T = TypeVar('T')

def process_with_retry(
    job, 
    func: Callable, 
    max_retries: int = 3, 
    base_delay: float = 1.0,
    max_delay: float = 60.0
) -> Any:
    """
    Executes a function with exponential backoff retry logic.
    
    Args:
        job: The Job object (used for logging/context).
        func: The function to execute.
        max_retries: Maximum number of attempts.
        base_delay: Initial delay in seconds.
        max_delay: Maximum delay cap in seconds.
        
    Returns:
        The result of the function execution.
    """
    attempts = 0
    last_error = None

    while attempts < max_retries:
        try:
            # Execute the function
            return func()
        except Exception as e:
            attempts += 1
            last_error = e
            
            if attempts < max_retries:
                # Calculate exponential backoff
                delay = min(base_delay * (2 ** (attempts - 1)), max_delay)
                print(f"Job {job.id} failed (attempt {attempts}/{max_retries}), retrying in {delay:.2f}s")
                time.sleep(delay)
            else:
                print(f"Job {job.id} failed permanently after {attempts} attempts.")

    # If we exit the loop, all retries were exhausted
    raise last_error

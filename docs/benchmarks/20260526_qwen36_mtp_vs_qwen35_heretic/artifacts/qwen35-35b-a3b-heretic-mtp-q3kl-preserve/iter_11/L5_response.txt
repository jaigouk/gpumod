### `queue/core.py`

```python
from dataclasses import dataclass, field
from typing import Callable, Any, Optional, List
import time
import uuid

@dataclass
class Job:
    """Represents a job to be processed."""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    name: str = ""
    payload: Any = None
    priority: int = 0
    created_at: float = field(default_factory=time.time)
    status: str = "pending"
    attempt: int = 0
    max_attempts: int = 3
    callback: Optional[Callable] = None
    error: Optional[str] = None

    def __post_init__(self):
        if not self.id:
            self.id = str(uuid.uuid4())
        if not self.status:
            self.status = "pending"

    def mark_started(self):
        self.status = "processing"
        self.attempt += 1

    def mark_completed(self):
        self.status = "completed"
        self.error = None

    def mark_failed(self, error: str):
        self.status = "failed"
        self.error = error

    def can_retry(self) -> bool:
        return self.attempt < self.max_attempts and self.status in ["pending", "processing", "failed"]
```

### `queue/priority.py`

```python
from .core import Job
import heapq
from typing import List, Optional
import time

class PriorityQueue:
    """Thread-safe priority queue using heapq."""

    def __init__(self):
        self._queue: List[Job] = []
        self._counter = 0

    def push(self, job: Job):
        entry = (job.priority, self._counter, job)
        heapq.heappush(self._queue, entry)
        self._counter += 1

    def pop(self, timeout: float = 0) -> Optional[Job]:
        start = time.time()
        while True:
            try:
                _, _, job = heapq.heappop(self._queue)
                job.mark_started()
                return job
            except IndexError:
                if timeout == 0:
                    return None
                if time.time() - start >= timeout:
                    return None
                time.sleep(0.01)

    def peek(self) -> Optional[Job]:
        if not self._queue:
            return None
        return self._queue[0][2]

    def size(self) -> int:
        return len(self._queue)

    def is_empty(self) -> bool:
        return len(self._queue) == 0

    def clear(self):
        self._queue.clear()
```

### `queue/retry.py`

```python
import time
import random
from .core import Job

def exponential_backoff(attempt: int, base_delay: float = 1.0, max_delay: float = 60.0, jitter: bool = True) -> float:
    delay = min(base_delay * (2 ** attempt), max_delay)
    if jitter:
        delay = delay * (0.5 + random.random())
    return delay

def process_with_retry(job: Job, func: callable, max_attempts: int = None, base_delay: float = 1.0) -> bool:
    if max_attempts is None:
        max_attempts = job.max_attempts

    for attempt in range(max_attempts):
        try:
            job.mark_started()
            func(job)
            job.mark_completed()
            return True
        except Exception as e:
            job.mark_failed(str(e))
            if attempt < max_attempts - 1:
                delay = exponential_backoff(attempt, base_delay)
                time.sleep(delay)
            else:
                return False
    return False
```

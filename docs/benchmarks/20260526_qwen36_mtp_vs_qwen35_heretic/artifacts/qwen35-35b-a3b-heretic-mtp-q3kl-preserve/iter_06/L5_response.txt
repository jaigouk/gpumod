### 2. `queue/core.py`
Contains the core data structures (`Job`) and the basic queue management logic (`JobQueue`). This module knows nothing about priority or retry logic.

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Optional, List
import uuid

@dataclass
class Job:
    """Represents a unit of work to be processed."""
    id: str
    payload: Any
    created_at: datetime = field(default_factory=datetime.now)
    status: str = "pending"  # pending, processing, completed, failed
    attempts: int = 0

    @classmethod
    def create(cls, payload: Any) -> "Job":
        return cls(id=str(uuid.uuid4()), payload=payload)

class JobQueue:
    """Basic FIFO queue for managing jobs."""

    def __init__(self):
        self._jobs: List[Job] = []
        self._index = 0

    def add(self, job: Job) -> None:
        """Add a job to the end of the queue."""
        self._jobs.append(job)

    def pop(self) -> Optional[Job]:
        """Remove and return the next job from the queue."""
        if self._jobs:
            return self._jobs.pop(0)
        return None

    def peek(self) -> Optional[Job]:
        """Return the next job without removing it."""
        if self._jobs:
            return self._jobs[0]
        return None

    def size(self) -> int:
        """Return the number of jobs in the queue."""
        return len(self._jobs)

    def clear(self) -> None:
        """Remove all jobs from the queue."""
        self._jobs.clear()
```

### 3. `queue/priority.py`
Implements priority-specific logic. It inherits from the base `JobQueue` to reuse basic operations but overrides how items are ordered.

```python
from typing import Any, Optional
import heapq

from .core import Job, JobQueue

class PriorityQueue(JobQueue):
    """A priority-based queue implementation."""

    def __init__(self):
        super().__init__()
        self._heap: list = []

    def add(self, job: Job, priority: int = 0) -> None:
        """Add a job with a specific priority (lower number = higher priority)."""
        # We store (priority, job) to maintain heap property
        heapq.heappush(self._heap, (priority, job))

    def pop(self) -> Optional[Job]:
        """Remove and return the job with the highest priority."""
        if self._heap:
            priority, job = heapq.heappop(self._heap)
            return job
        return None

    def peek(self) -> Optional[Job]:
        """Return the job with the highest priority without removing it."""
        if self._heap:
            return self._heap[0][1]
        return None

    def size(self) -> int:
        return len(self._heap)

    def clear(self) -> None:
        self._heap.clear()
```

### 4. `queue/retry.py`
Contains the execution logic, handling backoff strategies and retry attempts. It depends on `Job` and `JobQueue` from `core.py`.

```python
import time
import random
from typing import Callable, Any

from .core import Job, JobQueue

def process_with_retry(
    job: Job,
    func: Callable[[Any], Any],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0
) -> bool:
    """
    Process a job with exponential backoff retry logic.
    
    Args:
        job: The job to process.
        func: The function to execute.
        max_retries: Maximum number of retry attempts.
        base_delay: Initial delay in seconds.
        max_delay: Maximum delay cap in seconds.
        
    Returns:
        True if successful, False if max retries exceeded.
    """
    attempt = 0
    current_delay = base_delay

    while attempt <= max_retries:
        try:
            result = func(job.payload)
            job.status = "completed"
            job.attempts = attempt
            return True
        except Exception as e:
            job.attempts = attempt + 1
            job.status = "failed"
            
            if attempt == max_retries:
                return False
            
            # Exponential backoff with jitter
            sleep_time = min(current_delay * (2 ** attempt) + random.uniform(0, 0.5), max_delay)
            time.sleep(sleep_time)
            
            attempt += 1
            
    return False
```

### Key Refactoring Decisions

1.  **Dependency Flow**: `core.py` is the foundation. `priority.py` and `retry.py` depend on `core.py`, but `core.py` does not depend on them. This prevents circular imports.
2.  **Inheritance**: `PriorityQueue` extends `JobQueue`. This allows the base functionality (like `add`, `pop`) to be shared while overriding behavior where necessary (heap management).
3.  **Encapsulation**: `Job` is defined in `core.py` so it is available to both the queue implementations and the retry logic without duplication.
4.  **Public API**: `__init__.py` acts as the facade, allowing users to write `from queue import Job, JobQueue` instead of navigating internal modules.

```python
import heapq

class JobQueue:
    def __init__(self):
        self._heap = []
        self._counter = 0

    def add_job(self, name: str, payload: dict, priority: int = 0) -> None:
        heapq.heappush(self._heap, (-priority, self._counter, name, payload))
        self._counter += 1

    def get_next_job(self) -> tuple[str, dict] | None:
        if self._heap:
            _, _, name, payload = heapq.heappop(self._heap)
            return name, payload
        return None
```

    def __init__(self):

        if job_id not in self.jobs:

        data = job_record['data']
        delays = job_record['delays']
        
        backoff_delays = [1, 2, 4]
        
        for attempt in range(4):
            try:
                processor(data)
                return True
            except Exception:
                if attempt < 3:
                    delays.append(backoff_delays[attempt])
        
        return False

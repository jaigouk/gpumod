    def __init__(self):

    def add_job(self, job_id: str, data: dict):
        self.jobs[job_id] = data

    def process_job(self, job_id: str, processor: Callable) -> bool:

        if data is None:
            return False

        max_attempts = 4

        for attempt in range(max_attempts):

                return True

                if attempt < max_attempts - 1:
                    delay = backoffs[attempt]

        return False

from collections import deque

class JobQueue:
    def __init__(self):
        self.queue = deque()
        self.results = {}
        self.processed_ids = set()

    def add_job(self, job_id: str, data: dict) -> str:
        if job_id in self.processed_ids:
            raise ValueError(f"Job ID '{job_id}' already processed or exists.")
        self.queue.append({'job_id': job_id, 'data': data, 'status': 'pending'})
        return job_id

    def process(self):
        while self.queue:
            job = self.queue.popleft()
            job['status'] = 'completed'
            self.results[job['job_id']] = job
            self.processed_ids.add(job['job_id'])

    def get_result(self, job_id: str) -> dict | None:
        return self.results.get(job_id)
